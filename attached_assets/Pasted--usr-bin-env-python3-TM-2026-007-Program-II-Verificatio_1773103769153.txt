# ═══════════════════════════════════════════
# GF(27) ARITHMETIC
# ═══════════════════════════════════════════
# Elements are triples (a0, a1, a2) representing a0 + a1*t + a2*t^2
# where t^3 = -2t - 1 = t + 2 (in GF(3), -2 = 1, -1 = 2)
# So t^3 = t + 2

def gf27_add(a, b):
    return tuple((ai + bi) % 3 for ai, bi in zip(a, b))

def gf27_neg(a):
    return tuple((3 - ai) % 3 for ai in a)

def gf27_sub(a, b):
    return gf27_add(a, gf27_neg(b))

def gf27_mul(a, b):
    # (a0 + a1*t + a2*t^2)(b0 + b1*t + b2*t^2)
    # expand and reduce t^3 = t + 2, t^4 = t^2 + 2t
    a0, a1, a2 = a
    b0, b1, b2 = b
    
    # Direct expansion coefficients for t^0, t^1, t^2, t^3, t^4
    c0 = a0*b0
    c1 = a0*b1 + a1*b0
    c2 = a0*b2 + a1*b1 + a2*b0
    c3 = a1*b2 + a2*b1
    c4 = a2*b2
    
    # Reduce: t^3 = t + 2, so t^4 = t*t^3 = t(t+2) = t^2 + 2t
    # c3 * t^3 = c3*(t + 2) = 2*c3 + c3*t
    # c4 * t^4 = c4*(t^2 + 2t) = 2*c4*t + c4*t^2
    r0 = (c0 + 2*c3) % 3
    r1 = (c1 + c3 + 2*c4) % 3
    r2 = (c2 + c4) % 3
    
    return (r0, r1, r2)

def gf27_pow(a, n):
    if n == 0:
        return (1, 0, 0)
    if a == (0, 0, 0):
        return (0, 0, 0)
    result = (1, 0, 0)
    base = a
    while n > 0:
        if n % 2 == 1:
            result = gf27_mul(result, base)
        base = gf27_mul(base, base)
        n //= 2
    return result

def elem_to_int(a):
    return a[0] + 3*a[1] + 9*a[2]

def int_to_elem(n):
    return (n % 3, (n // 3) % 3, (n // 9) % 3)

# Generate all 27 elements
ALL_ELEMS = [int_to_elem(i) for i in range(27)]
ZERO = (0, 0, 0)
ONE = (1, 0, 0)

# Verify field structure: check t^3 + 2t + 1 = 0
t = (0, 1, 0)  # the element t
t3 = gf27_pow(t, 3)
check = gf27_add(t3, gf27_add(gf27_mul((2,0,0), t), ONE))
print(f"Irreducibility check: t^3 + 2t + 1 = {check} (should be (0,0,0))")

# Verify x^26 = 1 for all nonzero x (multiplicative group order = 26)
all_orders_ok = True
for x in ALL_ELEMS:
    if x != ZERO:
        if gf27_pow(x, 26) != ONE:
            all_orders_ok = False

# ═══════════════════════════════════════════
# DDT COMPUTATION
# ═══════════════════════════════════════════
def compute_ddt(exponent):
    """Compute DDT for chi(x) = x^exponent over GF(27)"""
    # First check it's a permutation
    images = set()
    chi = {}
    for x in ALL_ELEMS:
        y = gf27_pow(x, exponent)
        chi[x] = y
        images.add(y)
    
    is_perm = len(images) == 27
    
    # DDT[a][b] = |{x : chi(x+a) - chi(x) = b}| for a != 0
    ddt = {}
    max_entry = 0
    max_pair = None
    
    for a in ALL_ELEMS:
        if a == ZERO:
            continue
        ddt[a] = {}
        for b in ALL_ELEMS:
            ddt[a][b] = 0
        
        for x in ALL_ELEMS:
            xa = gf27_add(x, a)
            diff = gf27_sub(chi[xa], chi[x])
            ddt[a][diff] += 1
            if ddt[a][diff] > max_entry:
                max_entry = ddt[a][diff]
                max_pair = (a, diff)
    
    return is_perm, max_entry, max_pair, ddt

def theta_multiply(a):

    return tuple(result)

def hamming_weight(v):
    return sum(1 for x in v if x != 0)
