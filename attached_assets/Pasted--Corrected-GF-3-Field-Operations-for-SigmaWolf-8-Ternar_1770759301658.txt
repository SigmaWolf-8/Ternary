/**
 * Corrected GF(3) Field Operations for SigmaWolf-8/Ternary
 * 
 * This file provides the corrected GF(3) arithmetic operations
 * using the proper ring homomorphism mapping: toGF3(a) = ((a%3)+3)%3
 * 
 * Drop-in replacement for the arithmetic functions in:
 *   libternary/src/ternary-operations.ts
 * 
 * IMPORTANT: The representation conversion functions (ternaryAtoB, etc.)
 * are CORRECT as-is and should NOT be changed. Only the arithmetic
 * mapping needs this fix.
 * 
 * Bug reference: CONTRIBUTING.md notes that GF(3) arithmetic must use
 * ((a%3)+3)%3, not a+1
 */

// === Types (unchanged from repo) ===

export type TritA = -1 | 0 | 1;  // Computational representation (balanced ternary)
export type TritB = 0 | 1 | 2;   // Network representation
export type TritC = 1 | 2 | 3;   // Human/bijective representation

// === Representation Conversions (CORRECT - do not change) ===

export const ternaryAtoB = (a: TritA): TritB => (a + 1) as TritB;
export const ternaryAtoC = (a: TritA): TritC => (a + 2) as TritC;
export const ternaryBtoC = (b: TritB): TritC => (b + 1) as TritC;
export const ternaryBtoA = (b: TritB): TritA => (b - 1) as TritA;
export const ternaryCtoA = (c: TritC): TritA => (c - 2) as TritA;
export const ternaryCtoB = (c: TritC): TritB => (c - 1) as TritB;

// === GF(3) Arithmetic Mapping (FIXED) ===
// This is the proper ring homomorphism: balanced ternary {-1,0,+1} <-> GF(3) {2,0,1}
//
// Mapping:
//   -1  ->  2  (because ((-1 % 3) + 3) % 3 = 2)
//    0  ->  0  (because (( 0 % 3) + 3) % 3 = 0)  
//   +1  ->  1  (because (( 1 % 3) + 3) % 3 = 1)
//
// WHY NOT a+1? Because a+1 maps -1->0, 0->1, 1->2, which gives:
//   -1 * -1 = 0*0 = 0 -> maps back to -1  ... WRONG (should be +1)
//
// With ((a%3)+3)%3: -1->2, so 2*2 = 4%3 = 1 -> maps back to +1  ... CORRECT

export const toGF3 = (a: TritA): number => ((a % 3) + 3) % 3;

export const fromGF3 = (g: number): TritA => {
  const v = ((g % 3) + 3) % 3;
  if (v === 0) return 0;
  if (v === 1) return 1;
  return -1; // v === 2
};

// === GF(3) Arithmetic Operations ===

export const gf3Add = (a: TritA, b: TritA): TritA => {
  const aGF3 = toGF3(a);
  const bGF3 = toGF3(b);
  const resultGF3 = (aGF3 + bGF3) % 3;
  return fromGF3(resultGF3);
};

export const gf3Multiply = (a: TritA, b: TritA): TritA => {
  const aGF3 = toGF3(a);
  const bGF3 = toGF3(b);
  const resultGF3 = (aGF3 * bGF3) % 3;
  return fromGF3(resultGF3);
};

export const gf3Xor = (a: TritA, b: TritA): TritA => {
  return gf3Add(a, b); // In GF(3), XOR is equivalent to addition
};

export const gf3Not = (a: TritA): TritA => (-a) as TritA;

export const gf3Rotate = (value: TritA, steps: number = 1): TritA => {
  const gf3Val = toGF3(value);
  const rotated = (gf3Val + ((steps % 3) + 3) % 3) % 3;
  return fromGF3(rotated);
};

// === Representation A Operations (FIXED to use correct GF(3) mapping) ===

export const ternaryAdd = (a: TritA, b: TritA): TritA => {
  return gf3Add(a, b);
};

export const ternaryMultiply = (a: TritA, b: TritA): TritA => {
  return gf3Multiply(a, b);
};

export const ternaryNot = (value: TritA): TritA => (-value) as TritA;

// === GF(3) Operations on Representation B (unchanged, already correct) ===

export const GF3 = {
  add: (a: TritB, b: TritB): TritB => ((a + b) % 3) as TritB,
  subtract: (a: TritB, b: TritB): TritB => (((a - b) % 3 + 3) % 3) as TritB,
  multiply: (a: TritB, b: TritB): TritB => ((a * b) % 3) as TritB,
  negate: (a: TritB): TritB => ((3 - a) % 3) as TritB,
};

// === Quick Verification ===
// Run this to confirm the fix works:

export const verifyGF3 = (): boolean => {
  // Key test: -1 * -1 must equal +1 (not -1)
  const test1 = gf3Multiply(-1, -1) === 1;
  
  // Identity: a * 1 = a for all a
  const test2 = gf3Multiply(-1, 1) === -1;
  const test3 = gf3Multiply(0, 1) === 0;
  const test4 = gf3Multiply(1, 1) === 1;
  
  // Zero: a * 0 = 0 for all a
  const test5 = gf3Multiply(-1, 0) === 0;
  const test6 = gf3Multiply(1, 0) === 0;
  
  // Additive identity: a + 0 = a
  const test7 = gf3Add(-1, 0) === -1;
  const test8 = gf3Add(0, 0) === 0;
  const test9 = gf3Add(1, 0) === 1;
  
  // Additive inverse: a + (-a) wraps in GF(3)
  const test10 = gf3Add(1, -1) === 0;
  
  return test1 && test2 && test3 && test4 && test5 && 
         test6 && test7 && test8 && test9 && test10;
};
