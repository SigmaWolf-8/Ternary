Backend anomaly detection (simple version):

```ts
// in /stats endpoint
const dailyUsage = await db
  .select({
    date: sql`date_trunc('day', timestamp)`,
    key_id: apiKeyLogs.keyId,
    count: sql`count(*)`,
  })
  .from(apiKeyLogs)
  .groupBy(sql`date_trunc('day', timestamp)`, apiKeyLogs.keyId)
  .orderBy(apiKeyLogs.keyId, sql`date_trunc('day', timestamp)`);

const anomalies = [];
for (let i = 1; i < dailyUsage.length; i++) {
  const prev = dailyUsage[i-1];
  const curr = dailyUsage[i];
  if (curr.key_id === prev.key_id) {
    const pct = ((curr.count - prev.count) / prev.count) * 100;
    if (pct > 300 && curr.count > 50) {  // arbitrary thresholds
      anomalies.push({
        keyId: curr.key_id,
        date: curr.date,
        change: Math.round(pct),
        type: 'spike',
      });
    }
  }
}
```
