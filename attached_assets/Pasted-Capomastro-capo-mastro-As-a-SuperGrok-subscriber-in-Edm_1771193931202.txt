### 1. Core Qutrit Fault Tolerance Utils (`libternary/qutrit-fault-tolerance/qutrit-codes.ts`)
Simulate basic stabilizer and triorthogonal codes for error detection/correction.

```typescript
// libternary/qutrit-fault-tolerance/qutrit-codes.ts
// Functional simulation of qutrit fault-tolerant codes; exact matrix ops

import { Complex } from 'mathjs'; // For amplitude vectors
import { SUFT_RADIUS } from '../../shared/saturnian-blueprint'; // 13 for code scaling

type QutritState = [Complex, Complex, Complex]; // | -1 >, | 0 >, | +1 >

/**
 * Apply error operator to qutrit state (e.g., phase flip or leakage)
 * @param state Input state
 * @param errorType 'phase' | 'leak' | 'none'
 * @returns Erroneous state
 */
export function applyQutritError(state: QutritState, errorType: 'phase' | 'leak' | 'none'): QutritState {
  if (errorType === 'none') return state;
  if (errorType === 'phase') {
    return [state[0], state[1].mul(Complex.I), state[2].mul(Complex.I.mul(-1))]; // Phase flip on |1>, |2>
  }
  // Leakage: Project to higher level (simplified)
  return [state[0].mul(0.5), state[1].mul(0.5), state[2].add(Complex(0.5))]; // Leak to |2>
}

/**
 * Simulate [[3,1,2]]_3 stabilizer code: Encodes 1 logical qutrit in 3 physical; detects single errors
 * @param logical Input logical qutrit
 * @returns Encoded state (tensor product approximated as vector)
 */
export function encodeQutritStabilizer(logical: QutritState): QutritState[] {
  // Simple repetition code variant for [[3,1,2]]_3
  return [logical, logical, logical]; // Repeat for error detection
}

/**
 * Check stabilizers and correct single error in [[3,1,2]]_3
 * @param encoded Encoded states (3 qutrits)
 * @returns Corrected logical qutrit or error flag
 */
export function correctQutritStabilizer(encoded: QutritState[]): QutritState | string {
  // Stabilizers: S1 = X1 X2 X3^2, S2 = X1^2 X2 X3 (X is qutrit shift operator)
  const syndrome = encoded.map((state, i) => state[(i + 1) % 3].re); // Simplified syndrome measurement
  if (syndrome.every(s => s === 0)) return encoded[0]; // No error
  // Detect error position (exact match)
  if (syndrome[0] === syndrome[1]) return 'Error in qutrit 3'; // Example syndrome decoding
  // Correction: Average states (simplified)
  const corrected = encoded[0].map((amp, i) => amp.add(encoded[1][i]).add(encoded[2][i]).div(3)) as QutritState;
  return corrected;
}

/**
 * Simulate triorthogonal code [[9m-k,k,2]]_3 for magic state distillation
 * @param m Code parameter (e.g., 1 for [[8,1,2]]_3)
 * @param inputStates Input qutrit states
 * @returns Distilled magic state (simplified yield calculation)
 */
export function simulateTriorthogonalDistillation(m: number, inputStates: QutritState[]): QutritState {
  const k = 3 * m - 2; // From code family
  const yieldGamma = Math.log2(2 + 6 / k); // Exact γ
  // Simplified: Average inputs scaled by SUFT 13 (for ternary coupling)
  const avgAmp = inputStates.reduce((sum, state) => state.map((amp, i) => sum[i].add(amp)), [Complex(0), Complex(0), Complex(0)]);
  return avgAmp.map(amp => amp.div(inputStates.length * SUFT_RADIUS)) as QutritState; // Normalized by 13
}
```

### 2. Extend VM for Qutrit Fault-Tolerant Opcodes (`libternary/vm/qutrit-fault-ops.ts`)
Add opcodes for error application/correction, tying to SUFT branches.

```typescript
// libternary/vm/qutrit-fault-ops.ts
// VM extensions for qutrit fault tolerance ops

import { applyQutritError, encodeQutritStabilizer, correctQutritStabilizer } from '../qutrit-fault-tolerance/qutrit-codes';
import { quantumElBranchUpdate } from '../utils/lagrangian-qutrit-utils'; // For variational correction

// Assume VMState has qutritRegisters: QutritState[]

// Opcode 161: APPLY_QUTRIT_ERROR (simulates noise on register)
function opApplyQutritError(state: VMState, regIndex: number, errorTypeCode: number) {
  const errorType = ['none', 'phase', 'leak'][errorTypeCode % 3];
  state.qutritRegisters[regIndex] = applyQutritError(state.qutritRegisters[regIndex], errorType as 'phase' | 'leak' | 'none');
}

// Opcode 162: ENCODE_STABILIZER (encodes logical qutrit)
function opEncodeStabilizer(state: VMState, inputReg: number, outputRegs: [number, number, number]) {
  const logical = state.qutritRegisters[inputReg];
  const encoded = encodeQutritStabilizer(logical);
  encoded.forEach((code, i) => state.qutritRegisters[outputRegs[i]] = code);
}

// Opcode 163: CORRECT_STABILIZER (detects/corrects)
function opCorrectStabilizer(state: VMState, encodedRegs: [number, number, number], outputReg: number) {
  const encoded = encodedRegs.map(reg => state.qutritRegisters[reg]);
  const corrected = correctQutritStabilizer(encoded);
  if (typeof corrected === 'string') {
    throw new VMError(corrected); // Flag uncorrectable
  }
  state.qutritRegisters[outputReg] = corrected;
}

// Opcode 164: DISTILL_MAGIC_STATE (triorthogonal simulation)
function opDistillMagicState(state: VMState, inputRegs: number[], outputReg: number, m: number) {
  const inputs = inputRegs.map(reg => state.qutritRegisters[reg]);
  const distilled = simulateTriorthogonalDistillation(m, inputs);
  state.qutritRegisters[outputReg] = distilled;
}

// Variational correction hook (post-opcode)
function variationalQutritCorrect(state: VMState, branchRegs: [number, number, number]) {
  const branches = branchRegs.map(reg => ({
    t_alpha: Complex(state.qutritRegisters[reg][0].re), // Simplified mapping
    E_alpha: Complex(state.qutritRegisters[reg][1].re)
  }));
  const updated = quantumElBranchUpdate(branches, [0,0,0], [Complex(0), Complex(0), Complex(0)]);
  updated.forEach((branch, i) => {
    state.qutritRegisters[branchRegs[i]][0] = branch.t_alpha; // Update amplitudes
  });
}
```
