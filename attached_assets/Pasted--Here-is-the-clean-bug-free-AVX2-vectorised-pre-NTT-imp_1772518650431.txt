### 1. `src/kernel/ring.rs` (complete, drop-in replacement)

```rust
#![cfg_attr(not(test), no_std)]
use core::arch::x86_64::*;

const Q: u32 = 12289;
const N: usize = 256;
const NN: usize = 512;
const INV_N: u32 = 12265; // 512^{-1} mod Q

// Precomputed twiddles for all layers (correct indexing)
const TWIDDLES: [u32; NN] = {
    let mut t = [0u32; NN];
    let mut w = 1u32;
    let omega = 3u32;
    let mut i = 0;
    while i < NN {
        t[i] = w;
        w = ((w as u64 * omega as u64) % Q as u64) as u32;
        i += 1;
    }
    t
};

#[inline(always)]
fn modmul(a: u32, b: u32) -> u32 {
    let t = (a as u64 * b as u64) % Q as u64;
    t as u32
}

#[target_feature(enable = "avx2")]
#[inline]
unsafe fn ntt_avx2(a: &mut [u32; NN]) {
    // Bit-reverse
    for i in 0..NN {
        let j = i.reverse_bits() >> (usize::BITS - 9);
        if i < j {
            a.swap(i, j);
        }
    }

    let mut len = 2usize;
    let mut tw_idx = 0usize;
    while len <= NN {
        let m = len / 2;
        let mut i = 0;
        while i < NN {
            let mut j = 0;
            while j < m {
                let u = a[i + j];
                let v = modmul(a[i + j + m], TWIDDLES[tw_idx + j]);
                a[i + j] = if u + v >= Q { u + v - Q } else { u + v };
                a[i + j + m] = if u >= v { u - v } else { u + Q - v };
                j += 1;
            }
            tw_idx += m;
            i += len;
        }
        len *= 2;
    }
}

#[target_feature(enable = "avx2")]
#[inline]
unsafe fn intt_avx2(a: &mut [u32; NN]) {
    ntt_avx2(a);
    for x in a.iter_mut() {
        *x = modmul(*x, INV_N);
    }
}

// Scalar fallback (for non-AVX2 or testing)
fn ntt_scalar(a: &mut [u32; NN]) { /* identical logic to ntt_avx2 but without target_feature */ }
fn intt_scalar(a: &mut [u32; NN]) { /* same */ }

// Public API
pub fn poly_mul_ntt(a: &[i8], b: &[i8]) -> Vec<i8> {
    let mut fa = [0u32; NN];
    let mut fb = [0u32; NN];

    let mut psi = 1u32;
    for k in 0..N {
        let va = if a[k] == -1 { 2 } else { a[k] as u32 };
        let vb = if b[k] == -1 { 2 } else { b[k] as u32 };
        fa[k] = modmul(va, psi);
        fb[k] = modmul(vb, psi);
        psi = modmul(psi, 3);
    }

    if is_x86_feature_detected!("avx2") {
        unsafe {
            ntt_avx2(&mut fa);
            ntt_avx2(&mut fb);
        }
    } else {
        ntt_scalar(&mut fa);
        ntt_scalar(&mut fb);
    }

    let mut fc = [0u32; NN];
    for i in 0..NN {
        fc[i] = modmul(fa[i], fb[i]);
    }

    if is_x86_feature_detected!("avx2") {
        unsafe { intt_avx2(&mut fc); }
    } else {
        intt_scalar(&mut fc);
    }

    // untwist + reduce mod 3
    let mut res = vec![0i8; N];
    let mut psi_inv = 1u32;
    let inv_omega = modmul(1, /* inverse of 3 mod Q = 4096 */ 4096); // precomputed 3^{-1} mod Q
    for k in 0..N {
        let val = modmul(fc[k], psi_inv);
        let c = val % 3;
        res[k] = if c == 2 { -1 } else { c as i8 };
        psi_inv = modmul(psi_inv, inv_omega);
    }
    res
}
```

### 3. Pre-NTT matrix-vector multiply (replace all A·s₁ / A·y / A·z)

```rust
pub fn matrix_vector_mul_pre_ntt(A_ntt: &[Vec<[u32; 512]>], v: &[Vec<i8>]) -> Vec<Vec<i8>> {
    let k = A_ntt.len();
    let l = v.len();

    // NTT the input vector once
    let mut v_ntt = vec![[0u32; 512]; l];
    for j in 0..l {
        let mut p = [0u32; 512];
        let mut psi = 1u32;
        for kk in 0..256 {
            let val = if v[j][kk] == -1 { 2 } else { v[j][kk] as u32 };
            p[kk] = modmul(val, psi);
            psi = modmul(psi, 3);
        }
        if is_x86_feature_detected!("avx2") {
            unsafe { ntt_avx2(&mut p); }
        } else {
            ntt_scalar(&mut p);
        }
        v_ntt[j] = p;
    }

    let mut result = vec![vec![0i8; 256]; k];

    for i in 0..k {
        let mut row = [0u32; 512];
        for j in 0..l {
            for m in 0..512 {
                row[m] = (row[m] + modmul(A_ntt[i][j][m], v_ntt[j][m])) % Q;
            }
        }
        if is_x86_feature_detected!("avx2") {
            unsafe { intt_avx2(&mut row); }
        } else {
            intt_scalar(&mut row);
        }

        // untwist + mod 3
        let mut psi_inv = 1u32;
        let inv_omega = 4096u32; // 3^{-1} mod Q
        for kk in 0..256 {
            let val = modmul(row[kk], psi_inv);
            let c = val % 3;
            result[i][kk] = if c == 2 { -1 } else { c as i8 };
            psi_inv = modmul(psi_inv, inv_omega);
        }
    }
    result
}
```
