import type { Express, Request, Response } from "express";
import { createLogger } from "../logger";
import * as crypto from "crypto";

const log = createLogger("tdns");

// ── Types ─────────────────────────────────────────────────────────────────────
interface Dimension {
  number:     number;
  category:   string;
  question:   string;
  value:      number;
  label:      string;
  confidence: number;
}

interface ScanResult {
  status:         string;
  address:        string;
  scan_hash:      string;
  hptp_mandatory: boolean;
  crd:            number;
  dimensions:     Dimension[];
  meta:           Record<string, any>;
  securityScore:  number;
  privacyScore:   number;
  scannedAt:      string;
}

interface RegistryEntry extends ScanResult {
  name:           string;
  zone:           string;
  url:            string;
  registered_at:  string;
}

// ── In-memory registry ────────────────────────────────────────────────────────
const registry = new Map<string, RegistryEntry>();

// ── GF(3) derivation — exact formula from derive.rs ──────────────────────────
// gf3 = min(floor(3k/N), 2)   trit = gf3 + 1  →  Rep C {1,2,3}
function gf3(k: number, n: number): number {
  return Math.min(Math.floor(3 * k / n), 2) + 1;
}
function confDigit(k: number, n: number): number {
  const p = k / n;
  const d = Math.min(Math.abs(p - 1/3), Math.abs(p - 2/3));
  return Math.min(Math.floor(27 * d) + 1, 9);
}

// ── Scanner ───────────────────────────────────────────────────────────────────
async function scanUrl(rawUrl: string): Promise<ScanResult> {
  // Normalise URL
  if (!rawUrl.startsWith("http")) rawUrl = "https://" + rawUrl;
  const parsed   = new URL(rawUrl);
  const hostname = parsed.hostname.toLowerCase();
  const isHttps  = parsed.protocol === "https:";

  // Fetch with browser-like headers so sites don't block us
  let hdrs: Record<string, string> = {};
  let statusCode = 0;
  let body       = "";
  let ok         = false;

  try {
    const resp = await fetch(rawUrl, {
      method:  "GET",
      redirect: "follow",
      signal:  AbortSignal.timeout(10000),
      headers: {
        "User-Agent":      "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Safari/537.36",
        "Accept":          "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
        "Accept-Language": "en-US,en;q=0.5",
        "Accept-Encoding": "identity",
      },
    });
    ok         = resp.ok;
    statusCode = resp.status;
    resp.headers.forEach((v: string, k: string) => { hdrs[k.toLowerCase()] = v.toLowerCase(); });
    const buf  = await resp.arrayBuffer();
    body       = new TextDecoder().decode(buf.slice(0, 32768)).toLowerCase();
    log.info(`Scanned ${hostname}: HTTP ${statusCode}, ${body.length} bytes`);
  } catch (err: any) {
    log.warn(`Fetch failed for ${hostname}: ${err.message}`);
  }

  const h   = (k: string) => hdrs[k] || "";
  const has = (k: string) => !!hdrs[k];
  const b   = (p: string) => body.includes(p);

  // Enriched metadata for report
  const meta: Record<string, any> = {
    url:        rawUrl,
    hostname,
    isHttps:    isHttps && (statusCode > 0 ? true : false),
    statusCode,
    server:     h("server") || h("x-powered-by") || "—",
    ct:         h("content-type"),
    cacheCtrl:  h("cache-control"),
    hsts:       h("strict-transport-security"),
    csp:        h("content-security-policy"),
    cors:       h("access-control-allow-origin"),
    xfo:        h("x-frame-options"),
    xcto:       h("x-content-type-options"),
    via:        h("via"),
    cfRay:      has("cf-ray"),
    xCache:     has("x-cache") || has("x-amz-cf-id") || has("x-served-by"),
    altSvc:     h("alt-svc"),
    permsPolicy:h("permissions-policy"),
    nel:        has("nel"),
    reportTo:   has("report-to") || has("reporting-endpoints"),
    coep:       has("cross-origin-embedder-policy"),
    coop:       has("cross-origin-opener-policy"),
    setCookie:  h("set-cookie"),
    bodySize:   body.length,
    hasHtml:    b("<!doctype") || h("content-type").includes("html"),
    hasJson:    h("content-type").includes("json"),
    isApi:      parsed.pathname.includes("/api/"),
    isWs:       h("upgrade") === "websocket",
    isSse:      h("content-type").includes("event-stream"),
  };

  // Correct isHttps: if we got a response it was via HTTPS
  if (statusCode > 0 && rawUrl.startsWith("https://")) meta.isHttps = true;

  const isGov   = /\.(gov|mil|gc\.ca|gov\.uk)$/.test(hostname);
  const isEdu   = /\.(edu|ac\.uk|edu\.au)$/.test(hostname);
  // Major cloud providers AND companies that ARE their own cloud infra
  const isCloud = [
    "amazonaws","cloudflare","vercel","netlify","replit","fastly",
    "azurewebsites","googleusercontent","github.io","pages.dev",
    "fly.dev","render.com",
    // Major corps that run their own global infra
    "google.com","youtube.com","facebook.com","meta.com","x.com",
    "twitter.com","microsoft.com","apple.com","amazon.com",
    "netflix.com","linkedin.com","github.com","cloudflare.com",
  ].some(s => hostname === s || hostname.endsWith("." + s));

  const isLargeCorp = isCloud || [
    "google.com","microsoft.com","apple.com","amazon.com",
    "meta.com","netflix.com","github.com",
  ].some(s => hostname === s || hostname.endsWith("." + s));

  const ct        = meta.ct as string;
  const cacheCtrl = meta.cacheCtrl as string;
  const cookieVal = meta.setCookie as string;
  const maxAgeM   = cacheCtrl.match(/max-age=(\d+)/);
  const maxAge    = maxAgeM ? parseInt(maxAgeM[1]) : 3600;

  // ── 27 derivations (exact signal counts from derive.rs) ────────────────────

  // D1  WHO: What kind of entity?  [CATEGORICAL]
  const d1 = isGov || isEdu ? 3 : 2;

  // D2  WHO: Who's the audience?  [CATEGORICAL]
  const d2 = 3; // anything reachable on public internet = public

  // D3  WHO: Who operates it?  [QUANTITATIVE N=5]
  // about_page, contact_info, legal_entity, physical_address, gov_domain
  const k3 = +( b("/about") || b("about us") || b("about-us") || b("our company") )
           + +( b("contact") || b("contact us") || b("get in touch") )
           + +( b("inc.") || b(" ltd") || b("llc") || b("corp.") || b("s.a.") )
           + +( b("street") || b("avenue") || b("suite") || b("postal code") || b("zip code") )
           + +( isGov || isLargeCorp ); // known large public corps are transparent by definition
  const d3 = gf3(k3, 5);
  const c3 = confDigit(k3, 5);

  // D4  WHO: Hosting model?  [CATEGORICAL]
  const d4 = isCloud ? 3 : 2;

  // D5  WHAT: What form factor?  [CATEGORICAL]
  const d5 = (meta.isApi && !meta.hasHtml) ? 2 : 1;

  // D6  WHAT: Content type?  [CATEGORICAL]
  const d6 = meta.isWs || meta.isSse ? 3
           : ct.includes("video") || ct.includes("audio") || ct.includes("image") ? 2
           : 1;

  // D7  WHAT: Primary consumer?  [CATEGORICAL]
  const d7 = meta.hasHtml && (meta.hasJson || meta.isApi) ? 3
           : meta.hasJson || meta.isApi ? 2
           : 1;

  // D8  WHAT: AI / ML present?  [QUANTITATIVE N=5]
  const k8 = +( b("/predict") || b("/inference") || b("/model") || b("/embed") || b("/generate") )
           + +( b("tensorflow") || b("pytorch") || b("hugging face") || b("openai") || b("llm") )
           + +( b("recommendation") || b("personali") || b("suggested for you") )
           + +( b("ranking") || b("relevance score") )
           + +( has("x-ai") || has("x-model") || has("x-inference") );
  const d8 = gf3(k8, 5);
  const c8 = confDigit(k8, 5);

  // D9  WHERE: Visibility?  [QUANTITATIVE N=3]
  const k9 = +( ok || statusCode > 0 )
           + +( statusCode !== 401 && statusCode !== 403 && !has("www-authenticate") )
           + +( ok && (meta.hasHtml || meta.hasJson) );
  const d9 = gf3(k9, 3);
  const c9 = confDigit(k9, 3);

  // D10 WHERE: Authentication required?  [CATEGORICAL]
  const d10 = has("www-authenticate") ? 2
            : b("sign in") || b("log in") || b("login required") ? 2
            : 1;

  // D11 WHERE: Infrastructure scale?  [QUANTITATIVE N=6]
  const parts = hostname.split(".");
  const k11 = +( ok || statusCode > 0 )
            + +( parts.length >= 3 )
            + +( parts.length >= 4 )
            + +( meta.cfRay || has("x-cdn") || has("x-fastly") || isCloud )
            + +( meta.xCache || has("age") )
            + +( (meta.via as string) !== "" );
  const d11 = gf3(k11, 6);
  const c11 = confDigit(k11, 6);

  // D12 WHERE: Transport protocol?  [CATEGORICAL]
  const d12 = meta.isWs ? 2 : 1;

  // D13 WHEN: Technology era?  [QUANTITATIVE N=6]
  // alt_svc, permissions_policy, nel, report_to, cross_origin_policy, csp
  const k13 = +( has("alt-svc") )
            + +( has("permissions-policy") )
            + +( has("nel") )
            + +( has("report-to") || has("reporting-endpoints") )
            + +( has("cross-origin-opener-policy") || has("cross-origin-embedder-policy") )
            + +( has("content-security-policy") );
  const d13 = gf3(k13, 6);
  const c13 = confDigit(k13, 6);

  // D14 WHEN: Availability window?  [QUANTITATIVE N=3]
  const k14 = +( !b("maintenance mode") && !b("down for maintenance") )
            + +( !b("business hours") && !b("office hours") )
            + +( b("99.") || b("uptime") || b("always available") || ok );
  const d14 = gf3(k14, 3);
  const c14 = confDigit(k14, 3);

  // D15 WHEN: Data freshness?  [CATEGORICAL]
  const d15 = meta.isWs || meta.isSse ? 3
            : cacheCtrl.includes("no-store") || cacheCtrl.includes("no-cache") || cacheCtrl.includes("max-age=0") ? 2
            : cacheCtrl.includes("max-age") ? 2
            : 2;

  // D16 WHEN: Real-time capability?  [QUANTITATIVE N=6]
  const k16 = +( !cacheCtrl.includes("immutable") && maxAge < 3600 )
            + +( maxAge < 60 )
            + +( meta.isWs )
            + +( meta.isSse )
            + +( ct.includes("grpc") )
            + +( ct.includes("octet-stream") && b("stream") );
  const d16 = gf3(k16, 6);
  const c16 = confDigit(k16, 6);

  // D17 WHY: Financial transactions?  [CATEGORICAL]
  const d17 = b("swift") || b("wire transfer") || b("ach transfer") ? 3
            : b("stripe") || b("paypal") || b("checkout") || b("braintree") || b("buy now") ? 2
            : 1;

  // D18 WHY: Data collection appetite?  [QUANTITATIVE N=5]
  const k18 = +( b("<input") || b("input type=") )
            + +( b("sign up") || b("register") || b("create account") )
            + +( b("gtag(") || b("analytics.js") || b("fbq(") || b("_paq.push") || b("mixpanel") )
            + +( b("cookie consent") || b("gdpr") || b("we use cookies") )
            + +( b("third party") || b("data partner") );
  const d18 = gf3(k18, 5);
  const c18 = confDigit(k18, 5);

  // D19 WHY: Legal / policy presence?  [QUANTITATIVE N=5]
  const k19 = +( b("privacy policy") || b("/privacy") )
            + +( b("terms of service") || b("terms of use") || b("/terms") )
            + +( b("cookie policy") || b("cookie notice") )
            + +( b("gdpr") || b("data protection") )
            + +( b("accessibility") || b("wcag") );
  const d19 = gf3(k19, 5);
  const c19 = confDigit(k19, 5);

  // D20 WHY: Revenue model?  [CATEGORICAL]
  const d20 = b("subscribe") || b("per month") || b("/month") || b("annual plan") ? 3
            : b("pay per") || b("per use") || b("credits") ? 2
            : 1;

  // D21 HOW: Delivery topology?  [CATEGORICAL]
  const d21 = isCloud || meta.cfRay || meta.xCache ? 3
            : (meta.via as string) !== "" ? 2
            : 1;

  // D22 HOW: Data flow direction?  [CATEGORICAL]
  const d22 = meta.hasJson && (b('"data"') || b('"result"') || b('"id"')) ? 3
            : (meta.via as string) !== "" || meta.cfRay ? 2
            : 1;

  // D23 HOW: Update mechanism?  [CATEGORICAL]
  const d23 = meta.isWs || meta.isSse ? 3
            : b("rss") || b("atom") || b("</feed") ? 2
            : 1;

  // D24 HOW: Session persistence?  [QUANTITATIVE N=3]
  // Collect ALL Set-Cookie values — headers may arrive as comma-joined
  const allCookies = cookieVal;
  const hasCookie  = allCookies !== "";
  const hasExpires = allCookies.includes("expires=") || allCookies.includes("max-age=");
  const cookieMaxAge = parseInt((allCookies.match(/max-age=(\d+)/i) || [, "0"])[1]);
  const longLived  = cookieMaxAge > 86400 * 30
                  || (allCookies.includes("expires=") && !allCookies.includes("session"));
  const k24 = +hasCookie + +hasExpires + +longLived;
  const d24 = gf3(k24, 3);
  const c24 = confDigit(k24, 3);

  // D25 PEACE: Encryption posture?  [QUANTITATIVE N=6]
  // tls_present, hsts, csp, security_txt, xcto, xfo
  const k25 = +( meta.isHttps )
            + +( has("strict-transport-security") )
            + +( has("content-security-policy") )
            + +( b("security.txt") || b("/.well-known/security") )
            + +( has("x-content-type-options") )
            + +( has("x-frame-options") );
  const d25 = gf3(k25, 6);
  const c25 = confDigit(k25, 6);

  // D26 PEACE: Tracker density?  [QUANTITATIVE N=5, INVERTED — counts clean signals]
  const k26 = +( !b("google-analytics") && !b("gtag(") && !b("mixpanel") && !b("amplitude") )
            + +( !b("facebook.net") && !b("platform.twitter") && !b("snap.licdn") )
            + +( !b("doubleclick") && !b("googlesyndication") && !b("adsystem") )
            + +( !b("fullstory") && !b("hotjar") && !b("logrocket") && !b("clarity.ms") )
            + +( !b("hubspot") && !b("marketo") && !b("intercom") && !b("pardot") );
  const d26 = gf3(k26, 5);
  const c26 = confDigit(k26, 5);

  // D27 PEACE: Security audit status?  [CATEGORICAL]
  const d27 = b("iso 27001") || b("soc 2") || b("pci dss") || b("soc2") ? 3
            : b("penetration test") || b("bug bounty") || b("self-certified") ? 2
            : 1;

  const trits = [d1,d2,d3,d4,d5,d6,d7,d8,d9,d10,d11,d12,d13,d14,d15,d16,
                 d17,d18,d19,d20,d21,d22,d23,d24,d25,d26,d27];
  const confs  = [9,9,c3,9,9,9,9,c8,c9,9,c11,9,c13,c14,9,c16,
                  9,c18,c19,9,9,9,9,c24,c25,c26,9];

  const address = [
    `WO:${trits.slice(0,4).join("")}`,
    `WA:${trits.slice(4,8).join("")}`,
    `WR:${trits.slice(8,12).join("")}`,
    `WN:${trits.slice(12,16).join("")}`,
    `WY:${trits.slice(16,20).join("")}`,
    `HO:${trits.slice(20,24).join("")}`,
    `PE:${trits.slice(24,27).join("")}`,
  ].join(" ");

  const scan_hash = crypto.createHash("sha256")
    .update(Buffer.from(trits))
    .digest("hex");

  const SCHEMA = [
    {cat:"WHO",  q:"What kind of entity?",     vals:["Personal","Corporate","Governance"]},
    {cat:"WHO",  q:"Who's the audience?",       vals:["Just me","My group","Everyone"]},
    {cat:"WHO",  q:"Who operates it?",          vals:["Anonymous","Known","Transparent"]},
    {cat:"WHO",  q:"Hosting model?",            vals:["Self-hosted","Provider","Cloud"]},
    {cat:"WHAT", q:"What form factor?",         vals:["Website","App / API","Device"]},
    {cat:"WHAT", q:"Content type?",             vals:["Text / HTML","Media","Live stream"]},
    {cat:"WHAT", q:"Primary consumer?",         vals:["Humans","Machines","Both"]},
    {cat:"WHAT", q:"AI / ML present?",          vals:["No","Partially","Yes"]},
    {cat:"WHERE",q:"Visibility?",               vals:["Private","Group","Public"]},
    {cat:"WHERE",q:"Authentication required?",  vals:["None","Password","Strong ID"]},
    {cat:"WHERE",q:"Infrastructure scale?",     vals:["Single server","Several","CDN / Many"]},
    {cat:"WHERE",q:"Transport protocol?",       vals:["HTTP","WebSocket","Raw TCP"]},
    {cat:"WHEN", q:"Technology era?",           vals:["Pre-2010","2010s","2020s+"]},
    {cat:"WHEN", q:"Availability window?",      vals:["Business hours","Extended","24/7"]},
    {cat:"WHEN", q:"Data freshness?",           vals:["Historical","Current","Live"]},
    {cat:"WHEN", q:"Real-time capability?",     vals:["Batch","Near-real-time","Real-time"]},
    {cat:"WHY",  q:"Financial transactions?",   vals:["None","Accepts payment","Processes"]},
    {cat:"WHY",  q:"Data collection appetite?", vals:["Minimal","Moderate","Heavy"]},
    {cat:"WHY",  q:"Legal / policy presence?",  vals:["None","Basic","Comprehensive"]},
    {cat:"WHY",  q:"Revenue model?",            vals:["Free","Pay-per-use","Subscription"]},
    {cat:"HOW",  q:"Delivery topology?",        vals:["Unicast","Multicast","Anycast / Edge"]},
    {cat:"HOW",  q:"Data flow direction?",      vals:["Outbound","Relay / Proxy","Inbound"]},
    {cat:"HOW",  q:"Update mechanism?",         vals:["Pull / Poll","Subscribe","Push"]},
    {cat:"HOW",  q:"Session persistence?",      vals:["Stateless","Short session","Long-lived"]},
    {cat:"PEACE",q:"Encryption posture?",       vals:["Weak / None","Basic TLS","Hardened TLS"]},
    {cat:"PEACE",q:"Tracker density?",          vals:["Heavy","Moderate","Clean"]},
    {cat:"PEACE",q:"Security audit status?",    vals:["None","Self-assessed","Third-party"]},
  ];

  const dimensions: Dimension[] = SCHEMA.map((s, i) => ({
    number:     i + 1,
    category:   s.cat,
    question:   s.q,
    value:      trits[i],
    label:      s.vals[trits[i] - 1],
    confidence: confs[i],
  }));

  const hptp_mandatory = trits[14] === 3 && trits[15] === 3;
  const securityScore  = Math.round(((trits[24] + trits[25] + trits[26]) / 9) * 100);
  const privacyScore   = Math.round(((trits[25] + trits[18]) / 6) * 100);

  // CRD: collision resolution digit (1–9, derived from scan hash)
  const crd = (parseInt(scan_hash[0], 16) % 9) + 1;

  return {
    status: "ok",
    address,
    scan_hash,
    hptp_mandatory,
    crd,
    dimensions,
    meta,
    securityScore,
    privacyScore,
    scannedAt: new Date().toISOString(),
  };
}

// ── Route registration ────────────────────────────────────────────────────────
export function registerTdnsRoutes(app: Express) {

  // Health
  app.get("/api/tdns/health", (_req: Request, res: Response) => {
    res.json({ status: "ok", version: "2.3.3", entities: registry.size, engine: "server-js" });
  });

  // Scan — derive 27-dim address for any URL
  app.post("/api/tdns/scan", async (req: Request, res: Response) => {
    const { url } = req.body;
    if (!url) { res.status(400).json({ error: "url required" }); return; }
    try {
      const result = await scanUrl(url);
      res.json(result);
    } catch (err: any) {
      log.error(`Scan error: ${err.message}`);
      res.status(500).json({ error: err.message });
    }
  });

  // Register — scan + add to registry
  app.post("/api/tdns/register", async (req: Request, res: Response) => {
    const { name, zone, url } = req.body;
    if (!name || !url) { res.status(400).json({ error: "name and url required" }); return; }
    try {
      const scan    = await scanUrl(url);
      const plmName = name.endsWith(".plm") ? name : name + ".plm";
      const entry: RegistryEntry = {
        ...scan,
        name:          plmName,
        zone:          zone || "public",
        url,
        registered_at: new Date().toISOString(),
      };
      registry.set(plmName, entry);
      log.info(`Registered ${plmName} → ${scan.address}`);
      res.json({ status: "ok", name: plmName, address: scan.address,
                 scan_hash: scan.scan_hash, hptp_mandatory: scan.hptp_mandatory, crd: scan.crd });
    } catch (err: any) {
      log.error(`Register error: ${err.message}`);
      res.status(500).json({ error: err.message });
    }
  });

  // Resolve — look up registered .plm name
  app.get("/api/tdns/resolve/:name", (req: Request, res: Response) => {
    const name    = req.params.name.endsWith(".plm") ? req.params.name : req.params.name + ".plm";
    const entry   = registry.get(name);
    if (!entry) { res.status(404).json({ status: "not_found", name }); return; }
    res.json({ status: "ok", name: entry.name, address: entry.address,
               scan_hash: entry.scan_hash, hptp_mandatory: entry.hptp_mandatory,
               crd: entry.crd, registered_at: entry.registered_at });
  });

  // List all registered entities
  app.get("/api/tdns/list", (_req: Request, res: Response) => {
    const entries = Array.from(registry.values()).map(e => ({
      name: e.name, address: e.address, url: e.url, registered_at: e.registered_at,
    }));
    res.json({ status: "ok", count: entries.length, entries });
  });

  log.info("TDNS routes registered — server-side scanner active");
}
