<script>
  const TDNS_API = 'http://localhost:3927';
  const CATEGORIES = {
    'WO': 'WHO — Who Is Behind It?',
    'WA': 'WHAT — What Is It?',
    'WR': 'WHERE — Where Can I Find It?',
    'WN': 'WHEN — When Does It Operate?',
    'WY': 'WHY — Why Does It Exist?',
    'HO': 'HOW — How Does It Work?',
    'PE': 'PEACE — Can I Sleep at Night?'
  };

  function buildConfBar(conf) {
    const c = typeof conf === 'number' ? conf : parseInt(conf) || 5;
    let html = '<div class="conf-bar" title="Confidence: ' + c + '/9">';
    for (let i = 1; i <= 9; i++) {
      let cls = 'conf-pip';
      if (i <= c) {
        cls += ' active';
        if (c <= 3) cls += ' low';
        else if (c <= 6) cls += ' mid';
      }
      html += `<div class="${cls}"></div>`;
    }
    html += '</div>';
    return html;
  }

  function renderDimensions(dims) {
    const container = document.getElementById('dimensions');
    let currentCat = '';
    let html = '';

    for (const dim of dims) {
      const cat = dim.category;
      if (cat !== currentCat) {
        if (currentCat) html += '</div>';
        html += `<div class="category-group">`;
        html += `<div class="category-label">${CATEGORIES[cat] || cat}</div>`;
        currentCat = cat;
      }

      const conf = typeof dim.confidence === 'number'
        ? dim.confidence
        : parseInt(String(dim.confidence).replace(/\D/g, '')) || 9;

      html += `
        <div class="dim-row">
          <div class="dim-num">${dim.number}</div>
          <div class="dim-question">${dim.question}</div>
          <div class="dim-label">${dim.label}</div>
          <div class="dim-trit trit-${dim.value}">${dim.value}</div>
          ${buildConfBar(conf)}
        </div>
      `;
    }
    if (currentCat) html += '</div>';
    container.innerHTML = html;
  }

  async function resolveName(name) {
    // First try resolve (registered names)
    try {
      const resp = await fetch(`${TDNS_API}/api/v1/resolve/${name}`);
      if (resp.ok) {
        const data = await resp.json();
        if (data.status === 'ok') return { type: 'resolved', data };
      }
    } catch (e) {}

    // Fall back to scan (derive address without registration)
    // Extract likely URL from .plm name
    const baseName = name.replace(/\.plm$/, '');
    const scanUrl = `https://${baseName.replace(/\./g, '.')}.com`;

    try {
      const resp = await fetch(`${TDNS_API}/api/v1/scan`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ url: scanUrl })
      });
      if (resp.ok) {
        const data = await resp.json();
        if (data.status === 'ok') return { type: 'scanned', data };
      }
    } catch (e) {}

    return null;
  }

  async function init() {
    const params = new URLSearchParams(window.location.search);
    const name = params.get('name');

    if (!name) {
      document.getElementById('loading').style.display = 'none';
      document.getElementById('errorState').style.display = 'block';
      document.getElementById('errorState').innerHTML = `
        <h3>No name specified</h3>
        <p>Use the omnibox: type <strong>plm</strong> then a name.</p>
      `;
      return;
    }

    document.title = `${name} — PlenumNET TDNS`;

    const result = await resolveName(name);

    document.getElementById('loading').style.display = 'none';

    if (!result) {
      document.getElementById('errorState').style.display = 'block';
      document.getElementById('errorState').innerHTML = `
        <h3>${name}</h3>
        <p>Could not resolve or scan this name.</p>
        <p style="color:#5A5548;margin-top:8px;">
          Make sure the TDNS server is running on port 3927.
        </p>
      `;
      return;
    }

    const data = result.data;
    const isRegistered = result.type === 'resolved';

    document.getElementById('content').style.display = 'block';
    document.getElementById('nameDisplay').textContent =
      name + (isRegistered ? '' : '  (scanned — not registered)');
    document.getElementById('addressDisplay').textContent = data.address;
    document.getElementById('crdBadge').textContent =
      'CRD:' + (data.crd || '—');
    document.getElementById('metaHash').textContent = data.scan_hash || '—';
    document.getElementById('metaHptp').textContent =
      data.hptp_mandatory ? 'Yes — HPTP sync required' : 'No';
    document.getElementById('metaCanonical').textContent =
      data.address_canonical || '—';

    if (data.dimensions) {
      renderDimensions(data.dimensions);
    }
  }

  init();
