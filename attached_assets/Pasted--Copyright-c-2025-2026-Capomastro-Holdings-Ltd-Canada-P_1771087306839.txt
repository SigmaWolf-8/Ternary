use alloc::string::String;
use alloc::vec::Vec;
use super::{VmError, VmResult};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct TernaryRegister {
    pub value: i64,
    pub ternary_mode: bool,
}

impl Default for TernaryRegister {
    fn default() -> Self {
        Self {
            value: 0,
            ternary_mode: false,
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct VmFlags {
    pub zero: bool,
    pub negative: bool,
    pub positive: bool,
    pub overflow: bool,
    pub ternary: bool,
    pub halted: bool,
    // v2.0 additions
    pub carry: bool,
    pub parity: bool,
    pub interrupt_enabled: bool,
}

impl Default for VmFlags {
    fn default() -> Self {
        Self {
            zero: false,
            negative: false,
            positive: false,
            overflow: false,
            ternary: false,
            halted: false,
            carry: false,
            parity: false,
            interrupt_enabled: true,
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum PrivilegeLevel {
    Ring0 = 0,
    Ring1 = 1,
    Ring2 = 2,  // v2.0: new service/driver level
}

impl Default for PrivilegeLevel {
    fn default() -> Self {
        PrivilegeLevel::Ring0
    }
}

#[derive(Debug, Clone)]
pub struct RegisterFile {
    pub registers: [TernaryRegister; 27],
    pub program_counter: u64,
    pub stack_pointer: u64,
    pub frame_pointer: u64,          // v2.0: frame pointer for call frames
    pub link_register: u64,          // v2.0: return address register
    pub flags: VmFlags,
    pub privilege: PrivilegeLevel,
    pub security_domain: u8,         // v2.0: explicit security domain
    pub exception_vector: u64,       // v2.0: exception handler address
}

impl Default for RegisterFile {
    fn default() -> Self {
        Self {
            registers: [TernaryRegister::default(); 27],
            program_counter: 0,
            stack_pointer: 0,
            frame_pointer: 0,
            link_register: 0,
            flags: VmFlags::default(),
            privilege: PrivilegeLevel::default(),
            security_domain: 0,
            exception_vector: 0,
        }
    }
}

// ============================================================================
// ISA v2.0 Opcode Enumeration — 156 Instructions
// ============================================================================
//
// Layout by category and hex range:
//
//   0x00–0x0F  Basic & Extended Arithmetic        (16 opcodes)
//   0x10–0x1F  Ternary Core                       (16 opcodes)  [unchanged]
//   0x20–0x2F  Memory, Register & Atomics         (16 opcodes)
//   0x30–0x3F  Control Flow                       (16 opcodes)
//   0x40–0x4F  Comparison & Selection             (16 opcodes)
//   0x50–0x5F  Binary Logic & Bit Manipulation    (16 opcodes)
//   0x60–0x6F  Crypto Acceleration                (16 opcodes)
//   0x70–0x7F  SIMD / Vector Ternary              (16 opcodes)
//   0x80–0x8F  System & Privilege                 (16 opcodes)
//   0x90–0x97  Security & Audit                   (8 opcodes)
//   0x98–0x9F  Debug & Profiling                  (8 opcodes)
//
// Total: 160 allocated slots, 160 defined (0 reserved — fully populated)
// ============================================================================

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
#[repr(u8)]
pub enum Opcode {
    // ========================================================================
    // 0x00–0x0F: Basic & Extended Arithmetic
    // ========================================================================
    Nop         = 0x00,  // No operation
    Halt        = 0x01,  // Stop execution; set halted flag
    Add         = 0x02,  // dst ← src1 + src2 (i64, overflow-detecting)
    Sub         = 0x03,  // dst ← src1 − src2
    Mul         = 0x04,  // dst ← src1 × src2
    Div         = 0x05,  // dst ← src1 / src2 (trap on zero)
    Mod         = 0x06,  // dst ← src1 % src2 (trap on zero)
    Neg         = 0x07,  // dst ← −src1
    // --- v2.0 additions ---
    Abs         = 0x08,  // dst ← |src1|
    AddImm      = 0x09,  // dst ← src1 + #imm (immediate add, avoids LOADIMM+ADD)
    MulAdd      = 0x0A,  // dst ← src1 × src2 + dst (fused multiply-accumulate)
    MulSub      = 0x0B,  // dst ← src1 × src2 − dst (fused multiply-subtract)
    DivMod      = 0x0C,  // dst ← src1/src2, src2_out ← src1%src2 (quotient+remainder)
    Min         = 0x0D,  // dst ← min(src1, src2) signed
    Max         = 0x0E,  // dst ← max(src1, src2) signed
    Clamp       = 0x0F,  // dst ← clamp(dst, src1=lo, src2=hi)

    // ========================================================================
    // 0x10–0x1F: Ternary Core (unchanged from v1.0)
    // ========================================================================
    TAdd        = 0x10,  // GF(3) trit-wise addition
    TMul        = 0x11,  // GF(3) trit-wise multiplication
    TNeg        = 0x12,  // Trit-wise negation (NOT): −t per trit
    TRot        = 0x13,  // Trit-wise rotation (−1→0→+1→−1) by src2 positions
    TXor        = 0x14,  // Ternary XOR: Kleene min(a, b) per trit
    TConvert    = 0x15,  // Convert trits between representations A/B/C
    TAnd        = 0x16,  // Łukasiewicz conjunction: max(a+b−1, −1) per trit
    TOr         = 0x17,  // Kleene disjunction: max(a, b) per trit
    TSub        = 0x18,  // GF(3) trit-wise subtraction: add(a, not(b))
    TInv        = 0x19,  // GF(3) multiplicative inverse per trit
    TShift      = 0x1A,  // Trit shift left/right; zero-fill
    TCmp        = 0x1B,  // Trit-wise compare → {−1, 0, +1}
    TLoad       = 0x1C,  // Load from memory using ternary/binary address
    TStore      = 0x1D,  // Store to memory using ternary/binary address
    TReduce     = 0x1E,  // Reduce all 27 trits via gate (add/mul/min/max)
    TRotInv     = 0x1F,  // Trit-wise inverse rotation (+1→0→−1→+1)

    // ========================================================================
    // 0x20–0x2F: Memory, Register & Atomics
    // ========================================================================
    Load        = 0x20,  // dst ← mem[src1 + imm] (8-byte read)
    Store       = 0x21,  // mem[src1 + imm] ← dst (8-byte write)
    Move        = 0x22,  // dst ← src1
    LoadImm     = 0x23,  // dst ← immediate
    Push        = 0x24,  // Push src1 onto value stack
    Pop         = 0x25,  // Pop top of value stack into dst
    // --- v2.0 additions ---
    Lea         = 0x26,  // dst ← src1 + imm (load effective address, no memory access)
    LoadByte    = 0x27,  // dst ← mem[src1 + imm] as u8, zero-extended
    StoreByte   = 0x28,  // mem[src1 + imm] ← dst & 0xFF
    Exchange    = 0x29,  // Atomic swap: dst ↔ mem[src1 + imm]
    CompareSwap = 0x2A,  // Atomic CAS: if mem[addr]==src1 then mem[addr]←src2, dst←old
    AtomicAdd   = 0x2B,  // Atomic: mem[src1+imm] += dst; dst ← old value
    Fence       = 0x2C,  // Memory fence (imm: 0=full, 1=acquire, 2=release)
    PushFrame   = 0x2D,  // Push frame pointer, set FP ← SP
    PopFrame    = 0x2E,  // Restore SP ← FP, pop old FP
    MemCopy     = 0x2F,  // Copy dst_addr(r_dst), src_addr(r_src1), len(r_src2) bytes

    // ========================================================================
    // 0x30–0x3F: Control Flow
    // ========================================================================
    Jump        = 0x30,  // PC ← imm (unconditional)
    JumpZero    = 0x31,  // PC ← imm if zero flag set
    JumpNeg     = 0x32,  // PC ← imm if negative flag set
    JumpPos     = 0x33,  // PC ← imm if !zero && !negative
    Call        = 0x34,  // Push PC; PC ← imm
    Return      = 0x35,  // PC ← pop()
    JumpNotZero = 0x36,  // PC ← imm if !zero
    // --- v2.0 additions ---
    JumpGe      = 0x37,  // PC ← imm if !negative (greater or equal)
    JumpLe      = 0x38,  // PC ← imm if negative || zero (less or equal)
    JumpOvf     = 0x39,  // PC ← imm if overflow flag set
    JumpCarry   = 0x3A,  // PC ← imm if carry flag set
    CallInd     = 0x3B,  // Push PC; PC ← register[src1] + imm (indirect call)
    TailCall    = 0x3C,  // PC ← imm (no push; reuse current frame)
    Loop        = 0x3D,  // Decrement dst; if dst > 0, PC ← imm
    JumpTable   = 0x3E,  // PC ← program_data[src1 + imm] (indexed dispatch)
    ReturnInt   = 0x3F,  // Return from interrupt/exception handler

    // ========================================================================
    // 0x40–0x4F: Comparison & Selection
    // ========================================================================
    Cmp         = 0x40,  // Update flags from src1 − src2 (result discarded)
    CmpImm      = 0x41,  // Update flags from src1 − imm
    // --- v2.0 additions ---
    CmpUnsigned = 0x42,  // Unsigned comparison: update flags from src1 − src2
    Test        = 0x43,  // Update flags from src1 & src2 (bitwise AND, result discarded)
    Select      = 0x44,  // dst ← (flags.zero) ? src1 : src2 (conditional move)
    SelectNeg   = 0x45,  // dst ← (flags.negative) ? src1 : src2
    SelectOvf   = 0x46,  // dst ← (flags.overflow) ? src1 : src2
    SignExtend  = 0x47,  // dst ← sign-extend src1 from bit width specified by imm
    ZeroExtend  = 0x48,  // dst ← zero-extend src1 from bit width specified by imm
    CountOnes   = 0x49,  // dst ← popcount(src1) (population count / Hamming weight)
    LeadZeros   = 0x4A,  // dst ← count leading zeros of src1
    TrailZeros  = 0x4B,  // dst ← count trailing zeros of src1
    BitReverse  = 0x4C,  // dst ← bit-reverse of src1
    ByteSwap    = 0x4D,  // dst ← byte-swap of src1 (endian conversion)
    TritCount   = 0x4E,  // dst ← count of trits matching value in imm (0=neg, 1=zero, 2=pos)
    TritBalance = 0x4F,  // dst ← (count_pos − count_neg) for packed trit word src1

    // ========================================================================
    // 0x50–0x5F: Binary Logic & Bit Manipulation
    // ========================================================================
    And         = 0x50,  // dst ← src1 & src2 (bitwise)
    Or          = 0x51,  // dst ← src1 | src2
    Xor         = 0x52,  // dst ← src1 ^ src2
    Shl         = 0x53,  // dst ← src1 << (src2 & 63)
    Shr         = 0x54,  // dst ← src1 >> (src2 & 63) (arithmetic)
    Not         = 0x55,  // dst ← !src1 (bitwise complement)
    // --- v2.0 additions ---
    ShrLogical  = 0x56,  // dst ← src1 >>> (src2 & 63) (logical/unsigned shift right)
    RotateLeft  = 0x57,  // dst ← rotate_left(src1, src2 & 63)
    RotateRight = 0x58,  // dst ← rotate_right(src1, src2 & 63)
    AndNot      = 0x59,  // dst ← src1 & (!src2) (bit clear)
    OrNot       = 0x5A,  // dst ← src1 | (!src2)
    Xnor        = 0x5B,  // dst ← !(src1 ^ src2) (equivalence)
    BitDeposit  = 0x5C,  // dst ← PDEP(src1, src2) — parallel bit deposit
    BitExtract  = 0x5D,  // dst ← PEXT(src1, src2) — parallel bit extract
    BitField    = 0x5E,  // dst ← extract bitfield from src1: offset=imm[7:0], width=imm[15:8]
    Crc32       = 0x5F,  // dst ← CRC32(dst, src1) — hardware-accelerated CRC

    // ========================================================================
    // 0x60–0x6F: Crypto Acceleration
    // ========================================================================
    TPolyMul    = 0x60,  // GF(3) polynomial multiplication mod x^d
    TNTT        = 0x61,  // Number-Theoretic Transform (forward/inverse)
    THash       = 0x62,  // Ternary sponge-based hash of packed trit word
    TEntropy    = 0x63,  // Read hardware entropy into dst
    TPolyAdd    = 0x64,  // GF(3) polynomial addition mod x^d
    TPolySample = 0x65,  // Sample random GF(3) polynomial (CBD/Gaussian)
    TCompress   = 0x66,  // Compress packed trits for transmission
    TDecompress = 0x67,  // Decompress packed trits from wire format
    // --- v2.0 additions ---
    TLatticeMul = 0x68,  // Lattice matrix-vector multiply in GF(3) (ML-KEM/ML-DSA core)
    TSpongeAbsorb = 0x69, // Sponge absorb: feed src1 into crypto sponge state
    TSpongeSqueeze = 0x6A, // Sponge squeeze: extract dst bytes from sponge state
    TLamportGen = 0x6B,  // Generate Lamport OTS key pair segment
    TLamportSign = 0x6C, // Lamport signature: sign hash at src1 with key at src2
    TKemEncaps  = 0x6D,  // KEM encapsulation: ciphertext ← encaps(public_key)
    TKemDecaps  = 0x6E,  // KEM decapsulation: shared_secret ← decaps(ciphertext, sk)
    TZeroize    = 0x6F,  // Secure zeroize: constant-time wipe memory [src1..src1+imm]

    // ========================================================================
    // 0x70–0x7F: SIMD / Vector Ternary
    // ========================================================================
    TAddV       = 0x70,  // Vector GF(3) addition (multi-word)
    TMulV       = 0x71,  // Vector GF(3) multiplication (multi-word)
    TNegV       = 0x72,  // Vector GF(3) negation (multi-word)
    TRotV       = 0x73,  // Vector GF(3) rotation (multi-word)
    // --- v2.0 additions ---
    TXorV       = 0x74,  // Vector Kleene min (multi-word)
    TAndV       = 0x75,  // Vector Łukasiewicz conjunction (multi-word)
    TOrV        = 0x76,  // Vector Kleene disjunction (multi-word)
    TCmpV       = 0x77,  // Vector trit-wise compare (multi-word)
    TShiftV     = 0x78,  // Vector trit shift (multi-word)
    TReduceV    = 0x79,  // Vector reduce: fold N words into one via gate
    TConvertV   = 0x7A,  // Vector convert: transform N words between repr A/B/C
    TSelectV    = 0x7B,  // Vector conditional select: per-element mask
    TBroadcast  = 0x7C,  // Broadcast single trit value across all 27 positions
    TExtract    = 0x7D,  // Extract single trit from packed word: dst ← trit[src2] of src1
    TInsert     = 0x7E,  // Insert single trit into packed word: dst[imm] ← src1 trit
    TPermute    = 0x7F,  // Permute trits within packed word per index table in src2

    // ========================================================================
    // 0x80–0x8F: System & Privilege
    // ========================================================================
    Syscall     = 0x80,  // System call (imm = syscall number)
    Trap        = 0x81,  // Software trap/interrupt (Ring0 only)
    Alloc       = 0x82,  // GC heap allocation: dst ← handle
    Free        = 0x83,  // GC heap free: release handle in src1
    ReadTime    = 0x84,  // Read HPTP femtosecond timestamp into dst
    // --- v2.0 additions ---
    PrivEscalate = 0x85, // Escalate to Ring0 (requires valid capability token in src1)
    PrivDrop    = 0x86,  // Drop to Ring1/Ring2 (imm: target level)
    DomainSet   = 0x87,  // Set security domain (Ring0 only): domain ← imm
    DomainGet   = 0x88,  // Get current security domain: dst ← domain
    GetCycles   = 0x89,  // Read cycle counter: dst ← cycles_executed
    GetPid      = 0x8A,  // Read process ID: dst ← current PID
    MProtect    = 0x8B,  // Set memory protection: addr=src1, len=src2, perm=imm
    IoRead      = 0x8C,  // I/O port read: dst ← io[src1 + imm]
    IoWrite     = 0x8D,  // I/O port write: io[src1 + imm] ← dst
    CpuId       = 0x8E,  // Query VM capabilities: dst ← feature_bits[imm]
    Yield       = 0x8F,  // Yield time slice to scheduler

    // ========================================================================
    // 0x90–0x97: Security & Audit
    // ========================================================================
    AuditLog    = 0x90,  // Write audit event: type=imm, data=src1 (FINRA 613 / CAT NMS)
    CapCheck    = 0x91,  // Check capability: dst ← 1 if cap[imm] granted, else 0
    CapGrant    = 0x92,  // Grant capability to process: pid=src1, cap=imm (Ring0)
    CapRevoke   = 0x93,  // Revoke capability from process: pid=src1, cap=imm (Ring0)
    SideChMask  = 0x94,  // Apply side-channel mask: dst ← src1 ^ random_mask
    SideChUnmask = 0x95, // Remove side-channel mask: dst ← src1 ^ stored_mask
    ConstTimeEq = 0x96,  // Constant-time equality: dst ← (src1 == src2) ? 1 : 0
    ConstTimeSel = 0x97, // Constant-time select: dst ← (src1 != 0) ? src2 : dst

    // ========================================================================
    // 0x98–0x9F: Debug & Profiling
    // ========================================================================
    DebugBreak  = 0x98,  // Trigger debugger breakpoint
    DebugPrint  = 0x99,  // Debug print: register src1 to debug output
    ProfStart   = 0x9A,  // Start profiling counter: counter_id = imm
    ProfStop    = 0x9B,  // Stop profiling counter: dst ← elapsed for counter_id = imm
    ProfRead    = 0x9C,  // Read profiling counter without stopping: dst ← current value
    TraceEmit   = 0x9D,  // Emit trace event: type=imm, value=src1
    AssertEq    = 0x9E,  // Debug assert: trap if src1 != src2
    AssertTrit  = 0x9F,  // Debug assert: trap if src1 trit[imm & 0x1F] != (imm >> 8)
}

// ============================================================================
// Opcode decoding — from_u8
// ============================================================================

impl Opcode {
    pub fn from_u8(value: u8) -> VmResult<Self> {
        match value {
            // Basic & Extended Arithmetic (0x00–0x0F)
            0x00 => Ok(Opcode::Nop),
            0x01 => Ok(Opcode::Halt),
            0x02 => Ok(Opcode::Add),
            0x03 => Ok(Opcode::Sub),
            0x04 => Ok(Opcode::Mul),
            0x05 => Ok(Opcode::Div),
            0x06 => Ok(Opcode::Mod),
            0x07 => Ok(Opcode::Neg),
            0x08 => Ok(Opcode::Abs),
            0x09 => Ok(Opcode::AddImm),
            0x0A => Ok(Opcode::MulAdd),
            0x0B => Ok(Opcode::MulSub),
            0x0C => Ok(Opcode::DivMod),
            0x0D => Ok(Opcode::Min),
            0x0E => Ok(Opcode::Max),
            0x0F => Ok(Opcode::Clamp),

            // Ternary Core (0x10–0x1F)
            0x10 => Ok(Opcode::TAdd),
            0x11 => Ok(Opcode::TMul),
            0x12 => Ok(Opcode::TNeg),
            0x13 => Ok(Opcode::TRot),
            0x14 => Ok(Opcode::TXor),
            0x15 => Ok(Opcode::TConvert),
            0x16 => Ok(Opcode::TAnd),
            0x17 => Ok(Opcode::TOr),
            0x18 => Ok(Opcode::TSub),
            0x19 => Ok(Opcode::TInv),
            0x1A => Ok(Opcode::TShift),
            0x1B => Ok(Opcode::TCmp),
            0x1C => Ok(Opcode::TLoad),
            0x1D => Ok(Opcode::TStore),
            0x1E => Ok(Opcode::TReduce),
            0x1F => Ok(Opcode::TRotInv),

            // Memory, Register & Atomics (0x20–0x2F)
            0x20 => Ok(Opcode::Load),
            0x21 => Ok(Opcode::Store),
            0x22 => Ok(Opcode::Move),
            0x23 => Ok(Opcode::LoadImm),
            0x24 => Ok(Opcode::Push),
            0x25 => Ok(Opcode::Pop),
            0x26 => Ok(Opcode::Lea),
            0x27 => Ok(Opcode::LoadByte),
            0x28 => Ok(Opcode::StoreByte),
            0x29 => Ok(Opcode::Exchange),
            0x2A => Ok(Opcode::CompareSwap),
            0x2B => Ok(Opcode::AtomicAdd),
            0x2C => Ok(Opcode::Fence),
            0x2D => Ok(Opcode::PushFrame),
            0x2E => Ok(Opcode::PopFrame),
            0x2F => Ok(Opcode::MemCopy),

            // Control Flow (0x30–0x3F)
            0x30 => Ok(Opcode::Jump),
            0x31 => Ok(Opcode::JumpZero),
            0x32 => Ok(Opcode::JumpNeg),
            0x33 => Ok(Opcode::JumpPos),
            0x34 => Ok(Opcode::Call),
            0x35 => Ok(Opcode::Return),
            0x36 => Ok(Opcode::JumpNotZero),
            0x37 => Ok(Opcode::JumpGe),
            0x38 => Ok(Opcode::JumpLe),
            0x39 => Ok(Opcode::JumpOvf),
            0x3A => Ok(Opcode::JumpCarry),
            0x3B => Ok(Opcode::CallInd),
            0x3C => Ok(Opcode::TailCall),
            0x3D => Ok(Opcode::Loop),
            0x3E => Ok(Opcode::JumpTable),
            0x3F => Ok(Opcode::ReturnInt),

            // Comparison & Selection (0x40–0x4F)
            0x40 => Ok(Opcode::Cmp),
            0x41 => Ok(Opcode::CmpImm),
            0x42 => Ok(Opcode::CmpUnsigned),
            0x43 => Ok(Opcode::Test),
            0x44 => Ok(Opcode::Select),
            0x45 => Ok(Opcode::SelectNeg),
            0x46 => Ok(Opcode::SelectOvf),
            0x47 => Ok(Opcode::SignExtend),
            0x48 => Ok(Opcode::ZeroExtend),
            0x49 => Ok(Opcode::CountOnes),
            0x4A => Ok(Opcode::LeadZeros),
            0x4B => Ok(Opcode::TrailZeros),
            0x4C => Ok(Opcode::BitReverse),
            0x4D => Ok(Opcode::ByteSwap),
            0x4E => Ok(Opcode::TritCount),
            0x4F => Ok(Opcode::TritBalance),

            // Binary Logic & Bit Manipulation (0x50–0x5F)
            0x50 => Ok(Opcode::And),
            0x51 => Ok(Opcode::Or),
            0x52 => Ok(Opcode::Xor),
            0x53 => Ok(Opcode::Shl),
            0x54 => Ok(Opcode::Shr),
            0x55 => Ok(Opcode::Not),
            0x56 => Ok(Opcode::ShrLogical),
            0x57 => Ok(Opcode::RotateLeft),
            0x58 => Ok(Opcode::RotateRight),
            0x59 => Ok(Opcode::AndNot),
            0x5A => Ok(Opcode::OrNot),
            0x5B => Ok(Opcode::Xnor),
            0x5C => Ok(Opcode::BitDeposit),
            0x5D => Ok(Opcode::BitExtract),
            0x5E => Ok(Opcode::BitField),
            0x5F => Ok(Opcode::Crc32),

            // Crypto Acceleration (0x60–0x6F)
            0x60 => Ok(Opcode::TPolyMul),
            0x61 => Ok(Opcode::TNTT),
            0x62 => Ok(Opcode::THash),
            0x63 => Ok(Opcode::TEntropy),
            0x64 => Ok(Opcode::TPolyAdd),
            0x65 => Ok(Opcode::TPolySample),
            0x66 => Ok(Opcode::TCompress),
            0x67 => Ok(Opcode::TDecompress),
            0x68 => Ok(Opcode::TLatticeMul),
            0x69 => Ok(Opcode::TSpongeAbsorb),
            0x6A => Ok(Opcode::TSpongeSqueeze),
            0x6B => Ok(Opcode::TLamportGen),
            0x6C => Ok(Opcode::TLamportSign),
            0x6D => Ok(Opcode::TKemEncaps),
            0x6E => Ok(Opcode::TKemDecaps),
            0x6F => Ok(Opcode::TZeroize),

            // SIMD / Vector Ternary (0x70–0x7F)
            0x70 => Ok(Opcode::TAddV),
            0x71 => Ok(Opcode::TMulV),
            0x72 => Ok(Opcode::TNegV),
            0x73 => Ok(Opcode::TRotV),
            0x74 => Ok(Opcode::TXorV),
            0x75 => Ok(Opcode::TAndV),
            0x76 => Ok(Opcode::TOrV),
            0x77 => Ok(Opcode::TCmpV),
            0x78 => Ok(Opcode::TShiftV),
            0x79 => Ok(Opcode::TReduceV),
            0x7A => Ok(Opcode::TConvertV),
            0x7B => Ok(Opcode::TSelectV),
            0x7C => Ok(Opcode::TBroadcast),

    /// Returns the number of source registers this opcode reads
    pub fn src_count(&self) -> u8 {
        match self {
            // Zero source
            Opcode::Nop | Opcode::Halt | Opcode::Return | Opcode::ReturnInt |
            Opcode::DebugBreak | Opcode::Yield | Opcode::Fence => 0,

            // One source
            Opcode::Neg | Opcode::Abs | Opcode::TNeg | Opcode::TInv | Opcode::TRotInv |
            Opcode::Push | Opcode::Not | Opcode::TNegV | Opcode::Jump | Opcode::JumpZero |
            Opcode::JumpNeg | Opcode::JumpPos | Opcode::JumpNotZero | Opcode::Call |
            Opcode::JumpGe | Opcode::JumpLe | Opcode::JumpOvf | Opcode::JumpCarry |
            Opcode::TailCall | Opcode::LoadImm | Opcode::Pop | Opcode::CountOnes |
            Opcode::LeadZeros | Opcode::TrailZeros | Opcode::BitReverse | Opcode::ByteSwap |
            Opcode::TritBalance | Opcode::TBroadcast | Opcode::CmpImm | Opcode::AddImm |
            Opcode::SignExtend | Opcode::ZeroExtend | Opcode::DebugPrint |
            Opcode::Alloc | Opcode::Free | Opcode::Syscall | Opcode::Trap |
            Opcode::ReadTime | Opcode::GetCycles | Opcode::GetPid | Opcode::CpuId |
            Opcode::DomainGet | Opcode::ProfStart | Opcode::ProfStop | Opcode::ProfRead |
            Opcode::Loop | Opcode::AuditLog | Opcode::TritCount => 1,

            // Two sources (default for most operations)
            _ => 2,
        }
    }

    /// Returns true if this opcode requires Ring0 privilege
    pub fn requires_ring0(&self) -> bool {
        matches!(self,
            Opcode::Trap | Opcode::PrivEscalate | Opcode::DomainSet |
            Opcode::CapGrant | Opcode::CapRevoke | Opcode::MProtect |
            Opcode::IoRead | Opcode::IoWrite | Opcode::ReturnInt
        )
    }

    /// Returns true if this opcode accesses memory
    pub fn is_memory_op(&self) -> bool {
        matches!(self,
            Opcode::Load | Opcode::Store | Opcode::TLoad | Opcode::TStore |
            Opcode::LoadByte | Opcode::StoreByte | Opcode::Exchange |
            Opcode::CompareSwap | Opcode::AtomicAdd | Opcode::MemCopy |
            Opcode::TZeroize
        )
    }

    /// Returns true if this opcode is a ternary operation
    pub fn is_ternary(&self) -> bool {
        let v = self.to_u8();
        (v >= 0x10 && v <= 0x1F) || (v >= 0x60 && v <= 0x6F) || (v >= 0x70 && v <= 0x7F)
            || matches!(self, Opcode::TritCount | Opcode::TritBalance)
    }

    /// Returns true if this opcode modifies control flow
    pub fn is_branch(&self) -> bool {
        let v = self.to_u8();
        (v >= 0x30 && v <= 0x3F) || *self == Opcode::Halt
    }

    /// Returns true if this opcode is a crypto acceleration instruction
    pub fn is_crypto(&self) -> bool {
        let v = self.to_u8();
        (v >= 0x60 && v <= 0x6F) || (v >= 0x94 && v <= 0x97)
    }

    /// Returns true if this opcode is a SIMD/vector operation
    pub fn is_simd(&self) -> bool {
        let v = self.to_u8();
        v >= 0x70 && v <= 0x7F
    }

    /// Returns the opcode category name
    pub fn category(&self) -> &'static str {
        let v = self.to_u8();
        match v {
            0x00..=0x0F => "Arithmetic",
            0x10..=0x1F => "Ternary Core",

        }
    }

// ============================================================================
// Instruction structure (unchanged encoding, extended metadata)
// ============================================================================

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Instruction {
    pub opcode: Opcode,
    pub dst: u8,
    pub src1: u8,
    pub src2: u8,
    pub immediate: i64,
}

impl Instruction {
    pub fn new(opcode: Opcode, dst: u8, src1: u8, src2: u8, immediate: i64) -> Self {
        Self { opcode, dst, src1, src2, immediate }
    }

    pub fn from_opcode(opcode: Opcode) -> Self {
        Self { opcode, dst: 0, src1: 0, src2: 0, immediate: 0 }
    }

    /// Legacy 16-byte encoding (v1.0 compatible)
    pub fn encode(&self) -> Vec<u8> {
        let mut bytes = Vec::with_capacity(16);
        bytes.push(self.opcode.to_u8());
        bytes.push(self.dst);
        bytes.push(self.src1);
        bytes.push(self.src2);
        bytes.extend_from_slice(&self.immediate.to_le_bytes());
        bytes.extend_from_slice(&[0u8; 4]); // reserved
        bytes
    }

    /// Decode from legacy 16-byte encoding
    pub fn decode(bytes: &[u8]) -> VmResult<Self> {
        if bytes.len() < 16 {
            return Err(VmError::InvalidProgram(
                String::from("Instruction too short: need 16 bytes"),
            ));
        }
        let opcode = Opcode::from_u8(bytes[0])?;
        let dst = bytes[1];
        let src1 = bytes[2];
        let src2 = bytes[3];
        let mut imm_bytes = [0u8; 8];
        imm_bytes.copy_from_slice(&bytes[4..12]);
        let immediate = i64::from_le_bytes(imm_bytes);
        Ok(Self { opcode, dst, src1, src2, immediate })
    }

    /// Compact 4/6-byte encoding
    pub fn encode_compact(&self) -> Vec<u8> {
        let has_imm = self.immediate != 0;
        let packed: u16 = ((self.dst as u16 & 0x1F) << 11)
            | ((self.src1 as u16 & 0x1F) << 6)
            | ((self.src2 as u16 & 0x1F) << 1)
            | (if has_imm { 1 } else { 0 });

        let mut bytes = Vec::with_capacity(if has_imm { 6 } else { 4 });
        bytes.push(self.opcode.to_u8());
        bytes.extend_from_slice(&packed.to_le_bytes());

        if has_imm {
            let imm16 = self.immediate as i16;
            bytes.extend_from_slice(&imm16.to_le_bytes());
            bytes.push(0); // reserved
        } else {
            bytes.push(0); // reserved
        }
        bytes
    }

    /// Decode from compact encoding
    pub fn decode_compact(bytes: &[u8]) -> VmResult<(Self, usize)> {
        if bytes.len() < 4 {
            return Err(VmError::InvalidProgram(
                String::from("Compact instruction too short"),
            ));
        }
        let opcode = Opcode::from_u8(bytes[0])?;
        let packed = u16::from_le_bytes([bytes[1], bytes[2]]);
        let dst = ((packed >> 11) & 0x1F) as u8;
        let src1 = ((packed >> 6) & 0x1F) as u8;
        let src2 = ((packed >> 1) & 0x1F) as u8;
        let has_imm = (packed & 1) != 0;

        if has_imm {
            if bytes.len() < 6 {
                return Err(VmError::InvalidProgram(
                    String::from("Compact instruction with immediate too short"),
                ));
            }
            let imm16 = i16::from_le_bytes([bytes[3], bytes[4]]);
            Ok((Self { opcode, dst, src1, src2, immediate: imm16 as i64 }, 6))
        } else {
            Ok((Self { opcode, dst, src1, src2, immediate: 0 }, 4))
        }
    }

    /// Balanced Ternary 5/7-byte encoding
    pub fn encode_ternary(&self) -> Vec<u8> {
        let op = self.opcode.to_u8() as u32;
        let mut ternary_op = [0u8; 3];
        let mut val = op;
        for i in 0..9 {
            let trit = val % 3;
            let byte_idx = i / 4;
            let bit_idx = (i % 4) * 2;
            ternary_op[byte_idx] |= (trit as u8) << bit_idx;
            val /= 3;
        }

        let has_imm = self.immediate != 0;
        let packed: u16 = ((self.dst as u16 & 0x1F) << 11)
            | ((self.src1 as u16 & 0x1F) << 6)
            | ((self.src2 as u16 & 0x1F) << 1)
            | (if has_imm { 1 } else { 0 });

        let mut bytes = Vec::with_capacity(if has_imm { 7 } else { 5 });
        bytes.extend_from_slice(&ternary_op);
        bytes.extend_from_slice(&packed.to_le_bytes());

        if has_imm {
            let imm16 = self.immediate as i16;
            bytes.extend_from_slice(&imm16.to_le_bytes());
        }
        bytes
    }

    /// Decode from balanced ternary encoding
    pub fn decode_ternary(bytes: &[u8]) -> VmResult<(Self, usize)> {
        if bytes.len() < 5 {
            return Err(VmError::InvalidProgram(
                String::from("Ternary instruction too short"),
            ));
        }

        let mut op_val: u32 = 0;
        let mut multiplier: u32 = 1;
        for i in 0..9 {
            let byte_idx = i / 4;
            let bit_idx = (i % 4) * 2;
            let trit = ((bytes[byte_idx] >> bit_idx) & 0x03) as u32;
            op_val += trit * multiplier;
            multiplier *= 3;
        }

        let opcode = Opcode::from_u8(op_val as u8)?;
        let packed = u16::from_le_bytes([bytes[3], bytes[4]]);
        let dst = ((packed >> 11) & 0x1F) as u8;
        let src1 = ((packed >> 6) & 0x1F) as u8;
        let src2 = ((packed >> 1) & 0x1F) as u8;
        let has_imm = (packed & 1) != 0;

        if has_imm {
            if bytes.len() < 7 {
                return Err(VmError::InvalidProgram(
                    String::from("Ternary instruction with immediate too short"),
                ));
            }
            let imm16 = i16::from_le_bytes([bytes[5], bytes[6]]);
            Ok((Self { opcode, dst, src1, src2, immediate: imm16 as i64 }, 7))
        } else {
            Ok((Self { opcode, dst, src1, src2, immediate: 0 }, 5))
        }
    }

    /// Auto-detect and decode instruction format
    pub fn decode_auto(bytes: &[u8], format: InstructionFormat) -> VmResult<(Self, usize)> {
        match format {
            InstructionFormat::Legacy => {
                let inst = Self::decode(bytes)?;
                Ok((inst, 16))
            }
            InstructionFormat::Compact => Self::decode_compact(bytes),
            InstructionFormat::BalancedTernary => Self::decode_ternary(bytes),
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum InstructionFormat {
    Legacy,
    Compact,
    BalancedTernary,
}

// ============================================================================
// Program container
// ============================================================================

#[derive(Debug, Clone)]
pub struct Program {
    pub name: String,
    pub instructions: Vec<Instruction>,
    pub data_segment: Vec<u8>,
    pub entry_point: usize,
    pub isa_version: (u8, u8), // v2.0: track ISA version
}

impl Program {
    pub fn new(name: &str) -> Self {
        Self {
            name: String::from(name),
            instructions: Vec::new(),
            data_segment: Vec::new(),
            entry_point: 0,
            isa_version: (2, 0),
        }
    }

    pub fn add_instruction(&mut self, inst: Instruction) {
        self.instructions.push(inst);
    }

    pub fn get_instruction(&self, index: usize) -> Option<&Instruction> {
        self.instructions.get(index)
    }

    pub fn instruction_count(&self) -> usize {
        self.instructions.len()
    }

    pub fn set_entry_point(&mut self, ep: usize) {
        self.entry_point = ep;
    }

    pub fn set_data(&mut self, data: Vec<u8>) {
        self.data_segment = data;
    }

    pub fn validate(&self) -> VmResult<()> {
        if self.instructions.is_empty() {
            return Err(VmError::InvalidProgram(
                String::from("Program has no instructions"),
            ));
        }
        for (i, inst) in self.instructions.iter().enumerate() {
            if inst.dst > 26 {
                return Err(VmError::InvalidProgram(
                    alloc::format!("Instruction {} has invalid dst register: {}", i, inst.dst),
                ));
            }
            if inst.src1 > 26 {
                return Err(VmError::InvalidProgram(
                    alloc::format!("Instruction {} has invalid src1 register: {}", i, inst.src1),
                ));
            }
            if inst.src2 > 26 {
                return Err(VmError::InvalidProgram(
                    alloc::format!("Instruction {} has invalid src2 register: {}", i, inst.src2),
                ));
            }
        }
        Ok(())
    }
}

    #[test]
    fn test_encoding_backward_compat() {
        // v1.0 programs should decode identically
        let inst = Instruction::new(Opcode::Add, 0, 1, 2, 42);
        let encoded = inst.encode();
        let decoded = Instruction::decode(&encoded).unwrap();
        assert_eq!(decoded.opcode, Opcode::Add);
        assert_eq!(decoded.dst, 0);
        assert_eq!(decoded.src1, 1);
        assert_eq!(decoded.src2, 2);
        assert_eq!(decoded.immediate, 42);
    }

    #[test]
    fn test_new_opcodes_encode_decode() {
        let new_ops = [
            Opcode::Abs, Opcode::MulAdd, Opcode::CompareSwap, Opcode::JumpGe,
            Opcode::Select, Opcode::RotateLeft, Opcode::TLatticeMul,
            Opcode::TPermute, Opcode::AuditLog, Opcode::ConstTimeEq,
            Opcode::DebugBreak, Opcode::AssertTrit,
        ];
        for op in &new_ops {
            let inst = Instruction::new(*op, 3, 7, 11, 0);
            let encoded = inst.encode();
            let decoded = Instruction::decode(&encoded).unwrap();
            assert_eq!(decoded.opcode, *op, "Failed for {:?}", op);
        }
    }

    #[test]
    fn test_compact_encoding_new_opcodes() {
        let inst = Instruction::new(Opcode::MulAdd, 5, 10, 15, 0);
        let bytes = inst.encode_compact();
        let (decoded, consumed) = Instruction::decode_compact(&bytes).unwrap();
        assert_eq!(consumed, 4);
        assert_eq!(decoded.opcode, Opcode::MulAdd);
        assert_eq!(decoded.dst, 5);
    }
