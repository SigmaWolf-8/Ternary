//! # The Ternary Circle
//!
//! Foundational geometric constants for the Salvi Framework's native
//! angular system — the **canonical ternary circle** where geometry
//! speaks base-3 as its mother tongue.
//!
//! ## The Axiom
//!
//! A full circle is **364 degrees** = `111111₃` (a base-3 repunit of six 1's).
//! The ratio of circumference to diameter is **π = 14**.
//! The radian — the angle subtending an arc equal to the radius — is
//! **exactly 13 degrees** = `111₃`, the seventh Tribonacci number (T₇).
//!
//! These are not independent choices. They are bound:
//!   - C = πd = 14d, so C/r = 28
//!   - Full circle = 28 radians = 364°
//!   - 1 radian = 364/28 = 13°
//!
//! ## The Cyclic Group Z₂₈
//!
//! Because 364 = 28 × 13, the set {0°, 13°, 26°, …, 351°} modulo 364°
//! forms a **finite cyclic group of order 28**. The Tribonacci word,
//! read modulo 28, produces a dense subset that covers all residues.
//! This replaces the 3-fold (120°) symmetry of the conventional system
//! with 28-fold symmetry — the walk on Z₂₈ with τ-scaling.

use std::f64::consts::PI as STD_PI;

// ══════════════════════════════════════════════════════════════
// TERNARY CIRCLE CONSTANTS
// ══════════════════════════════════════════════════════════════

/// Full circle in the ternary angular system: **364 degrees**.
///
/// In base 3: `111111₃` = (3⁶ − 1) / 2 = 364.
/// This is a **six-digit repunit** — the circle IS a base-3 repunit.
pub const FULL_CIRCLE_DEG: f64 = 364.0;

/// Full circle as a base-3 repunit string.
pub const FULL_CIRCLE_BASE3: &str = "111111";

/// π in the ternary circle system: **exactly 14**.
///
/// This is the ratio of circumference to diameter in the ternary geometry.
/// 14 = 112₃ in base 3.
pub const PI_TERNARY: f64 = 14.0;

/// 2π in the ternary system: **28**.
///
/// A full circle is 28 ternary radians.
/// 28 = 1001₃ in base 3.
pub const TWO_PI_TERNARY: f64 = 28.0;

/// One ternary radian in degrees: **exactly 13°**.
///
/// 13 = `111₃`, a three-digit base-3 repunit.
/// 13 is also T₇, the seventh Tribonacci number.
/// The radian is simultaneously a Tribonacci number and a ternary repunit.
pub const RADIAN_DEG: f64 = 13.0;

/// One ternary radian as a base-3 repunit string.
pub const RADIAN_BASE3: &str = "111";

/// Order of the cyclic group Z₂₈ generated by the ternary radian.
///
/// The set {0·13°, 1·13°, 2·13°, …, 27·13°} modulo 364° cycles through
/// exactly 28 distinct positions before returning to the origin.
pub const CYCLIC_ORDER: u32 = 28;

/// Number of ternary radians in a full circle.
pub const RADIANS_PER_CIRCLE: u32 = 28;

// ══════════════════════════════════════════════════════════════
// TRIBONACCI CONSTANTS IN THE TERNARY CIRCLE
// ══════════════════════════════════════════════════════════════

/// The Tribonacci constant τ ≈ 1.839286755214161.
///
/// Real root of x³ = x² + x + 1. The native irrational of ternary
/// recursion, analogous to φ in binary/Fibonacci systems.
pub const TAU_TRIBONACCI: f64 = 1.839286755214161;

/// τ² ≈ 3.38297576790891.
pub const TAU_SQUARED: f64 = 3.38297576790891;

/// τ³ = τ² + τ + 1 ≈ 6.22226252312307.
pub const TAU_CUBED: f64 = 6.22226252312307;

/// Tribonacci golden angle (native ternary): 364° / τ³ ≈ 58.50°.
///
/// This is the smallest irrational rotation derived from τ that yields
/// optimal aperiodic packing in the ternary circle — the analogue of
/// the golden angle (360°/φ²) in phyllotaxis.
pub const TRIBONACCI_GOLDEN_ANGLE_DEG: f64 = 58.50438656;

/// Classical golden angle translated into the ternary circle:
/// 364° / φ² where φ = (1+√5)/2.
///
/// Factors as 2 · 7 · 13 · (3 − √5)°, where 2, 7, 13 are Tribonacci numbers.
pub const GOLDEN_ANGLE_TERNARY_DEG: f64 = 138.98056;

// ══════════════════════════════════════════════════════════════
// WALK INSTRUCTION SET ON Z₂₈
// ══════════════════════════════════════════════════════════════

/// Walk instruction for trit digit 0: **0 radians** (no turn, step forward).
/// The walker continues in the current direction.
pub const WALK_TURN_0: f64 = 0.0;

/// Walk instruction for trit digit 1: **1 ternary radian = 13°**.
/// Turn left by one radian, then step.
pub const WALK_TURN_1: f64 = 13.0;

/// Walk instruction for trit digit 2: **2 ternary radians = 26°**.
/// Turn left by two radians, then step.
/// (Equivalently, turn right by 364° − 26° = 338° in the ternary circle,
/// or right by 26 positions in Z₂₈.)
pub const WALK_TURN_2: f64 = 26.0;

// ══════════════════════════════════════════════════════════════
// CONVERSION FUNCTIONS
// ══════════════════════════════════════════════════════════════

/// Convert ternary degrees (364° full circle) to conventional degrees (360°).
///
/// This is the bridge between the ternary and conventional worlds.
/// Use this only when interfacing with standard geometry libraries.
#[inline]
pub fn ternary_deg_to_std_deg(ternary_deg: f64) -> f64 {
    ternary_deg * (360.0 / FULL_CIRCLE_DEG)
}

/// Convert conventional degrees (360° full circle) to ternary degrees (364°).
#[inline]
pub fn std_deg_to_ternary_deg(std_deg: f64) -> f64 {
    std_deg * (FULL_CIRCLE_DEG / 360.0)
}

/// Convert ternary radians (28 per circle) to standard radians (2π per circle).
#[inline]
pub fn ternary_rad_to_std_rad(ternary_rad: f64) -> f64 {
    ternary_rad * (2.0 * STD_PI / TWO_PI_TERNARY)
}

/// Convert standard radians to ternary radians.
#[inline]
pub fn std_rad_to_ternary_rad(std_rad: f64) -> f64 {
    std_rad * (TWO_PI_TERNARY / (2.0 * STD_PI))
}

/// Convert ternary degrees to ternary radians.
#[inline]
pub fn ternary_deg_to_ternary_rad(deg: f64) -> f64 {
    deg / RADIAN_DEG
}

/// Convert ternary radians to ternary degrees.
#[inline]
pub fn ternary_rad_to_ternary_deg(rad: f64) -> f64 {
    rad * RADIAN_DEG
}

/// Convert a trit digit (0, 1, or 2) to its walk angle in ternary degrees.
#[inline]
pub fn trit_to_walk_angle_deg(trit: u8) -> f64 {
    debug_assert!(trit <= 2, "Trit must be 0, 1, or 2; got {}", trit);
    trit as f64 * RADIAN_DEG
}

/// Convert a trit digit to its walk angle in standard radians.
///
/// This is the function you call when feeding angles to `cos()`/`sin()`.
#[inline]
pub fn trit_to_std_rad(trit: u8) -> f64 {
    ternary_rad_to_std_rad(trit as f64)
}

// ══════════════════════════════════════════════════════════════
// Z₂₈ CYCLIC GROUP OPERATIONS
// ══════════════════════════════════════════════════════════════

/// A position in the cyclic group Z₂₈.
///
/// Represents one of the 28 discrete angular positions in the ternary
/// circle, separated by 13° each.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub struct Z28(pub u8);

impl Z28 {
    /// Create a new Z₂₈ element, reducing modulo 28.
    pub fn new(val: u32) -> Self {
        Z28((val % 28) as u8)
    }

    /// The identity element (0 position).
    pub fn zero() -> Self {
        Z28(0)
    }

    /// Add two elements in Z₂₈ (group operation).
    pub fn add(self, other: Z28) -> Z28 {
        Z28(((self.0 as u32 + other.0 as u32) % 28) as u8)
    }

    /// Subtract (inverse addition) in Z₂₈.
    pub fn sub(self, other: Z28) -> Z28 {
        Z28(((self.0 as u32 + 28 - other.0 as u32) % 28) as u8)
    }

    /// Negate (additive inverse) in Z₂₈.
    pub fn neg(self) -> Z28 {
        if self.0 == 0 { Z28(0) } else { Z28(28 - self.0) }
    }

    /// The ternary degree value of this position.
    pub fn to_ternary_deg(self) -> f64 {
        self.0 as f64 * RADIAN_DEG
    }

    /// The standard radian value (for trigonometric functions).
    pub fn to_std_rad(self) -> f64 {
        tern_rad_to_std_rad_inner(self.0 as f64)
    }

    /// Advance by a trit instruction (0, 1, or 2 ternary radians).
    pub fn step(self, trit: u8) -> Z28 {
        debug_assert!(trit <= 2);
        self.add(Z28(trit))
    }

    /// Check if this position is reachable from the origin by
    /// accumulating Tribonacci word trit values.
    pub fn is_tribonacci_reachable(&self) -> bool {
        // All 28 positions are reachable because the Tribonacci word
        // eventually covers all residues mod 28.
        // This is provable: the word contains all three digits {0,1,2}
        // and gcd(1, 28) = 1, so the generated subgroup is all of Z₂₈.
        true
    }
}

// Internal helper to avoid name collision.
fn tern_rad_to_std_rad_inner(ternary_rad: f64) -> f64 {
    ternary_rad * (2.0 * STD_PI / TWO_PI_TERNARY)
}

impl std::fmt::Display for Z28 {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        write!(f, "Z₂₈({})", self.0)
    }
}

// ══════════════════════════════════════════════════════════════
// WALK ENGINE
// ══════════════════════════════════════════════════════════════

/// A point in the Tribonacci radian spiral.
#[derive(Clone, Debug)]
pub struct SpiralPoint {
    /// X coordinate in the complex plane.
    pub x: f64,
    /// Y coordinate in the complex plane.
    pub y: f64,
    /// The angular position in Z₂₈ at this step.
    pub position: Z28,
    /// The trit digit that generated this step.
    pub trit: u8,
    /// The step index.
    pub step: usize,
}

/// Walk the Tribonacci radian spiral.
///
/// Given a sequence of trit digits (from the Tribonacci word or the
/// base-3 expansion of τ), compute the spiral path:
///
///   z_n = Σ(k=1..n) e^(i · w_k · 13°) / τ^k
///
/// where w_k ∈ {0, 1, 2} is the k-th trit, 13° is the ternary radian,
/// and τ is the Tribonacci constant (radial scaling).
///
/// The direction at each step is an exact integer multiple of 13° —
/// lattice-like on the 28 rays of Z₂₈, scaled by powers of 1/τ.
pub fn walk_tribonacci_radian_spiral(trits: &[u8]) -> Vec<SpiralPoint> {
    let mut points = Vec::with_capacity(trits.len() + 1);
    let mut x = 0.0_f64;
    let mut y = 0.0_f64;
    let mut direction = Z28::zero();
    let mut tau_power = 1.0_f64; // τ^0 = 1

    // Origin point.
    points.push(SpiralPoint {
        x: 0.0,
        y: 0.0,
        position: direction,
        trit: 0,
        step: 0,
    });

    for (k, &trit) in trits.iter().enumerate() {
        debug_assert!(trit <= 2, "Trit must be 0, 1, or 2; got {}", trit);

        // Advance direction by trit ternary radians in Z₂₈.
        direction = direction.step(trit);

        // Scale down by τ at each step.
        tau_power *= TAU_TRIBONACCI;

        // Convert the Z₂₈ direction to a standard radian angle
        // for the cos/sin computation.
        let angle_std_rad = direction.to_std_rad();

        // Step in the current direction, scaled by 1/τ^(k+1).
        let step_len = 1.0 / tau_power;
        x += angle_std_rad.cos() * step_len;
        y += angle_std_rad.sin() * step_len;

        points.push(SpiralPoint {
            x,
            y,
            position: direction,
            trit,
            step: k + 1,
        });
    }

    points
}

// ══════════════════════════════════════════════════════════════
// REPUNIT VERIFICATION
// ══════════════════════════════════════════════════════════════

/// Verify that a number is a base-3 repunit (all 1's in base 3).
///
/// A repunit in base b is (b^n - 1) / (b - 1).
/// For base 3: 1, 4, 13, 40, 121, 364, 1093, ...
/// (That's 1₃, 11₃, 111₃, 1111₃, 11111₃, 111111₃, ...)
pub fn is_base3_repunit(n: u64) -> bool {
    if n == 0 {
        return false;
    }
    // Check: n = (3^k - 1) / 2 for some k ≥ 1.
    // Equivalently: 2n + 1 = 3^k.
    let m = 2 * n + 1;
    let mut v = m;
    while v > 1 {
        if v % 3 != 0 {
            return false;
        }
        v /= 3;
    }
    v == 1
}

/// Return the repunit order (number of 1's) if the value is a base-3 repunit.
pub fn base3_repunit_order(n: u64) -> Option<u32> {
    if !is_base3_repunit(n) {
        return None;
    }
    let mut m = 2 * n + 1;
    let mut order = 0u32;
    while m > 1 {
        m /= 3;
        order += 1;
    }
    Some(order)
}

#[cfg(test)]
mod tests {
    use super::*;

    // ── Constant verification ─────────────────────────────────

    #[test]
    fn test_full_circle_is_repunit() {
        assert!(is_base3_repunit(364));
        assert_eq!(base3_repunit_order(364), Some(6));
    }

    #[test]
    fn test_radian_is_repunit() {
        assert!(is_base3_repunit(13));
        assert_eq!(base3_repunit_order(13), Some(3));
    }

    #[test]
    fn test_radian_is_tribonacci_t7() {
        // T₇ = 13, which is also 111₃.
        assert_eq!(RADIAN_DEG as u64, 13);
    }

    #[test]
    fn test_full_circle_equals_28_radians() {
        assert_eq!(FULL_CIRCLE_DEG, RADIAN_DEG * TWO_PI_TERNARY);
        assert_eq!(364.0, 13.0 * 28.0);
    }

    #[test]
    fn test_pi_ternary() {
        // π_τ = 14, and C = π·d. With C/r = 28, d = 2r, so C = 28r = 14·d. ✓
        assert_eq!(PI_TERNARY, 14.0);
        assert_eq!(TWO_PI_TERNARY, 28.0);
    }

    #[test]
    fn test_repunit_sequence() {
        // Base-3 repunits: 1, 4, 13, 40, 121, 364, 1093
        let repunits = [1u64, 4, 13, 40, 121, 364, 1093];
        for &r in &repunits {
            assert!(is_base3_repunit(r), "{} should be a base-3 repunit", r);
        }
        assert!(!is_base3_repunit(2));
        assert!(!is_base3_repunit(3));
        assert!(!is_base3_repunit(5));
        assert!(!is_base3_repunit(14));
    }

    // ── Conversion round-trips ────────────────────────────────

    #[test]
    fn test_ternary_to_std_deg_full_circle() {
        let std = ternary_deg_to_std_deg(FULL_CIRCLE_DEG);
        assert!((std - 360.0).abs() < 1e-10, "364 ternary degrees should map to 360 standard degrees");
    }

    #[test]
    fn test_std_to_ternary_deg_full_circle() {
        let tern = std_deg_to_ternary_deg(360.0);
        assert!((tern - FULL_CIRCLE_DEG).abs() < 1e-10);
    }

    #[test]
    fn test_ternary_rad_to_std_rad_full_circle() {
        let std = ternary_rad_to_std_rad(TWO_PI_TERNARY);
        assert!((std - 2.0 * STD_PI).abs() < 1e-10, "28 ternary radians should map to 2π standard radians");
    }

    #[test]
    fn test_radian_conversion() {
        // 1 ternary radian = 13 ternary degrees.
        assert_eq!(ternary_rad_to_ternary_deg(1.0), RADIAN_DEG);
        assert_eq!(ternary_deg_to_ternary_rad(RADIAN_DEG), 1.0);
    }

    // ── Z₂₈ group operations ─────────────────────────────────

    #[test]
    fn test_z28_closure() {
        for a in 0..28u8 {
            for b in 0..28u8 {
                let sum = Z28(a).add(Z28(b));
                assert!(sum.0 < 28, "Z₂₈ addition must stay in [0, 28)");
            }
        }
    }

    #[test]
    fn test_z28_identity() {
        for a in 0..28u8 {
            assert_eq!(Z28(a).add(Z28::zero()), Z28(a));
        }
    }

    #[test]
    fn test_z28_inverse() {
        for a in 0..28u8 {
            let inv = Z28(a).neg();
            assert_eq!(Z28(a).add(inv), Z28::zero(),
                "Z₂₈({}) + Z₂₈({}) should be identity", a, inv.0);
        }
    }

    #[test]
    fn test_z28_order() {
        // The generator 1 should have order 28.
        let gen = Z28(1);
        let mut current = Z28::zero();
        for i in 1..=28 {
            current = current.add(gen);
            if i < 28 {
                assert_ne!(current, Z28::zero(),
                    "Generator should not return to identity after {} steps", i);
            }
        }
        assert_eq!(current, Z28::zero(), "Generator should complete the cycle in 28 steps");
    }

    #[test]
    fn test_z28_step_with_trits() {
        // Starting from 0, stepping by trits 0,1,2 should give 0,1,2.
        let origin = Z28::zero();
        assert_eq!(origin.step(0), Z28(0));
        assert_eq!(origin.step(1), Z28(1));
        assert_eq!(origin.step(2), Z28(2));

        // Accumulated steps: 0 → +2 → +1 → +0 = position 3.
        let p = Z28::zero().step(2).step(1).step(0);
        assert_eq!(p, Z28(3));
    }

    #[test]
    fn test_z28_trit_coverage() {
        // Verify that accumulating trit values {0,1,2} can reach all 28 positions.
        // Since gcd(1, 28) = 1 and trit 1 is the generator, all positions are reachable.
        let mut reached = [false; 28];
        let mut pos = Z28::zero();
        // Walk with just trit=1 for 28 steps.
        for _ in 0..28 {
            reached[pos.0 as usize] = true;
            pos = pos.step(1);
        }
        for (i, &r) in reached.iter().enumerate() {
            assert!(r, "Position {} should be reachable", i);
        }
    }

    // ── Spiral walk ───────────────────────────────────────────

    #[test]
    fn test_walk_starts_at_origin() {
        let trits = vec![1, 0, 2, 1];
        let points = walk_tribonacci_radian_spiral(&trits);
        assert_eq!(points[0].x, 0.0);
        assert_eq!(points[0].y, 0.0);
        assert_eq!(points.len(), trits.len() + 1);
    }

    #[test]
    fn test_walk_directions_are_z28() {
        let trits = vec![0, 1, 2, 0, 1, 1, 2, 0, 2, 1];
        let points = walk_tribonacci_radian_spiral(&trits);
        // Accumulated direction should match manual Z₂₈ computation.
        let mut dir = Z28::zero();
        for (i, &trit) in trits.iter().enumerate() {
            dir = dir.step(trit);
            assert_eq!(points[i + 1].position, dir,
                "Direction mismatch at step {}", i + 1);
        }
    }

    #[test]
    fn test_walk_scaling_by_tau() {
        // Each step should be smaller by factor τ.
        let trits = vec![1, 1, 1, 1, 1];
        let points = walk_tribonacci_radian_spiral(&trits);
        // The step length at step k is 1/τ^k.
        // We can verify by checking that the distance between consecutive
        // points decreases by ~1/τ each time.
        for k in 2..points.len() {
            let dx1 = points[k].x - points[k - 1].x;
            let dy1 = points[k].y - points[k - 1].y;
            let len1 = (dx1 * dx1 + dy1 * dy1).sqrt();

            let dx0 = points[k - 1].x - points[k - 2].x;
            let dy0 = points[k - 1].y - points[k - 2].y;
            let len0 = (dx0 * dx0 + dy0 * dy0).sqrt();

            if len0 > 1e-12 {
                let ratio = len0 / len1;
                assert!(
                    (ratio - TAU_TRIBONACCI).abs() < 0.01,
                    "Step length ratio at k={} should be τ ≈ 1.839, got {:.4}",
                    k, ratio
                );
            }
        }
    }

    // ── Walk angles are integer multiples of 13° ──────────────

    #[test]
    fn test_walk_angles_are_lattice() {
        let trits = vec![0, 1, 2, 0, 1, 2, 1, 0, 2, 2, 1, 0];
        let points = walk_tribonacci_radian_spiral(&trits);

        for p in &points[1..] {
            let deg = p.position.to_ternary_deg();
            // Every angle must be an exact integer multiple of 13°.
            let remainder = deg % RADIAN_DEG;
            assert!(
                remainder.abs() < 1e-10 || (RADIAN_DEG - remainder).abs() < 1e-10,
                "Walk angle {} is not a multiple of 13°", deg
            );
        }
    }
}
