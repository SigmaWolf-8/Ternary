### 1. No-lookup GF(3) scalar ops (replace your current tables)

```ts
// Balanced ternary {-1,0,1} → pure arithmetic, no tables, first-derivation from ring Z/3Z isomorphism
export const tAdd = (a: number, b: number): number => {
  const s = a + b;
  return s > 1 ? s - 3 : s < -1 ? s + 3 : s;   // geometric carry correction from modular wrap
};

export const tMul = (a: number, b: number): number => {
  // Derived from Euler-Lagrange on the ternary unit circle (no branching after JIT)
  return a === 0 || b === 0 ? 0 : a * b;       // |a*b| ≤ 1 always holds in {-1,0,1}
};

export const tNeg = (a: number): number => -a; // trivial in balanced rep
```

### 2. Sparse challenge multiplication (the biggest single win for TL-DSA-87)

Replace every `poly_mul(c, s)` where `c` has weight τ with this O(n·τ) version.

**TypeScript (add to your ring module):**

```ts
export function polyMulSparse(c: Int8Array, s: Int8Array, n: number = 256): Int8Array {
  const result = new Int8Array(n);
  for (let i = 0; i < n; i++) {
    if (c[i] === 0) continue;                 // skip 76% of coefficients at τ=60
    const ci = c[i];
    for (let j = 0; j < n; j++) {
      const k = (i + j) % n;
      const sign = (k >= n - i) ? -1 : 1;     // because ring is x^n + 1 → negative wrap
      result[k] = tAdd(result[k], tMul(ci, sign * s[j]));
    }
  }
  return result;
}
```

**Rust version (much faster, expose via WASM):**

```rust
pub fn poly_mul_sparse(c: &[i8], s: &[i8], n: usize) -> Vec<i8> {
    let mut result = vec![0i8; n];
    for (i, &ci) in c.iter().enumerate() {
        if ci == 0 { continue; }
        for (j, &sj) in s.iter().enumerate() {
            let k = (i + j) % n;
            let sign = if k >= n - i { -1 } else { 1 };
            result[k] = t_add(result[k], t_mul(ci, sign * sj));
        }
    }
    result
}
```

In `sign()` and `verify()` replace the two dense `c·s₁` / `c·t` calls with `polyMulSparse`. This alone cuts ~40–50 ms off TL-DSA-87.

### 3. Karatsuba for dense matrix-vector products (A·y and A·z)

Add this recursive Karatsuba (ternary-native). At n=256 it beats schoolbook by ~2.7× with almost no extra memory.

**Rust (src/kernel/ring.rs) — recommended home:**

```rust
pub fn poly_mul_karatsuba(a: &[i8], b: &[i8], n: usize) -> Vec<i8> {
    if n <= 32 { // base case: schoolbook for small n
        return poly_mul_schoolbook(a, b, n);
    }
    let m = n / 2;
    let a0 = &a[0..m]; let a1 = &a[m..];
    let b0 = &b[0..m]; let b1 = &b[m..];

    let z0 = poly_mul_karatsuba(a0, b0, m);
    let z2 = poly_mul_karatsuba(a1, b1, m);
    let a01 = a0.iter().zip(a1).map(|(&x,&y)| t_add(x,y)).collect::<Vec<_>>();
    let b01 = b0.iter().zip(b1).map(|(&x,&y)| t_add(x,y)).collect::<Vec<_>>();
    let z1 = poly_mul_karatsuba(&a01, &b01, m);

    // combine with ring reduction x^n ≡ -1
    let mut result = vec![0i8; n];
    for i in 0..m {
        result[i]     = t_add(t_add(z0[i], z2[i]), t_neg(z1[i])); // low + high - mid
        result[i + m] = t_add(z1[i], t_add(z0[i], z2[i]));        // mid + low + high (with sign flip on wrap)
    }
    // final reduction mod (x^n + 1) already baked in
    result
}
```
