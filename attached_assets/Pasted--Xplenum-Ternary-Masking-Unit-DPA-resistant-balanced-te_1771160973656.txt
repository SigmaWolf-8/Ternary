// Xplenum Ternary Masking Unit — DPA-resistant balanced ternary masking
// Trit encoding: 00=0, 01=+1, 10=-1, 11=invalid
`include "xplenum_pkg.vh"
module xplenum_mask_unit (
    input wire clk, input wire rst_n, input wire enable,
    input wire [6:0] funct7, input wire [31:0] rs1_data, input wire [31:0] rs2_data,
    output reg [31:0] rd_data, output reg valid, output reg exception, output reg [3:0] exc_cause,
    input wire [31:0] mask_seed, output reg [31:0] mask_state
);
    reg [31:0] lfsr;
    function [1:0] trit_add; input [1:0] a, b;
        begin case ({a, b})
            4'b0000: trit_add=2'b00; 4'b0001: trit_add=2'b01; 4'b0010: trit_add=2'b10;
            4'b0100: trit_add=2'b01; 4'b0101: trit_add=2'b10; 4'b0110: trit_add=2'b00;
            4'b1000: trit_add=2'b10; 4'b1001: trit_add=2'b00; 4'b1010: trit_add=2'b01;
            default: trit_add=2'b11;
        endcase end
    endfunction
    function [1:0] trit_sub; input [1:0] a, b;
        begin case (b)
            2'b00: trit_sub=trit_add(a,2'b00); 2'b01: trit_sub=trit_add(a,2'b10);
            2'b10: trit_sub=trit_add(a,2'b01); default: trit_sub=2'b11;
        endcase end
    endfunction
    function invalid_trits; input [31:0] word; integer i;
        begin invalid_trits=0;
            for(i=0;i<16;i=i+1) if(word[i*2+:2]==2'b11) invalid_trits=1;
        end
    endfunction
    function [31:0] mask_word; input [31:0] d,m; integer i;
        begin for(i=0;i<16;i=i+1) mask_word[i*2+:2]=trit_add(d[i*2+:2],m[i*2+:2]); end
    endfunction
    function [31:0] unmask_word; input [31:0] d,m; integer i;
        begin for(i=0;i<16;i=i+1) unmask_word[i*2+:2]=trit_sub(d[i*2+:2],m[i*2+:2]); end
    endfunction
    task advance_lfsr; integer i;
        begin for(i=0;i<32;i=i+1) lfsr={lfsr[30:0],lfsr[31]^lfsr[21]^lfsr[1]^lfsr[0]}; end
    endtask
    function [31:0] lfsr_to_trits; input [31:0] raw; integer i;
        begin for(i=0;i<16;i=i+1) lfsr_to_trits[i*2+:2]=(raw[i*2+:2]==2'b11)?2'b00:raw[i*2+:2]; end
    endfunction
    always @(posedge clk or negedge rst_n) begin
        if(!rst_n) begin rd_data<=0;valid<=0;exception<=0;exc_cause<=`XPEXC_NONE;lfsr<=32'hDEADBEEF;mask_state<=0;
        end else begin
            valid<=0; exception<=0; exc_cause<=`XPEXC_NONE;
            if(enable) case(funct7)
                `F7_TMASK_APPLY: if(invalid_trits(rs1_data)||invalid_trits(rs2_data))
                    begin exception<=1;exc_cause<=`XPEXC_MASK_FAULT; end
                    else begin rd_data<=mask_word(rs1_data,rs2_data);mask_state<=rs2_data;valid<=1; end
                `F7_TUNMASK: if(invalid_trits(rs1_data)||invalid_trits(rs2_data))
                    begin exception<=1;exc_cause<=`XPEXC_MASK_FAULT; end
                    else begin rd_data<=unmask_word(rs1_data,rs2_data);mask_state<=0;valid<=1; end
                `F7_TMASK_RNG: if(invalid_trits(rs1_data))
                    begin exception<=1;exc_cause<=`XPEXC_MASK_FAULT; end
                    else begin rd_data<=mask_word(rs1_data,lfsr_to_trits(lfsr));mask_state<=lfsr_to_trits(lfsr);advance_lfsr;valid<=1; end
                `F7_TMASK_REFRESH: begin
                    rd_data<=mask_word(unmask_word(rs1_data,mask_state),lfsr_to_trits(lfsr));
                    mask_state<=lfsr_to_trits(lfsr);advance_lfsr;valid<=1; end
                default: begin exception<=1;exc_cause<=`XPEXC_MASK_FAULT; end
            endcase
            if(mask_seed!=0) lfsr<=mask_seed;
        end
    end
endmodule
