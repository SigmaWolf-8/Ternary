```ts
// server/db/schema.ts
import { pgTable, text, timestamp, boolean, jsonb, uuid, integer } from 'drizzle-orm/pg-core';
import { relations } from 'drizzle-orm';

export const apiKeys = pgTable('api_keys', {
  id: uuid('id').primaryKey().defaultRandom(),
  keyHash: text('key_hash').notNull().unique(),  // Argon2 or SHA-256
  name: text('name').notNull(),
  owner: text('owner').notNull(),
  scopes: jsonb('scopes').notNull().$type<string[]>(),
  createdAt: timestamp('created_at').defaultNow().notNull(),
  expiresAt: timestamp('expires_at'),
  revokedAt: timestamp('revoked_at'),
  lastUsedAt: timestamp('last_used_at'),
  usageCount: integer('usage_count').default(0),
  
  // Priority 1: Rotation fields
  rotationScheduledAt: timestamp('rotation_scheduled_at'),  // When next rotation happens
  previousKeyHash: text('previous_key_hash'),  // For dual-key overlap
  isRotated: boolean('is_rotated').default(false),
  
  // Priority 2: Per-key limits (jsonb for flexibility)
  limits: jsonb('limits').$type<{ tier: 'research' | 'pro' | 'admin'; rpm: number; daily: number }>()
    .default({ tier: 'research', rpm: 100, daily: 5000 }),
});

export const apiKeyLogs = pgTable('api_key_logs', { /* your existing */ });

// Relations (for queries)
export const apiKeysRelations = relations(apiKeys, ({ many }) => ({
  logs: many(apiKeyLogs),
}));
```

### **Priority 1: Automated Key Rotation & Dual-Key Overlap**

#### 1.1: Extended Service (`server/services/api-key.service.ts`)

```ts
// server/services/api-key.service.ts
import { randomBytes } from 'crypto';
import { hash, verify } from 'argon2';
import { db } from '../db';
import { apiKeys } from '../db/schema';
import { eq, and, lt, isNull, desc } from 'drizzle-orm';
import cron from 'node-cron';

export const apiKeyService = {
  // ... your existing generate, validate, etc.

  /** Schedule rotation for a key (called on generation or manually) */
  async scheduleRotation(keyId: string, daysUntil = 90) {
    const scheduledAt = new Date(Date.now() + daysUntil * 86400000);
    await db.update(apiKeys)
      .set({ rotationScheduledAt: scheduledAt })
      .where(eq(apiKeys.id, keyId));
    return scheduledAt;
  },

  /** Generate new key as rotation (dual-key overlap) */
  async rotateKey(oldKeyId: string, newName?: string) {
    const [oldKey] = await db.select().from(apiKeys).where(eq(apiKeys.id, oldKeyId));
    if (!oldKey) throw new Error('Key not found');

    // Generate new key with same owner/scopes
    const newKeyData = await this.generate(oldKey.owner, newName || `${oldKey.name} (v2)`, oldKey.scopes, 90);
    
    // Link to old (for overlap)
    await db.update(apiKeys)
      .set({ 
        previousKeyHash: oldKey.keyHash, 
        isRotated: true 
      })
      .where(eq(apiKeys.id, newKeyData.id));  // Assuming generate returns id now

    // Schedule revocation of old after 7-day grace
    setTimeout(async () => {
      await this.revoke(oldKeyId);
    }, 7 * 86400000);

    return { newKey: newKeyData, oldKeyId };
  },

  /** Revoke (existing, but enhanced) */
  async revoke(keyId: string) {
    await db.update(apiKeys)
      .set({ revokedAt: new Date() })
      .where(eq(apiKeys.id, keyId));
  },

  // Cron job setup (call once in server startup)
  startRotationCron() {
    cron.schedule('0 0 * * *', async () => {  // Daily at midnight
      console.log('Running key rotation check...');
      const dueKeys = await db.select()
        .from(apiKeys)
        .where(and(
          isNull(apiKeys.revokedAt),
          lt(apiKeys.rotationScheduledAt, new Date())
        ));

      for (const key of dueKeys) {
        await this.rotateKey(key.id);
        console.log(`Rotated key: ${key.id}`);
      }
    });
  },
};
```

```ts
    ))

});
```

#### 2.2: Rate Limiting Middleware (`server/middleware/rate-limit.middleware.ts`)

New/updated file (extend your existing one).

```ts
// server/middleware/rate-limit.middleware.ts
import { rateLimit } from 'express-rate-limit';
import { Request, Response, NextFunction } from 'express';
import { db } from '../db';
import { apiKeys } from '../db/schema';
import { eq } from 'drizzle-orm';

// Dynamic per-key limiter factory
export const createKeyLimiter = () => rateLimit({
  windowMs: 60 * 1000,  // 1 min
  max: (req: Request) => {
    const key = (req as any).apiKey;  // From apiKeyAuth
    return key?.limits?.rpm || 100;  // Default
  },
  keyGenerator: (req: Request) => {
    const key = (req as any).apiKey;
    return key?.id || req.ip;  // Fallback to IP if no key (legacy)
  },
  handler: (req: Request, res: Response) => {
    res.status(429).json({ 
      error: 'Rate limit exceeded for this API key',
      retryAfter: Math.ceil((req as any).rateLimit.resetTime - Date.now()) / 1000 
    });
  },
  skip: (req: Request) => {
    // Skip for admin dashboard or internal
    return req.path.startsWith('/api/keys') || (req as any).apiKey?.scopes.includes('admin:*');
  },
});

// Daily global fallback (optional)
export const dailyLimiter = rateLimit({
  windowMs: 24 * 60 * 60 * 1000,
  max: (req: Request) => (req as any).apiKey?.limits?.daily || 5000,
  keyGenerator: (req: Request) => (req as any).apiKey?.id || req.ip,
});
```
