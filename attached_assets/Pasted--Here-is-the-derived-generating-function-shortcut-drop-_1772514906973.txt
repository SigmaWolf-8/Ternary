### Drop-in code (TypeScript — libternary/src/ternary-ring.ts)

```ts
// ====================== COMPLEX HELPERS ======================
type Complex = [number, number]; // [re, im]

const C_ZERO: Complex = [0, 0];
const C_ONE: Complex = [1, 0];

function cAdd(a: Complex, b: Complex): Complex {
  return [a[0] + b[0], a[1] + b[1]];
}

function cSub(a: Complex, b: Complex): Complex {
  return [a[0] - b[0], a[1] - b[1]];
}

function cMul(a: Complex, b: Complex): Complex {
  return [a[0] * b[0] - a[1] * b[1], a[0] * b[1] + a[1] * b[0]];
}

function cScale(c: Complex, s: number): Complex {
  return [c[0] * s, c[1] * s];
}

// ====================== POWER-OF-2 FFT (iterative) ======================
// Size must be power of 2. Works for 512 (2n).
function fft(a: Complex[], n: number, inverse: boolean = false): Complex[] {
  const N = a.length;
  if (N !== n) throw new Error("FFT size mismatch");

  // Bit-reverse copy
  const out = new Array<Complex>(N);
  for (let i = 0; i < N; i++) {
    const rev = bitReverse(i, N);
    out[rev] = a[i];
  }

  // Butterfly
  const sign = inverse ? 1 : -1;
  for (let len = 2; len <= N; len <<= 1) {
    const angle = (2 * Math.PI / len) * sign;
    const wlen: Complex = [Math.cos(angle), Math.sin(angle)];
    for (let i = 0; i < N; i += len) {
      let w: Complex = C_ONE;
      for (let j = 0; j < len / 2; j++) {
        const u = out[i + j];
        const v = cMul(out[i + j + len / 2], w);
        out[i + j] = cAdd(u, v);
        out[i + j + len / 2] = cSub(u, v);
        w = cMul(w, wlen);
      }
    }
  }

  if (inverse) {
    const scale = 1 / N;
    for (let i = 0; i < N; i++) out[i] = cScale(out[i], scale);
  }
  return out;
}

function bitReverse(k: number, N: number): number {
  let rev = 0;
  for (let i = 0; i < Math.log2(N); i++) {
    rev = (rev << 1) | (k & 1);
    k >>= 1;
  }
  return rev;
}

// ====================== NEGACYCLIC MUL VIA GENERATING-FUNCTION DFT ======================
export function polyMulGeometric(a: Int8Array, b: Int8Array, n: number = 256): Int8Array {
  if (a.length !== n || b.length !== n) throw new Error("Poly size mismatch");

  const N = 2 * n;                    // 512
  const psiAngle = Math.PI / n;       // torsion twist angle

  // 1. Twist generating functions (geometric phase from torsion)
  const aTwist = new Array<Complex>(n);
  const bTwist = new Array<Complex>(n);
  for (let k = 0; k < n; k++) {
    const phase: Complex = [Math.cos(k * psiAngle), Math.sin(k * psiAngle)];
    aTwist[k] = cMul([a[k], 0], phase);   // a_k * ψ^k
    bTwist[k] = cMul([b[k], 0], phase);
  }

  // 2. Pad to length 2n
  const fa = new Array<Complex>(N).fill(C_ZERO);
  const fb = new Array<Complex>(N).fill(C_ZERO);
  for (let k = 0; k < n; k++) {
    fa[k] = aTwist[k];
    fb[k] = bTwist[k];
  }

  // 3. DFT (generating-function → frequency domain)
  const Fa = fft(fa, N);
  const Fb = fft(fb, N);

  // 4. Pointwise product (diagonalised multiplication)
  const Fc = new Array<Complex>(N);
  for (let i = 0; i < N; i++) {
    Fc[i] = cMul(Fa[i], Fb[i]);
  }

  // 5. Inverse DFT
  const fc = fft(Fc, N, true);

  // 6. Untwist + extract first n coeffs + round + mod 3 → balanced ternary
  const result = new Int8Array(n);
  for (let k = 0; k < n; k++) {
    const phaseInv: Complex = [Math.cos(-k * psiAngle), Math.sin(-k * psiAngle)]; // ψ^(-k)
    const val = cMul(fc[k], phaseInv)[0];   // take real part (imag should be ~0)
    let coeff = Math.round(val);            // exact integer because of small coeffs
    coeff = ((coeff % 3) + 3) % 3;          // 0,1,2
    result[k] = coeff === 2 ? -1 : coeff;   // balanced {-1,0,1}
  }
  return result;
}
```
