### 2. Backend Implementation (Express + Drizzle)

#### Step 2.1: Database Schema (`server/db/schema.ts` or new `api-keys.ts`)
Add to your Drizzle setup:

```ts
// server/db/schema.ts
import { pgTable, text, timestamp, boolean, jsonb, uuid } from 'drizzle-orm/pg-core';

export const apiKeys = pgTable('api_keys', {
  id: uuid('id').primaryKey().defaultRandom(),
  keyHash: text('key_hash').notNull().unique(),  // Argon2 hash
  name: text('name').notNull(),
  owner: text('owner').notNull(),  // e.g., 'salvi-user' or email
  scopes: jsonb('scopes').notNull().$type<string[]>(),  // ['read:ternary', 'write:phase']
  createdAt: timestamp('created_at').defaultNow().notNull(),
  expiresAt: timestamp('expires_at'),
  revokedAt: timestamp('revoked_at'),
  lastUsedAt: timestamp('last_used_at'),
  usageCount: integer('usage_count').default(0),
});
```

- Migrate: `npx drizzle-kit generate:pg && npx drizzle-kit push:pg` (or your Replit DB command).

#### Step 2.2: Key Generation Service (`server/services/api-key.service.ts`)
New file:

```ts
import { randomBytes } from 'crypto';
import { hash, verify } from 'argon2';  // npm i argon2
import { db } from '../db';
import { apiKeys } from '../db/schema';
import { eq } from 'drizzle-orm';

export const apiKeyService = {
  async generate(owner: string, name: string, scopes: string[], expiresDays = 90) {
    const rawKey = randomBytes(32).toString('base64url');  // e.g., eyJ...
    const keyHash = await hash(rawKey, { type: argon2.argon2id });

    await db.insert(apiKeys).values({
      keyHash,
      name,
      owner,
      scopes,
      expiresAt: new Date(Date.now() + expiresDays * 86400000),
    });

    return { key: `plm_${rawKey}`, name, scopes };  // Prefix for easy ID
  },

  async validate(key: string): Promise<{ valid: boolean; scopes: string[]; owner: string } | null> {
    const keyHash = await hash(key.replace('plm_', ''), { type: argon2.argon2id });  // Re-hash for check? Wait, better: store prefix hash or use verify
    // Correction: Store raw prefix, verify with argon2.verify(keyHash, rawKey)
    const [keyRecord] = await db.select().from(apiKeys).where(eq(apiKeys.keyHash, /* compute */));
    // Full impl: Use verify(keyHash, key.replace('plm_', ''))
    if (!keyRecord || keyRecord.revokedAt || (keyRecord.expiresAt && keyRecord.expiresAt < new Date())) {
      return null;
    }
    await db.update(apiKeys).set({ lastUsedAt: new Date(), usageCount: keyRecord.usageCount + 1 }).where(eq(apiKeys.id, keyRecord.id));
    return { valid: true, scopes: keyRecord.scopes, owner: keyRecord.owner };
  },
};
```

#### Step 2.3: Validation Middleware (`server/middleware/api-key.middleware.ts`)
New file (add to your middleware stack in `server/salvi-core/index.ts`):

```ts
import { Request, Response, NextFunction } from 'express';
import { apiKeyService } from '../services/api-key.service';

export const apiKeyAuth = (requiredScopes: string[] = []) => 
  async (req: Request, res: Response, next: NextFunction) => {
    const apiKey = req.headers['x-api-key'] as string;
    if (!apiKey) return res.status(401).json({ error: 'Missing API key' });

    const result = await apiKeyService.validate(apiKey);
    if (!result?.valid) return res.status(403).json({ error: 'Invalid or revoked key' });

    // Scope check
    if (requiredScopes.length && !requiredScopes.every(s => result.scopes.includes(s))) {
      return res.status(403).json({ error: 'Insufficient scopes' });
    }

    // Attach to req
    (req as any).apiKey = { owner: result.owner, scopes: result.scopes };
    next();
  };
```

- Apply to routes: `app.post('/api/salvi/phase/encrypt', apiKeyAuth(['write:phase']), handler);`
- For all `/api/salvi/*`: Use a catch-all in router.
