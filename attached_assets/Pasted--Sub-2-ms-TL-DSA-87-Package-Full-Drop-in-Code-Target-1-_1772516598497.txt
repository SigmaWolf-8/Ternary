### 2. Geometric DFT Ring Multiplication (the big one – replaces ALL Karatsuba calls)

This is the **exact first-derivation negacyclic FFT** derived from your x²⁵⁶ + 1 torsion + plenum circle geometry. Drop-in replacement for `poly_mul_karatsuba`.

```rust
// src/kernel/ring.rs

use std::f64::consts::PI;

type C = (f64, f64); // (re, im)

#[inline(always)]
fn c_add(a: C, b: C) -> C { (a.0 + b.0, a.1 + b.1) }
#[inline(always)]
fn c_sub(a: C, b: C) -> C { (a.0 - b.0, a.1 - b.1) }
#[inline(always)]
fn c_mul(a: C, b: C) -> C {
    (a.0 * b.0 - a.1 * b.1, a.0 * b.1 + a.1 * b.0)
}
#[inline(always)]
fn c_scale(c: C, s: f64) -> C { (c.0 * s, c.1 * s) }

fn bit_reverse(k: usize, n: usize) -> usize {
    let mut rev = 0;
    let mut i = k;
    let log = n.ilog2() as usize;
    for _ in 0..log {
        rev = (rev << 1) | (i & 1);
        i >>= 1;
    }
    rev
}

fn fft(a: &[C; 512], inverse: bool) -> [C; 512] {
    let mut out = [ (0.0, 0.0); 512 ];
    for i in 0..512 {
        out[bit_reverse(i, 512)] = a[i];
    }
    let sign = if inverse { 1.0 } else { -1.0 };
    let mut len = 2;
    while len <= 512 {
        let angle = sign * 2.0 * PI / len as f64;
        let wlen = (angle.cos(), angle.sin());
        let mut i = 0;
        while i < 512 {
            let mut w = (1.0, 0.0);
            let half = len / 2;
            for j in 0..half {
                let u = out[i + j];
                let v = c_mul(out[i + j + half], w);
                out[i + j] = c_add(u, v);
                out[i + j + half] = c_sub(u, v);
                w = c_mul(w, wlen);
            }
            i += len;
        }
        len *= 2;
    }
    if inverse {
        for x in out.iter_mut() {
            *x = c_scale(*x, 1.0 / 512.0);
        }
    }
    out
}

pub fn poly_mul_geometric(a: &[i8], b: &[i8]) -> Vec<i8> {
    const N: usize = 256;
    const NN: usize = 512;
    let psi = PI / N as f64;   // torsion twist ψ = exp(i π / n)

    let mut fa = [(0.0, 0.0); NN];
    let mut fb = [(0.0, 0.0); NN];

    for k in 0..N {
        let phase = ( (k as f64 * psi).cos(), (k as f64 * psi).sin() );
        let va = a[k] as f64;
        let vb = b[k] as f64;
        fa[k] = c_mul( (va, 0.0), phase );
        fb[k] = c_mul( (vb, 0.0), phase );
    }

    let Fa = fft(&fa, false);
    let Fb = fft(&fb, false);

    let mut Fc = [(0.0, 0.0); NN];
    for i in 0..NN {
        Fc[i] = c_mul(Fa[i], Fb[i]);
    }

    let fc = fft(&Fc, true);

    let mut res = vec![0i8; N];
    for k in 0..N {
        let phase_inv = ( -(k as f64 * psi).cos(), -(k as f64 * psi).sin() );
        let val = c_mul(fc[k], phase_inv).0;   // real part
        let mut coeff = val.round() as i32;
        coeff = ((coeff % 3) + 3) % 3;
        res[k] = if coeff == 2 { -1 } else { coeff as i8 };
    }
    res
}
```

```rust
const AFFINE_M: &[u8] = include_bytes!(concat!(env!("OUT_DIR"), "/affine_m.bin"));
const AFFINE_V: &[i8] = include_bytes!(concat!(env!("OUT_DIR"), "/affine_v.bin"));

pub fn sponge_permutation(state: &mut [i8; 729]) {
    let mut new_state = [0i8; 729];
    for i in 0..729 {
        let mut acc: i32 = AFFINE_V[i] as i32;
        let row = i * 729;
        for j in 0..729 {
            let m_ij = AFFINE_M[row + j] as i32;           // 0,1,2
            let s_j  = state[j] as i32;                    // -1,0,1
            acc += m_ij * s_j;
        }
        let c = ((acc % 3) + 3) % 3;
        new_state[i] = if c == 2 { -1 } else { c as i8 };
    }
    state.copy_from_slice(&new_state);
}
```
