# ═══════════════════════════════════════════
# GF(27) = GF(3)[t] / (t^3 + 2t + 1)
# ═══════════════════════════════════════════
def gf27_add(a, b):
    return tuple((ai + bi) % 3 for ai, bi in zip(a, b))
def gf27_neg(a):
    return tuple((3 - ai) % 3 for ai in a)
def gf27_sub(a, b):
    return gf27_add(a, gf27_neg(b))
def gf27_mul(a, b):
    a0,a1,a2 = a; b0,b1,b2 = b
    c0=a0*b0; c1=a0*b1+a1*b0; c2=a0*b2+a1*b1+a2*b0; c3=a1*b2+a2*b1; c4=a2*b2
    return ((c0+2*c3)%3, (c1+c3+2*c4)%3, (c2+c4)%3)
def gf27_pow(a, n):
    if n==0: return (1,0,0)
    if a==(0,0,0): return (0,0,0)
    r=(1,0,0); b=a
    while n>0:
        if n%2==1: r=gf27_mul(r,b)
        b=gf27_mul(b,b); n//=2
    return r

def elem_to_int(a): return a[0]+3*a[1]+9*a[2]
def int_to_elem(n): return (n%3,(n//3)%3,(n//9)%3)
ALL = [int_to_elem(i) for i in range(27)]
ZERO=(0,0,0); ONE=(1,0,0)

def compute_ddt(exp):
    chi = {x: gf27_pow(x, exp) for x in ALL}
    is_perm = len(set(chi.values())) == 27
    ddt = {}
    mx = 0; mp = None
    for a in ALL:
        if a == ZERO: continue
        ddt[a] = {b: 0 for b in ALL}
        for x in ALL:
            d = gf27_sub(chi[gf27_add(x,a)], chi[x])
            ddt[a][d] += 1
            if ddt[a][d] > mx:
                mx = ddt[a][d]; mp = (a, d)
    return is_perm, mx, mp, ddt

def print_ddt(exp, ddt, mx):
    print(f"\n{'='*80}")
    print(f"FULL DDT FOR chi(x) = x^{exp} OVER GF(27)   [DP_max = {mx}/27 = {mx/27:.6f}]")

        a = int_to_elem(ai)
        print(f"{ai:>3}", end="")
        for bi in range(27):
            b = int_to_elem(bi)
            v = ddt[a][b]
            vc[v] += 1
            print(f"{v:>3}", end="")
        print()
    print(f"\nValue distribution:")
    for v in sorted(vc): print(f"  {v}: {vc[v]} entries ({vc[v]/(26*27)*100:.1f}%)")

# For GF(27), the Walsh transform uses the absolute trace Tr: GF(27) -> GF(3)
# Tr(x) = x + x^3 + x^9 for GF(3^3)
def trace(x):
    return gf27_add(x, gf27_add(gf27_pow(x,3), gf27_pow(x,9)))

# Walsh coefficient W_f(a,b) = sum_{x in GF(27)} omega^(Tr(b*f(x) - a*x))
# where omega = exp(2pi i/3)
# For linearity analysis, we compute |W_f(a,b)| for all a,b
# The nonlinearity NL = (27 - max|W|) / ... but let's just get max|W|

import cmath
omega = cmath.exp(2j * cmath.pi / 3)

chi17 = {x: gf27_pow(x, 17) for x in ALL}
max_walsh = 0
for a in ALL:
    for b in ALL:
        if a == ZERO and b == ZERO: continue
        W = 0
        for x in ALL:
            bfx = gf27_mul(b, chi17[x])
            ax = gf27_mul(a, x)
            arg = gf27_sub(bfx, ax)
            tr_val = trace(arg)
            # tr_val is in GF(3), so it's (t,0,0) with t in {0,1,2}
            t = tr_val[0]  # since trace maps to GF(3) embedded as (t,0,0)
            W += omega**t
        mag = abs(W)
        if mag > max_walsh:
            max_walsh = mag

print(f"Max |Walsh coefficient|: {max_walsh:.4f}")

print(f"Linearity measure: {max_walsh:.4f} / 27 = {max_walsh/27:.4f}")
