### 1. Generalized Qudit Correction Opcode (supports arbitrary d ≥ 3)

This opcode (let's assign **opcode 166: QUDIT_CORRECT_D**) takes two operands:
- `reg_start`: u8 — first register of the block
- `d`: u8 — dimension (3..=13, clamped)

It treats `d × 3` consecutive registers as one logical qudit encoded in a 3-physical-qudit repetition code (each physical qudit uses d registers: amplitude real + imag for each level, but flattened for simplicity).

```rust
// src/kernel/src/vm/opcodes.rs (add to your opcode handler match)

// Generalized Qudit Correction Opcode 166: QUDIT_CORRECT_D reg_start: u8, d: u8
pub fn op_qudit_correct_d(vm: &mut VMState, reg_start: u8, d_param: u8) -> Result<(), VMError> {
    let d = d_param.clamp(3, 13) as usize; // SUFT radius max 13

    let regs_needed = d * 3; // 3 physical qudits × d amplitudes each
    if (reg_start as usize) + regs_needed * 2 > vm.registers.len() {
        return Err(VMError::RegisterOutOfBounds);
    }

    // ─── Extract amplitudes ──────────────────────────────────────
    // Each physical qudit: d complex numbers (real, imag packed in registers)
    // Total: 3 × d complex → 6d f32 registers used
    let mut state = DVector::zeros(d * 3);
    for i in 0..(d * 3) {
        let reg_idx = (reg_start as usize + i) * 2;
        state[i] = Complex::new(
            vm.registers[reg_idx] as f64,
            vm.registers[reg_idx + 1] as f64,
        );
    }

    // Normalize input state
    let norm = state.norm();
    if norm > 0.0 {
        state /= norm;
    }

    // ─── Apply random single error (for realism; in real use this would be external noise) ───
    let mut rng = rand::thread_rng();
    let target = rng.gen_range(0..3);
    let error_level = rng.gen_range(0..d);
    let shift_amount = rng.gen_range(1..d) as i32;

    // Simplified shift error on target qudit
    let mut erroneous = state.clone();
    for i in 0..d {
        let idx = target * d + i;
        let new_idx = (i as i32 + shift_amount).rem_euclid(d as i32) as usize;
        erroneous[target * d + new_idx] += erroneous[idx] * Complex::new(0.0, 1.0); // phase-like error
    }

    // ─── Syndrome extraction ─────────────────────────────────────
    // For repetition code, compare differences between copies
    let mut syndrome = vec![0.0f64; 2];

    // S1: difference between copy 0 and 1
    for i in 0..d {
        let idx0 = i;
        let idx1 = d + i;
        syndrome[0] += (erroneous[idx0] - erroneous[idx1]).norm_squared();
    }

    // S2: difference between copy 1 and 2
    for i in 0..d {
        let idx1 = d + i;
        let idx2 = 2 * d + i;
        syndrome[1] += (erroneous[idx1] - erroneous[idx2]).norm_squared();
    }

    // ─── Correction ──────────────────────────────────────────────
    let mut corrected = erroneous.clone();
    if syndrome[0] > 0.01 || syndrome[1] > 0.01 {
        let target_corr = if syndrome[0] > syndrome[1] { 0 } else { 1 };
        // Simple correction: average the three copies
        for i in 0..d {
            let avg = (erroneous[i] + erroneous[d + i] + erroneous[2 * d + i]) / 3.0;
            corrected[i] = avg;
            corrected[d + i] = avg;
            corrected[2 * d + i] = avg;
        }
        corrected = normalize(corrected);
    }

    // ─── Write back to registers ─────────────────────────────────
    for i in 0..(d * 3) {
        let reg_idx = (reg_start as usize + i) * 2;
        vm.registers[reg_idx]     = corrected[i].re as f32;
        vm.registers[reg_idx + 1] = corrected[i].im as f32;
    }

    Ok(())
}
```

### 2. Noether-Invariant Check After Correction

This is a post-correction hook that verifies the three main SUFT Noether invariants on the affected registers.

```rust
// Check all three Noether invariants after correction
pub fn check_noether_invariants_after_correction(
    vm: &VMState,
    reg_start: usize,
    d: usize,
    tolerance: f64,
) -> Result<(), VMError> {
    // 1. Ternary gauge symmetry invariant: sum of branch shifts ≈ 0
    // We assume the three logical branches are mapped to the first three complex values after correction
    let branch_sum = 
        vm.registers[reg_start * 2] as f64 + 
        vm.registers[(reg_start + d) * 2] as f64 + 
        vm.registers[(reg_start + 2 * d) * 2] as f64;

    if branch_sum.abs() > tolerance {
        return Err(VMError::InvariantViolation("Ternary gauge sum ≠ 0"));
    }

    // 2. Reparametrization / energy-like invariant (simplified: total "mass-shell" ≈ 0)
    let mut sum_e2 = 0.0f64;
    for i in 0..(d * 3) {
        let re = vm.registers[(reg_start + i) * 2] as f64;
        let im = vm.registers[(reg_start + i) * 2 + 1] as f64;
        sum_e2 += re * re + im * im;
    }
    let energy_invariant = (13.0 / 28.0) * sum_e2; // from SUFT Φ term
    if energy_invariant.abs() > tolerance {
        return Err(VMError::InvariantViolation("Reparametrization energy invariant broken"));
    }

    // 3. Periodicity invariant: each branch value mod 364 should be consistent
    // We use the real part as "time coordinate" (scaled)
    for i in 0..3 {
        let t_val = vm.registers[(reg_start + i * d) * 2] as f64;
        let mod_val = t_val.rem_euclid(364.0);
        if (t_val - mod_val).abs() > tolerance * 364.0 {
            return Err(VMError::InvariantViolation("Periodicity violation"));
        }
    }

    Ok(())
}
```
