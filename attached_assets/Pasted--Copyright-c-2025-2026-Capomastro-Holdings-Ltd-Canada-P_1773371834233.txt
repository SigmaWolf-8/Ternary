//! # Sponge Block Shuffles (T-18, SPEC-2026-NEXT)
//!
//! Applies the four σ permutations from the Plenum Square (T-04) to the
//! sponge state as a coarse-grain diffusion step. Each round uses a
//! different permutation selected by `SIGMAS[round % 4]`.
//!
//! ## Block Decomposition
//!
//! The 729-trit sponge state is decomposed into 9 blocks of 81 trits:
//!
//! ```text
//! State[729] = Block_0[81] ‖ Block_1[81] ‖ ... ‖ Block_8[81]
//! ```
//!
//! 81 = 3⁴ trits per block. 9 × 81 = 729 = 3⁶ total.
//! The block boundaries are natural in base-3: each block is a
//! complete 4th-order ternary subspace.
//!
//! ## Shuffle Operation
//!
//! Given σ = [2, 6, 7, 8, 4, 0, 1, 5, 3] (example: σ_C from T-04):
//!
//! ```text
//! new_block[0] = old_block[σ[0]] = old_block[2]
//! new_block[1] = old_block[σ[1]] = old_block[6]
//! ...
//! new_block[8] = old_block[σ[8]] = old_block[3]
//! ```
//!
//! This moves entire 81-trit blocks without touching individual trits.
//! The per-trit mixing is handled by the stride-13 permutation π and
//! the χ S-box in subsequent steps.
//!
//! ## Round Schedule
//!
//! - **TIS-27 (4 rounds)**: σ_A, σ_B, σ_C, σ_D — one per round
//! - **TLSponge-385 (9 rounds)**: cycling `SIGMAS[round % 4]`
//!   Round 0: σ_A, Round 1: σ_B, ..., Round 4: σ_A again, etc.
//!
//! σ_A (the full derangement) runs on round 0 for maximum first-round
//! disruption — every block moves, including the center.
//!
//! ## Performance
//!
//! Block shuffle = 9 memcpy of 81 bytes = 729 bytes total.
//! At ~1 byte/ns on modern hardware: ~1µs per shuffle.
//! The overhead target is <10% of total permutation time (T-26 benchmark).
//!
//! ## Feature Gate
//!
//! Controlled by `PlenumConfig.enable_sponge_shuffles` (T-05).
//! When disabled, the shuffle is a no-op (identity permutation).
//!
//! ## Integration
//!
//! The sponge permutation function calls `apply_block_shuffle()` BEFORE
//! the stride-13 trit permutation π in each round:
//!
//! ```text
//! for round in 0..num_rounds:
//!     σ_shuffle(state, round)   ← T-18 (coarse-grain block move)
//!     θ_diffusion(state)        ← existing (7-neighbor XOR)
//!     χ_sbox(state)             ← existing (x¹⁷ over GF(27))
//!     π_permutation(state)      ← existing (stride-13 trit move)
//! ```

use crate::plenum_square::{SIGMAS, SIGMA_A, SIGMA_B, SIGMA_C, SIGMA_D};

// ═══════════════════════════════════════════════════════════════════════
// CONSTANTS
// ═══════════════════════════════════════════════════════════════════════

/// Sponge state size in trits (3⁶ = 729).
pub const STATE_TRITS: usize = 729;

/// Number of blocks in the decomposition.
pub const NUM_BLOCKS: usize = 9;

/// Trits per block (3⁴ = 81).
pub const BLOCK_SIZE: usize = 81;

/// Number of σ permutations in the cycle.
pub const NUM_SIGMAS: usize = 4;

/// TIS-27 round count.
pub const TIS27_ROUNDS: usize = 4;

/// TLSponge-385 round count.
pub const TLSPONGE_ROUNDS: usize = 9;

// Compile-time verification
const _: () = {
    assert!(NUM_BLOCKS * BLOCK_SIZE == STATE_TRITS);
    assert!(SIGMAS.len() == NUM_SIGMAS);
    assert!(TIS27_ROUNDS == NUM_SIGMAS); // One σ per TIS-27 round
};

// ═══════════════════════════════════════════════════════════════════════
// BLOCK SHUFFLE — Core operation
// ═══════════════════════════════════════════════════════════════════════

/// Apply a σ block permutation to a 729-trit sponge state.
///
/// Decomposes the state into 9 blocks of 81 trits, permutes the blocks
/// according to σ, and writes the result back.
///
/// This operates on `i8` (balanced ternary {-1, 0, 1}) which is the
/// internal representation of the Rust sponge.
pub fn apply_block_shuffle_i8(state: &mut [i8; STATE_TRITS], sigma: &[usize; NUM_BLOCKS]) {
    // Read blocks into temp buffer
    let mut temp = [0i8; STATE_TRITS];

    for new_pos in 0..NUM_BLOCKS {
        let old_pos = sigma[new_pos];
        let src_start = old_pos * BLOCK_SIZE;
        let dst_start = new_pos * BLOCK_SIZE;
        temp[dst_start..dst_start + BLOCK_SIZE]
            .copy_from_slice(&state[src_start..src_start + BLOCK_SIZE]);
    }

    state.copy_from_slice(&temp);
}

/// Apply a σ block permutation to a 729-element u8 state.
///
/// Same operation for the N-API layer which uses `u8` representation.
pub fn apply_block_shuffle_u8(state: &mut [u8; STATE_TRITS], sigma: &[usize; NUM_BLOCKS]) {
    let mut temp = [0u8; STATE_TRITS];

    for new_pos in 0..NUM_BLOCKS {
        let old_pos = sigma[new_pos];
        let src_start = old_pos * BLOCK_SIZE;
        let dst_start = new_pos * BLOCK_SIZE;
        temp[dst_start..dst_start + BLOCK_SIZE]
            .copy_from_slice(&state[src_start..src_start + BLOCK_SIZE]);
    }

    state.copy_from_slice(&temp);
}

/// Apply the inverse σ block permutation.
///
/// If `apply_block_shuffle(state, σ)` moves block i to position σ[i],
/// then `apply_inverse_shuffle(state, σ)` restores the original order.
///
/// Useful for verification and testing.
pub fn apply_inverse_shuffle_i8(state: &mut [i8; STATE_TRITS], sigma: &[usize; NUM_BLOCKS]) {
    let inverse = invert_permutation(sigma);
    apply_block_shuffle_i8(state, &inverse);
}

/// Compute the inverse of a permutation.
///
/// If σ[i] = j, then σ⁻¹[j] = i.
pub fn invert_permutation(sigma: &[usize; NUM_BLOCKS]) -> [usize; NUM_BLOCKS] {
    let mut inv = [0usize; NUM_BLOCKS];
    for i in 0..NUM_BLOCKS {
        inv[sigma[i]] = i;
    }
    inv
}

// ═══════════════════════════════════════════════════════════════════════
// ROUND-DEPENDENT SHUFFLE SELECTION
// ═══════════════════════════════════════════════════════════════════════

/// Get the σ permutation for a given round number.
///
/// Uses `SIGMAS[round % 4]` for cycling through the four permutations.
/// σ_A (full derangement) is always round 0.
#[inline]
pub fn sigma_for_round(round: usize) -> &'static [usize; NUM_BLOCKS] {
    &SIGMAS[round % NUM_SIGMAS]
}

/// Apply the round-dependent σ shuffle to a sponge state.
///
/// This is the function called by the sponge permutation loop:
///
/// ```rust,ignore
/// for round in 0..num_rounds {
///     sponge_shuffle::shuffle_round_i8(&mut state, round);
///     // ... θ, χ, π steps ...
/// }
/// ```
#[inline]
pub fn shuffle_round_i8(state: &mut [i8; STATE_TRITS], round: usize) {
    let sigma = sigma_for_round(round);
    apply_block_shuffle_i8(state, sigma);
}

/// Apply the round-dependent σ shuffle (u8 variant for N-API).
#[inline]
pub fn shuffle_round_u8(state: &mut [u8; STATE_TRITS], round: usize) {
    let sigma = sigma_for_round(round);
    apply_block_shuffle_u8(state, sigma);
}

// ═══════════════════════════════════════════════════════════════════════
// LOOKUP TABLES — Compile-time selectable block mappings
// ═══════════════════════════════════════════════════════════════════════

/// Precomputed block offset mappings for each σ permutation.
///
/// `BLOCK_OFFSETS[sigma_idx][new_block] = (src_byte_offset, dst_byte_offset)`
///
/// These allow the shuffle to be expressed as a fixed sequence of
/// memcpy operations without runtime index computation.
pub const BLOCK_OFFSETS: [[(usize, usize); NUM_BLOCKS]; NUM_SIGMAS] = {
    let mut table = [[(0usize, 0usize); NUM_BLOCKS]; NUM_SIGMAS];
    let mut s = 0;
    while s < NUM_SIGMAS {
        let mut b = 0;
        while b < NUM_BLOCKS {
            let old_pos = SIGMAS[s][b];
            table[s][b] = (old_pos * BLOCK_SIZE, b * BLOCK_SIZE);
            b += 1;
        }
        s += 1;
    }
    table
};

/// Apply a σ shuffle using precomputed offset table (i8 variant).
///
/// Functionally identical to `apply_block_shuffle_i8` but uses the
/// compile-time offset table. May be slightly faster due to branch
/// elimination in tight loops.
pub fn apply_shuffle_from_table_i8(state: &mut [i8; STATE_TRITS], sigma_idx: usize) {
    let offsets = &BLOCK_OFFSETS[sigma_idx % NUM_SIGMAS];
    let mut temp = [0i8; STATE_TRITS];

    for &(src, dst) in offsets {
        temp[dst..dst + BLOCK_SIZE].copy_from_slice(&state[src..src + BLOCK_SIZE]);
    }

    state.copy_from_slice(&temp);
}

// ═══════════════════════════════════════════════════════════════════════
// FULL-ROUND SHUFFLE SEQUENCES
// ═══════════════════════════════════════════════════════════════════════

/// Apply the complete σ shuffle sequence for TIS-27 (4 rounds).
///
/// Applies σ_A, σ_B, σ_C, σ_D in order. Used for testing the
/// cumulative effect of all four shuffles.
pub fn apply_tis27_sequence_i8(state: &mut [i8; STATE_TRITS]) {
    for round in 0..TIS27_ROUNDS {
        shuffle_round_i8(state, round);
    }
}

/// Apply the complete σ shuffle sequence for TLSponge-385 (9 rounds).
pub fn apply_tlsponge_sequence_i8(state: &mut [i8; STATE_TRITS]) {
    for round in 0..TLSPONGE_ROUNDS {
        shuffle_round_i8(state, round);
    }
}

/// Compute the composed permutation after N rounds of cycling shuffles.
///
/// Returns the single σ that is equivalent to applying σ_A∘σ_B∘...∘σ_N.
/// Useful for analysis and verification.
pub fn compose_rounds(num_rounds: usize) -> [usize; NUM_BLOCKS] {
    // Start with identity
    let mut composed = [0, 1, 2, 3, 4, 5, 6, 7, 8];

    for round in 0..num_rounds {
        let sigma = sigma_for_round(round);
        let mut next = [0usize; NUM_BLOCKS];
        for i in 0..NUM_BLOCKS {
            next[i] = composed[sigma[i]];
        }
        composed = next;
    }
    composed
}

// ═══════════════════════════════════════════════════════════════════════
// ANALYSIS UTILITIES
// ═══════════════════════════════════════════════════════════════════════

/// Count the number of fixed points in a permutation.
///
/// A fixed point is a position where σ[i] = i (block doesn't move).
pub fn fixed_point_count(sigma: &[usize; NUM_BLOCKS]) -> usize {
    sigma.iter().enumerate().filter(|&(i, &s)| s == i).count()
}

/// Compute the order (period) of a permutation.
///
/// The smallest k such that σ^k = identity. Tells you how many
/// applications of σ before the state returns to its original order.
pub fn permutation_order(sigma: &[usize; NUM_BLOCKS]) -> usize {
    let identity = [0, 1, 2, 3, 4, 5, 6, 7, 8];
    let mut current = *sigma;
    for k in 1..=362880 { // 9! upper bound
        if current == identity {
            return k;
        }
        let mut next = [0usize; NUM_BLOCKS];
        for i in 0..NUM_BLOCKS {
            next[i] = sigma[current[i]];
        }
        current = next;
    }
    0 // Should never happen for a valid permutation
}

/// Compute the cycle structure of a permutation.
///
/// Returns a sorted vector of cycle lengths. For example,
/// a permutation with cycles (0 2 7)(1 6)(3 8 5)(4) has
/// cycle structure [1, 2, 3, 3].
pub fn cycle_structure(sigma: &[usize; NUM_BLOCKS]) -> Vec<usize> {
    let mut visited = [false; NUM_BLOCKS];
    let mut cycles = Vec::new();

    for start in 0..NUM_BLOCKS {
        if visited[start] {
            continue;
        }
        let mut len = 0;
        let mut pos = start;
        loop {
            visited[pos] = true;
            len += 1;
            pos = sigma[pos];
            if pos == start {
                break;
            }
        }
        cycles.push(len);
    }

    cycles.sort();
    cycles
}

#[cfg(test)]
mod tests {
    use super::*;

    /// Create a test state where block i has all trits set to i as i8.
    fn labeled_state() -> [i8; STATE_TRITS] {
        let mut state = [0i8; STATE_TRITS];
        for block in 0..NUM_BLOCKS {
            for trit in 0..BLOCK_SIZE {
                state[block * BLOCK_SIZE + trit] = block as i8;
            }
        }
        state
    }

    /// Extract the label (first trit) of each block.
    fn block_labels(state: &[i8; STATE_TRITS]) -> [i8; NUM_BLOCKS] {
        let mut labels = [0i8; NUM_BLOCKS];
        for b in 0..NUM_BLOCKS {
            labels[b] = state[b * BLOCK_SIZE];
        }
        labels
    }

    // ── Basic shuffle ───────────────────────────────────────────

    #[test]
    fn test_identity_shuffle() {
        let mut state = labeled_state();
        let identity = [0, 1, 2, 3, 4, 5, 6, 7, 8];
        apply_block_shuffle_i8(&mut state, &identity);
        assert_eq!(block_labels(&state), [0, 1, 2, 3, 4, 5, 6, 7, 8]);
    }

    #[test]
    fn test_sigma_a_shuffle() {
        let mut state = labeled_state();
        apply_block_shuffle_i8(&mut state, &SIGMA_A);
        // σ_A = [4, 8, 3, 2, 0, 7, 5, 6, 1]
        // new[0] = old[4], new[1] = old[8], new[2] = old[3], ...
        let expected: [i8; 9] = [4, 8, 3, 2, 0, 7, 5, 6, 1];
        assert_eq!(block_labels(&state), expected);
    }

    #[test]
    fn test_shuffle_preserves_block_contents() {
        let mut state = labeled_state();
        apply_block_shuffle_i8(&mut state, &SIGMA_A);

        // Block 0 now contains what was block σ_A[0]=4
        // Every trit in block 0 should be 4
        for trit in 0..BLOCK_SIZE {
            assert_eq!(state[trit], 4, "Block 0 should contain old block 4");
        }
    }

    #[test]
    fn test_all_four_sigmas_differ() {
        let original = labeled_state();
        let mut results = Vec::new();

        for sigma in &SIGMAS {
            let mut state = original;
            apply_block_shuffle_i8(&mut state, sigma);
            results.push(block_labels(&state));
        }

        // All four results should be different
        for i in 0..4 {
            for j in (i + 1)..4 {
                assert_ne!(results[i], results[j],
                    "σ_{} and σ_{} must produce different shuffles", i, j);
            }
        }
    }

    // ── Inverse ─────────────────────────────────────────────────

    #[test]
    fn test_inverse_roundtrip() {
        let mut state = labeled_state();
        let original = state;

        apply_block_shuffle_i8(&mut state, &SIGMA_A);
        assert_ne!(block_labels(&state), block_labels(&original));

        apply_inverse_shuffle_i8(&mut state, &SIGMA_A);
        assert_eq!(state, original, "Inverse must restore original state");
    }

    #[test]
    fn test_invert_permutation() {
        let inv = invert_permutation(&SIGMA_A);
        // σ_A[inv[i]] = i for all i
        for i in 0..NUM_BLOCKS {
            assert_eq!(SIGMA_A[inv[i]], i);
        }
    }

    // ── Round-dependent selection ───────────────────────────────

    #[test]
    fn test_sigma_for_round_cycling() {
        assert_eq!(sigma_for_round(0), &SIGMA_A);
        assert_eq!(sigma_for_round(1), &SIGMA_B);
        assert_eq!(sigma_for_round(2), &SIGMA_C);
        assert_eq!(sigma_for_round(3), &SIGMA_D);
        assert_eq!(sigma_for_round(4), &SIGMA_A); // Cycles back
        assert_eq!(sigma_for_round(8), &SIGMA_A); // Round 8 = 8 % 4 = 0
    }

    #[test]
    fn test_tis27_uses_all_four() {
        let mut used = std::collections::HashSet::new();
        for round in 0..TIS27_ROUNDS {
            used.insert(sigma_for_round(round) as *const _);
        }
        assert_eq!(used.len(), 4, "TIS-27 must use all four σ permutations");
    }

    // ── Lookup table ────────────────────────────────────────────

    #[test]
    fn test_table_matches_direct() {
        let mut state_direct = labeled_state();
        let mut state_table = labeled_state();

        apply_block_shuffle_i8(&mut state_direct, &SIGMA_A);
        apply_shuffle_from_table_i8(&mut state_table, 0); // σ_A is index 0

        assert_eq!(state_direct, state_table,
            "Table-based and direct shuffle must produce identical results");
    }

    #[test]
    fn test_table_all_sigmas() {
        for sigma_idx in 0..NUM_SIGMAS {
            let mut state_direct = labeled_state();
            let mut state_table = labeled_state();

            apply_block_shuffle_i8(&mut state_direct, &SIGMAS[sigma_idx]);
            apply_shuffle_from_table_i8(&mut state_table, sigma_idx);

            assert_eq!(state_direct, state_table,
                "Table mismatch for σ index {}", sigma_idx);
        }
    }

    // ── Composition ─────────────────────────────────────────────

    #[test]
    fn test_compose_zero_rounds_is_identity() {
        let composed = compose_rounds(0);
        assert_eq!(composed, [0, 1, 2, 3, 4, 5, 6, 7, 8]);
    }

    #[test]
    fn test_compose_one_round_is_sigma_a() {
        let composed = compose_rounds(1);
        assert_eq!(composed, SIGMA_A);
    }

    #[test]
    fn test_compose_matches_sequential() {
        // Composing 4 rounds should match applying 4 shuffles sequentially
        let mut state = labeled_state();
        apply_tis27_sequence_i8(&mut state);
        let sequential_labels = block_labels(&state);

        let composed = compose_rounds(TIS27_ROUNDS);
        let mut state2 = labeled_state();
        apply_block_shuffle_i8(&mut state2, &composed);
        let composed_labels = block_labels(&state2);

        assert_eq!(sequential_labels, composed_labels,
            "4-round composition must match sequential application");
    }

    // ── Analysis ────────────────────────────────────────────────

    #[test]
    fn test_sigma_a_is_derangement() {
        assert_eq!(fixed_point_count(&SIGMA_A), 0,
            "σ_A must be a derangement (no fixed points)");
    }

    #[test]
    fn test_sigma_bcd_fix_center() {
        assert_eq!(fixed_point_count(&SIGMA_B), 1);
        assert_eq!(fixed_point_count(&SIGMA_C), 1);
        assert_eq!(fixed_point_count(&SIGMA_D), 1);
        // The fixed point should be index 4 (center)
        assert_eq!(SIGMA_B[4], 4);
        assert_eq!(SIGMA_C[4], 4);
        assert_eq!(SIGMA_D[4], 4);
    }

    #[test]
    fn test_permutation_orders() {
        let order_a = permutation_order(&SIGMA_A);
        let order_b = permutation_order(&SIGMA_B);
        let order_c = permutation_order(&SIGMA_C);
        let order_d = permutation_order(&SIGMA_D);

        // All orders must be > 1 (not identity) and divide 9!
        assert!(order_a > 1, "σ_A order must be > 1");
        assert!(order_b > 1, "σ_B order must be > 1");
        assert!(order_c > 1, "σ_C order must be > 1");
        assert!(order_d > 1, "σ_D order must be > 1");
    }

    #[test]
    fn test_cycle_structure() {
        let cycles_a = cycle_structure(&SIGMA_A);
        // σ_A is a derangement → no cycle of length 1
        assert!(!cycles_a.contains(&1), "σ_A derangement has no fixed-point cycles");

        let cycles_b = cycle_structure(&SIGMA_B);
        // σ_B fixes center → has a cycle of length 1
        assert!(cycles_b.contains(&1), "σ_B has center fixed point");
    }

    // ── u8 variant ──────────────────────────────────────────────

    #[test]
    fn test_u8_matches_i8() {
        let mut state_i8 = [0i8; STATE_TRITS];
        let mut state_u8 = [0u8; STATE_TRITS];
        for i in 0..STATE_TRITS {
            state_i8[i] = (i % 3) as i8 - 1; // {-1, 0, 1}
            state_u8[i] = (i % 3) as u8;      // {0, 1, 2}
        }

        apply_block_shuffle_i8(&mut state_i8, &SIGMA_A);
        apply_block_shuffle_u8(&mut state_u8, &SIGMA_A);

        // Verify same block ordering (modulo representation)
        for b in 0..NUM_BLOCKS {
            let i8_first = state_i8[b * BLOCK_SIZE];
            let u8_first = state_u8[b * BLOCK_SIZE];
            // They should have the same relative block ordering
            assert_eq!(
                (i8_first + 1) as u8, u8_first,
                "Block {} ordering mismatch between i8 and u8 variants", b
            );
        }
    }

    // ── Constants ───────────────────────────────────────────────

    #[test]
    fn test_constants() {
        assert_eq!(STATE_TRITS, 729);
        assert_eq!(NUM_BLOCKS, 9);
        assert_eq!(BLOCK_SIZE, 81);
        assert_eq!(NUM_BLOCKS * BLOCK_SIZE, STATE_TRITS);
        assert_eq!(NUM_SIGMAS, 4);
        assert_eq!(TIS27_ROUNDS, 4);
        assert_eq!(TLSPONGE_ROUNDS, 9);
    }

    // ── Precomputed offsets ─────────────────────────────────────

    #[test]
    fn test_block_offsets_valid() {
        for s in 0..NUM_SIGMAS {
            for b in 0..NUM_BLOCKS {
                let (src, dst) = BLOCK_OFFSETS[s][b];
                assert!(src + BLOCK_SIZE <= STATE_TRITS,
                    "σ_{} block {} src out of bounds", s, b);
                assert!(dst + BLOCK_SIZE <= STATE_TRITS,
                    "σ_{} block {} dst out of bounds", s, b);
            }
        }
    }
}
