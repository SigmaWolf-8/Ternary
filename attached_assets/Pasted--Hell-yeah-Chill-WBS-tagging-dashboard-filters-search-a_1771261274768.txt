### 2. WBS Selector Component (salvi-sign/client/src/components/WBSSelector.tsx)

Searchable, multi-level dropdown (13 levels) with tags.

```tsx
'use client';

import { useState } from 'react';
import { Command, CommandEmpty, CommandGroup, CommandInput, CommandItem, CommandList } from '@/components/ui/command';
import { Popover, PopoverContent, PopoverTrigger } from '@/components/ui/popover';
import { Button } from '@/components/ui/button';
import { Badge } from '@/components/ui/badge';
import { Check, ChevronsUpDown, X } from 'lucide-react';
import { useEditorStore } from '@/store/editorStore';

const LEVELS = Array.from({ length: 13 }, (_, i) => `L${i + 1}`);

interface WBSSelectorProps {
  onChange: (tags: string[]) => void;
  value: string[];
}

export default function WBSSelector({ onChange, value }: WBSSelectorProps) {
  const [open, setOpen] = useState(false);
  const [search, setSearch] = useState('');

  const handleSelect = (level: string) => {
    const newTag = `${level}:${search || 'Untitled'}`;
    if (!value.includes(newTag)) {
      const updated = [...value, newTag];
      onChange(updated);
    }
    setSearch('');
  };

  const removeTag = (tag: string) => {
    onChange(value.filter(t => t !== tag));
  };

  return (
    <div className="space-y-2">
      <Popover open={open} onOpenChange={setOpen}>
        <PopoverTrigger asChild>
          <Button variant="outline" className="w-full justify-between border-gold text-gold hover:bg-zinc-900">
            Add WBS Tag (13 Levels)
            <ChevronsUpDown className="ml-2 h-4 w-4" />
          </Button>
        </PopoverTrigger>
        <PopoverContent className="w-80 p-0 bg-zinc-950 border-gold">
          <Command>
            <CommandInput
              placeholder="Search or type level..."
              value={search}
              onValueChange={setSearch}
            />
            <CommandList>
              <CommandEmpty>No results.</CommandEmpty>
              <CommandGroup heading="Levels">
                {LEVELS.map(level => (
                  <CommandItem key={level} onSelect={() => handleSelect(level)}>
                    {level}
                  </CommandItem>
                ))}
              </CommandGroup>
            </CommandList>
          </Command>
        </PopoverContent>
      </Popover>

      {/* Selected Tags */}
      <div className="flex flex-wrap gap-2">
        {value.map(tag => (
          <Badge key={tag} variant="secondary" className="bg-gold/10 text-gold border border-gold/30 px-3 py-1">
            {tag}
            <X className="ml-2 h-3 w-3 cursor-pointer" onClick={() => removeTag(tag)} />
          </Badge>
        ))}
      </div>
    </div>
  );
}
```

### 3. Dashboard Updates (salvi-sign/client/src/app/dashboard/page.tsx)

Assume you have a basic dashboard — here's the enhanced version with **WBS toggles + search**.

```tsx
'use client';

import { useState, useMemo } from 'react';
import { Input } from '@/components/ui/input';
import { Button } from '@/components/ui/button';
import { Badge } from '@/components/ui/badge';
import { useEditorStore } from '@/store/editorStore'; // or fetch from API

export default function Dashboard() {
  const [search, setSearch] = useState('');
  const [activeWBSLevels, setActiveWBSLevels] = useState<string[]>([]); // e.g. ['L1', 'L3']
  const [envelopes, setEnvelopes] = useState([]); // fetch from /api/envelopes

  // Toggle WBS filter
  const toggleWBS = (level: string) => {
    setActiveWBSLevels(prev =>
      prev.includes(level) ? prev.filter(l => l !== level) : [...prev, level]
    );
  };

  // Filtered envelopes
  const filteredEnvelopes = useMemo(() => {
    return envelopes.filter((env: any) => {
      const matchesSearch = 
        env.name.toLowerCase().includes(search.toLowerCase()) ||
        env.wbsTags.some((tag: string) => tag.toLowerCase().includes(search.toLowerCase()));
      
      const matchesWBS = activeWBSLevels.length === 0 || 
        env.wbsTags.some((tag: string) => activeWBSLevels.some(level => tag.startsWith(level)));
      
      return matchesSearch && matchesWBS;
    });
  }, [envelopes, search, activeWBSLevels]);

  return (
    <div className="p-8 bg-black min-h-screen text-white">
      <div className="flex gap-4 mb-8">
        {/* Search */}
        <Input
          placeholder="Search envelopes, WBS, signers..."
          value={search}
          onChange={e => setSearch(e.target.value)}
          className="w-96 border-gold bg-zinc-900"
        />

        {/* WBS Toggles (13 levels) */}
        <div className="flex gap-1 flex-wrap">
          {Array.from({ length: 13 }, (_, i) => `L${i+1}`).map(level => (
            <Button
              key={level}
              variant={activeWBSLevels.includes(level) ? 'default' : 'outline'}
              size="sm"
              onClick={() => toggleWBS(level)}
              className="border-gold hover:bg-gold/10"
            >
              {level}
            </Button>
          ))}
        </div>
      </div>

      {/* Envelopes Grid */}
      <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
        {filteredEnvelopes.map((env: any) => (
          <div key={env.id} className="bg-zinc-900 border border-gold/30 rounded-2xl p-6 hover:border-gold transition-all">
            <h3 className="font-mono text-gold">{env.name}</h3>
            <div className="flex gap-2 mt-4 flex-wrap">
              {env.wbsTags.map((tag: string) => (
                <Badge key={tag} className="bg-zinc-800 text-[10px]">{tag}</Badge>
              ))}
            </div>
            <div className="text-xs text-zinc-500 mt-4">Status: {env.status}</div>
          </div>
        ))}
      </div>
    </div>
  );
}
```

### 4. Multi-PDF Merge (in EnvelopeEditor.tsx upload handler)

Add to upload flow:

```tsx
// In EnvelopeEditor or new upload component
const handleMultiUpload = async (files: FileList) => {
  const mergedPdf = await mergePdfs(Array.from(files));
  
  // Then load merged into editor
  const url = URL.createObjectURL(new Blob([mergedPdf]));
  // setDocUrl(url)
};

async function mergePdfs(files: File[]) {
  const pdfDoc = await PDFDocument.create();
  for (const file of files) {
    const bytes = await file.arrayBuffer();
    const donor = await PDFDocument.load(bytes);
    const copiedPages = await pdfDoc.copyPages(donor, donor.getPageIndices());
    copiedPages.forEach(page => pdfDoc.addPage(page));
  }
  return pdfDoc.save();
}
```
