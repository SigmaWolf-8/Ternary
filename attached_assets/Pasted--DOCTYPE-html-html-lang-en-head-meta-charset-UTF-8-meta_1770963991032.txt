let activeFilter = 'all';
let searchQuery = '';
let selectedModules = new Set();

function tierToClass(t) {
  return { core:'tag-core', developer:'tag-dev', enterprise:'tag-enterprise', user:'tag-user' }[t] || '';
}

function renderCard(mod) {
  const isSelected = selectedModules.has(mod.id);
  const matchesFilter = activeFilter === 'all' || mod.tiers.includes(activeFilter);
  const matchesSearch = !searchQuery ||
    mod.name.toLowerCase().includes(searchQuery) ||
    mod.desc.toLowerCase().includes(searchQuery) ||
    mod.id.includes(searchQuery);

  const hidden = (!matchesFilter || !matchesSearch) ? 'hidden' : '';
  const selected = isSelected ? 'selected' : '';

  const tags = mod.tiers.map(t =>
    `<span class="tag ${tierToClass(t)}">${t}</span>`
  ).join('');

  return `
    <div class="card ${hidden} ${selected}" data-id="${mod.id}" style="animation-delay: ${MODULES.indexOf(mod) * 40}ms">
      <div class="card-head">
        <div class="card-name">${mod.name}</div>
        <div class="card-version">${mod.version}</div>
      </div>
      <div class="card-desc">${mod.desc}</div>
      <div class="card-meta">
        ${tags}
        <span class="card-size">${mod.size}</span>
        <span class="card-size">${mod.format}</span>
      </div>
      <div class="card-deps">${mod.deps}</div>
      <div class="card-actions" style="margin-top:0.75rem">
        <button class="btn btn-download" onclick="downloadModule('${mod.id}')">Download</button>
        <button class="btn btn-add ${isSelected ? 'added' : ''}"
                onclick="toggleBundle('${mod.id}')">
          ${isSelected ? '✓ In Bundle' : '+ Bundle'}
        </button>
      </div>
    </div>
  `;
}

function render() {
  const grid = document.getElementById('moduleGrid');
  grid.innerHTML = MODULES.map(renderCard).join('');
  document.getElementById('bundleCount').textContent = selectedModules.size;

  const btn = document.getElementById('bundleBtn');
  if (selectedModules.size > 0) {
    btn.classList.add('ready');
  } else {
    btn.classList.remove('ready');
  }
}

  setTimeout(() => {
    // In production: window.location.href = baseUrl + filename;
    showToast(`${mod.name} — download ready. SHA-384 verified.`);
    console.log(`[DIST] Download: ${baseUrl}${filename}`);
  }, 600);
}

function toggleBundle(id) {
  if (selectedModules.has(id)) {
    selectedModules.delete(id);
  } else {
    selectedModules.add(id);
    // Auto-resolve dependencies
    const mod = MODULES.find(m => m.id === id);
    if (mod.deps.includes('Kernel')) selectedModules.add('kernel');
    if (mod.deps.includes('Constants')) selectedModules.add('constants');
    if (mod.deps.includes('Ternary Circle')) selectedModules.add('ternary-circle');
  }
  render();
}

function downloadBundle() {
  if (selectedModules.size === 0) return;
  const names = [...selectedModules].map(id => {
    const m = MODULES.find(mod => mod.id === id);
    return m ? m.name : id;
  }).join(', ');

  showToast(`Building bundle: ${selectedModules.size} modules...`);

  setTimeout(() => {
    showToast(`Bundle ready — ${selectedModules.size} modules. SHA-384 manifest included.`);
    console.log('[DIST] Bundle:', [...selectedModules]);
  }, 800);
}
