//
// The plumbing that turns a URL into 27 raw observations.
//
// Architecture:
//   1. ScanTarget: what to scan (URL, domain)
//   2. ScanContext: raw network data collected from the target
//      - Primary HTTP response (status, headers, body)
//      - DNS records (A/AAAA count)
//      - TLS metadata (version, certificates)
//      - Auxiliary page fetches (/robots.txt, /privacy, /terms, /security.txt)
//      - Timing measurements (TCP connect, TLS handshake, TTFB)
//   3. 27 probe functions: each extracts a RawValue from the ScanContext
//   4. Orchestrator: runs the full pipeline → ScanResult
//
// The scanner makes real network calls. No fixtures. No simulations.
// Efficiency: ONE primary request + a handful of auxiliary fetches,
// then all 27 probes extract from the shared context.

// ─── ScanTarget ──────────────────────────────────────────────────────────────

/// What to scan.
#[derive(Debug, Clone)]
pub struct ScanTarget {
    /// The URL to probe (e.g., "https://github.com").
    pub url: String,
    /// Extracted domain (e.g., "github.com").
    pub domain: String,
    /// Extracted scheme (e.g., "https").
    pub scheme: String,
    /// Port (443 for HTTPS, 80 for HTTP, or explicit).
    pub port: u16,
}

impl ScanTarget {
    /// Parse a URL into a scan target.
    pub fn from_url(url: &str) -> Result<Self, String> {
        let url_str = if !url.contains("://") {
            format!("https://{}", url)
        } else {
            url.to_string()
        };

        let after_scheme = url_str
            .split("://")
            .nth(1)
            .ok_or("invalid URL: no scheme")?;
        let scheme = url_str
            .split("://")
            .next()
            .unwrap_or("https")
            .to_lowercase();

        let host_port = after_scheme.split('/').next().unwrap_or(after_scheme);
        let (domain, port) = if host_port.contains(':') {
            let parts: Vec<&str> = host_port.rsplitn(2, ':').collect();
            let port = parts[0].parse::<u16>().unwrap_or(if scheme == "https" { 443 } else { 80 });
            (parts[1].to_string(), port)
        } else {
            let port = if scheme == "https" { 443 } else { 80 };
            (host_port.to_string(), port)
        };

        Ok(Self {
            url: url_str,
            domain,
            scheme,
            port,
        })
    }

    pub fn is_https(&self) -> bool {
        self.scheme == "https"
    }
}

// ─── ScanContext ──────────────────────────────────────────────────────────────

/// Raw network data collected from the target. All 27 probes extract from this.
#[derive(Debug, Clone)]
pub struct ScanContext {
    pub target: ScanTarget,

    // ── Primary HTTP response ────────────────────────────────────────
    /// HTTP status code from unauthenticated GET.
    pub http_status: u16,
    /// Response headers (lowercase keys).
    pub headers: HashMap<String, String>,
    /// Response body (first 256KB — enough for analysis).
    pub body: String,
    /// Content-Type header value.
    pub content_type: String,

    // ── Timing ───────────────────────────────────────────────────────
    /// Time to first byte (milliseconds).
    pub ttfb_ms: f64,
    /// Total response time (milliseconds).
    pub total_ms: f64,

    // ── DNS ──────────────────────────────────────────────────────────
    /// Number of A/AAAA records resolved.
    pub dns_record_count: u32,

    // ── TLS ──────────────────────────────────────────────────────────
    /// Whether TLS is available.
    pub tls_available: bool,
    /// TLS protocol version string (e.g., "TLSv1.3").
    pub tls_version: String,

    // ── Auxiliary fetches ─────────────────────────────────────────────
    /// /robots.txt content (if accessible).
    pub robots_txt: Option<String>,
    /// Whether /privacy or /privacy-policy exists (HTTP 200).
    pub has_privacy_page: bool,
    /// Whether /terms or /terms-of-service exists (HTTP 200).
    pub has_terms_page: bool,
    /// Whether /security.txt or /.well-known/security.txt exists.
    pub has_security_txt: bool,
    /// Whether /about or /about-us exists.
    pub has_about_page: bool,

    // ── Raw TCP probes ───────────────────────────────────────────────
    /// Ports that accepted connections (from a small probe set).
    pub open_ports: Vec<u16>,
}

impl ScanContext {
    /// Check if a header exists (case-insensitive).
    pub fn has_header(&self, name: &str) -> bool {
        self.headers.contains_key(&name.to_lowercase())
    }

    /// Get a header value (case-insensitive).
    pub fn header(&self, name: &str) -> Option<&str> {
        self.headers.get(&name.to_lowercase()).map(|s| s.as_str())
    }

    /// Check if body contains a string (case-insensitive).
    pub fn body_contains(&self, needle: &str) -> bool {
        let body_lower = self.body.to_lowercase();
        let needle_lower = needle.to_lowercase();
        body_lower.contains(needle_lower.as_str())
    }

    /// Count occurrences of a pattern in the body (case-insensitive).
    pub fn body_count(&self, needle: &str) -> usize {
        let body_lower = self.body.to_lowercase();
        let needle_lower = needle.to_lowercase();
        body_lower.matches(needle_lower.as_str()).count()
    }
}

// ─── Network Collection ──────────────────────────────────────────────────────

/// Collect all raw network data from a live target.
/// This is where real network calls happen.
pub fn collect(target: &ScanTarget, timeout: Duration) -> Result<ScanContext, String> {
    // ── DNS resolution ───────────────────────────────────────────────
    let dns_record_count = resolve_dns_count(&target.domain, target.port);

    // ── Primary HTTP GET ─────────────────────────────────────────────
    let start = Instant::now();

    let agent = ureq::AgentBuilder::new()
        .timeout_connect(timeout)
        .timeout_read(timeout)
        .redirects(5)
        .user_agent("CRS-Scanner/2.3 (PlenumNET TDNS)")
        .build();

    let response = agent.get(&target.url).call();
    let ttfb_ms = start.elapsed().as_secs_f64() * 1000.0;

    let (http_status, headers, body, content_type) = match response {
        Ok(resp) => {
            let status = resp.status();
            let mut hdrs = HashMap::new();
            for name in resp.headers_names() {
                if let Some(val) = resp.header(&name) {
                    hdrs.insert(name.to_lowercase(), val.to_string());
                }
            }
            let ct = resp
                .header("content-type")
                .unwrap_or("")
                .to_string();

            // Read body (cap at 256KB)
            let mut body_bytes = Vec::with_capacity(262144);
            let _ = resp.into_reader().take(262144).read_to_end(&mut body_bytes);
            let body = String::from_utf8_lossy(&body_bytes).to_string();

            (status, hdrs, body, ct)
        }
        Err(ureq::Error::Status(code, resp)) => {
            let mut hdrs = HashMap::new();
            for name in resp.headers_names() {
                if let Some(val) = resp.header(&name) {
                    hdrs.insert(name.to_lowercase(), val.to_string());
                }
            }
            let ct = resp.header("content-type").unwrap_or("").to_string();
            let mut body_bytes = Vec::new();
            let _ = resp.into_reader().take(262144).read_to_end(&mut body_bytes);
            let body = String::from_utf8_lossy(&body_bytes).to_string();
            (code, hdrs, body, ct)
        }
        Err(e) => {
            return Err(format!("HTTP request failed: {}", e));
        }
    };
    let total_ms = start.elapsed().as_secs_f64() * 1000.0;

    // ── TLS detection ────────────────────────────────────────────────
    let tls_available = target.is_https() && http_status > 0;
    let tls_version = if tls_available {
        // ureq doesn't expose TLS version directly; infer from headers
        if headers.values().any(|v| v.contains("h3") || v.contains("quic")) {
            "TLSv1.3".to_string()
        } else {
            "TLSv1.2+".to_string() // HTTPS succeeded, at least 1.2
        }
    } else {
        "none".to_string()
    };

    // ── Auxiliary fetches (quick, tolerant of failure) ────────────────
    let base = format!("{}://{}", target.scheme, target.domain);
    let robots_txt = quick_fetch(&agent, &format!("{}/robots.txt", base));
    let has_privacy_page = quick_exists(&agent, &format!("{}/privacy", base))
        || quick_exists(&agent, &format!("{}/privacy-policy", base));
    let has_terms_page = quick_exists(&agent, &format!("{}/terms", base))
        || quick_exists(&agent, &format!("{}/terms-of-service", base));
    let has_security_txt = quick_exists(&agent, &format!("{}/.well-known/security.txt", base))
        || quick_exists(&agent, &format!("{}/security.txt", base));
    let has_about_page = quick_exists(&agent, &format!("{}/about", base))
        || quick_exists(&agent, &format!("{}/about-us", base));

    // ── Port probes ──────────────────────────────────────────────────
    let probe_ports = [80, 443, 8080, 8443, 1883, 5683]; // HTTP, HTTPS, Alt-HTTP, Alt-HTTPS, MQTT, CoAP
    let open_ports = probe_ports
        .iter()
        .filter(|&&port| tcp_probe(&target.domain, port, Duration::from_secs(2)))
        .copied()
        .collect();

    Ok(ScanContext {
        target: target.clone(),
        http_status,
        headers,
        body,
        content_type,
        ttfb_ms,
        total_ms,
        dns_record_count,
        tls_available,
        tls_version,
        robots_txt,
        has_privacy_page,
        has_terms_page,
        has_security_txt,
        has_about_page,
        open_ports,
    })
}

// ─── Network Helpers ─────────────────────────────────────────────────────────

/// Count DNS A/AAAA records for a domain.
fn resolve_dns_count(domain: &str, port: u16) -> u32 {
    let addr_str = format!("{}:{}", domain, port);
    match addr_str.to_socket_addrs() {
        Ok(addrs) => addrs.count() as u32,
        Err(_) => 0,
    }
}

/// Quick GET fetch — returns body on 200, None otherwise.
fn quick_fetch(agent: &ureq::Agent, url: &str) -> Option<String> {
    match agent.get(url).call() {
        Ok(resp) if resp.status() == 200 => {
            let mut body = String::new();
            let _ = resp.into_reader().take(65536).read_to_string(&mut body);
            Some(body)
        }
        _ => None,
    }
}

/// Quick HEAD/GET check — returns true if 200-299.
fn quick_exists(agent: &ureq::Agent, url: &str) -> bool {
    match agent.head(url).call() {
        Ok(resp) => (200..300).contains(&resp.status()),
        Err(_) => {
            // Some servers reject HEAD, fall back to GET
            match agent.get(url).call() {
                Ok(resp) => (200..300).contains(&resp.status()),
                _ => false,
            }
        }
    }
}

/// TCP connect probe — returns true if port accepts connections.
fn tcp_probe(domain: &str, port: u16, timeout: Duration) -> bool {
    let addr_str = format!("{}:{}", domain, port);
    match addr_str.to_socket_addrs() {
        Ok(mut addrs) => {
            if let Some(addr) = addrs.next() {
                TcpStream::connect_timeout(&addr, timeout).is_ok()
            } else {
                false
            }
        }
        Err(_) => false,
    }
}

// ── WHO Probes ───────────────────────────────────────────────────────────────

/// Trit 1: What kind? — Entity type from TLD, WHOIS patterns, page content.
fn probe_entity_kind(ctx: &ScanContext) -> RawValue {
    let domain = &ctx.target.domain;

    // TLD-based classification
    if domain.ends_with(".gov") || domain.ends_with(".mil") || domain.ends_with(".gc.ca") {
        return RawValue::Pattern("governance".into());
    }
    if domain.ends_with(".edu") || domain.ends_with(".ac.uk") {
        return RawValue::Pattern("governance".into());
    }
    // Prefix/infix patterns: gov.uk, www.gov.uk, service.gov.uk
    if domain.starts_with("gov.") || domain.contains(".gov.") {
        return RawValue::Pattern("governance".into());
    }

    // Content-based signals for personal sites
    let personal_signals = ["personal", "my blog", "my site", "portfolio", "about me"];
    let personal_score: usize = personal_signals.iter().filter(|s| ctx.body_contains(s)).count();

    // Corporate signals
    let corp_signals = [
        "inc.", "corp.", "ltd.", "llc", "gmbh", "company", "enterprise",
        "careers", "investor", "press release", "annual report",
    ];
    let corp_score: usize = corp_signals.iter().filter(|s| ctx.body_contains(s)).count();

    if personal_score > corp_score && corp_score == 0 {
        RawValue::Pattern("personal".into())
    } else {
        RawValue::Pattern("corporate".into())
    }
}

/// Trit 2: Who's it for? — Audience from robots.txt, access patterns.
fn probe_audience(ctx: &ScanContext) -> RawValue {
    // 401/403 with no public content → private/group
    if ctx.http_status == 401 || ctx.http_status == 403 {
        // Check if there's any public-facing content
        if ctx.body.len() < 100 {
            return RawValue::Pattern("private".into());
        }
        return RawValue::Pattern("group".into());
    }

    // robots.txt analysis
    if let Some(robots) = &ctx.robots_txt {
        let disallow_count = robots.matches("Disallow:").count();
        let allow_count = robots.matches("Allow:").count();
        if disallow_count > 10 && allow_count < 3 {
            return RawValue::Pattern("group".into()); // Heavy restrictions
        }
    }

    // Default: if it responded with 200, it's public
    if ctx.http_status == 200 {
        return RawValue::Pattern("public".into());
    }

    RawValue::Pattern("public".into())
}

/// Trit 3: Who runs it? — Operator transparency score.
fn probe_operator_transparency(ctx: &ScanContext) -> RawValue {
    // Count discrete binary signals. Each is yes/no, no scoring.
    let mut signals: u32 = 0;

    // Signal 1: About page exists
    if ctx.has_about_page {
        signals += 1;
    }

    // Signal 2: Contact information present
    let contact_signals = ["contact@", "mailto:", "contact us", "phone:", "tel:"];
    if contact_signals.iter().any(|s| ctx.body_contains(s)) {
        signals += 1;
    }

    // Signal 3: Legal entity name present
    let legal_signals = ["inc.", "corp.", "ltd.", "llc", "gmbh", "registered"];
    if legal_signals.iter().any(|s| ctx.body_contains(s)) {
        signals += 1;
    }

    // Signal 4: Physical address present
    let address_patterns = [" street", " avenue", " road", " blvd", " drive"];
    if address_patterns.iter().any(|s| ctx.body_contains(s)) {
        signals += 1;
    }

    // Signal 5: Government/education domains are always transparent
    let domain = &ctx.target.domain;
    if domain.ends_with(".gov") || domain.ends_with(".edu")
        || domain.starts_with("gov.") || domain.contains(".gov.")
    {
        signals = 5; // Full transparency for government entities
    }

    RawValue::Numeric(signals as f64)
}

/// Trit 4: Who hosts it? — Hosting model from headers, ASN patterns.
fn probe_hosting_model(ctx: &ScanContext) -> RawValue {
    let headers_str: String = ctx
        .headers
        .iter()
        .map(|(k, v)| format!("{}: {}", k, v))
        .collect::<Vec<_>>()
        .join(" ")
        .to_lowercase();

    // Cloud provider fingerprints
    let cloud_signals = [
        "cloudflare", "amazonaws", "azure", "google", "gcp", "akamai",
        "fastly", "vercel", "netlify", "heroku", "fly.io", "railway",
        "cf-ray", "x-amz-", "x-goog-", "x-azure-", "x-vercel-",
    ];
    if cloud_signals.iter().any(|s| headers_str.contains(s)) {
        return RawValue::Pattern("cloud".into());
    }

    // Hosting provider signals
    let provider_signals = [
        "nginx", "apache", "litespeed", "cpanel", "plesk",
        "x-powered-by", "digitalocean", "linode", "vultr", "ovh",
    ];
    if provider_signals.iter().any(|s| headers_str.contains(s)) {
        return RawValue::Pattern("provider".into());
    }

    // Single IP, no cloud headers → likely self-hosted or small provider
    if ctx.dns_record_count <= 1 {
        return RawValue::Pattern("self".into());
    }

    RawValue::Pattern("provider".into()) // Default
}

// ── WHAT Probes ──────────────────────────────────────────────────────────────

/// Trit 5: What is it? — Form factor from Content-Type, protocol behavior.
fn probe_form_factor(ctx: &ScanContext) -> RawValue {
    let ct = ctx.content_type.to_lowercase();

    // IoT/device protocols
    if ctx.open_ports.iter().any(|&p| p == 1883 || p == 5683) {
        return RawValue::Pattern("device".into());
    }

    // API-only (JSON/XML, no HTML)
    if (ct.contains("application/json") || ct.contains("application/xml"))
        && !ct.contains("html")
        && !ctx.body_contains("<html")
    {
        return RawValue::Pattern("app".into());
    }

    // SPA detection (minimal HTML shell + heavy JS)
    if ctx.body_contains("<html") {
        let script_count = ctx.body_count("<script");
        let body_text_len = ctx.body.len();
        // SPA: lots of scripts, minimal body text relative to scripts
        if script_count > 5 && body_text_len < 50000 {
            return RawValue::Pattern("app".into());
        }
    }

    // Default: if it serves HTML, it's a website
    if ct.contains("html") || ctx.body_contains("<html") {
        return RawValue::Pattern("website".into());
    }

    RawValue::Pattern("website".into())
}

/// Trit 6: What's on it? — Content type from MIME analysis.
fn probe_content_type(ctx: &ScanContext) -> RawValue {
    let ct = ctx.content_type.to_lowercase();

    // Live/streaming indicators
    let live_signals = [
        "text/event-stream", "application/x-ndjson", "multipart/x-mixed-replace",
    ];
    if live_signals.iter().any(|s| ct.contains(s)) {
        return RawValue::Pattern("live".into());
    }
    if ctx.body_contains("websocket") || ctx.has_header("upgrade") {
        return RawValue::Pattern("live".into());
    }

    // Media indicators
    let media_signals = [
        "image/", "video/", "audio/", "application/octet-stream",
    ];
    if media_signals.iter().any(|s| ct.contains(s)) {
        return RawValue::Pattern("media".into());
    }

    // Check body for embedded media density
    let media_tags = ctx.body_count("<video")
        + ctx.body_count("<audio")
        + ctx.body_count("<img")
        + ctx.body_count("<picture")
        + ctx.body_count("<canvas");
    let text_len = ctx.body.len();
    if media_tags > 10 && text_len > 0 {
        return RawValue::Pattern("media".into());
    }

    RawValue::Pattern("text".into())
}

/// Trit 7: Who uses it? — People vs software vs both.
fn probe_consumer_type(ctx: &ScanContext) -> RawValue {
    let has_html = ctx.body_contains("<html") || ctx.content_type.contains("html");
    let has_api = ctx.body_contains("\"api\"")
        || ctx.body_contains("/api/")
        || ctx.content_type.contains("json")
        || ctx.has_header("x-ratelimit-limit"); // API rate limiting

    let has_css = ctx.body_contains("<link") && ctx.body_contains("stylesheet");
    let has_ui = has_html && (has_css || ctx.body_contains("<button") || ctx.body_contains("<form"));

    match (has_ui, has_api) {
        (true, true) => RawValue::Pattern("both".into()),
        (true, false) => RawValue::Pattern("people".into()),
        (false, true) => RawValue::Pattern("software".into()),
        (false, false) => {
            if has_html {
                RawValue::Pattern("people".into())
            } else {
                RawValue::Pattern("software".into())
            }
        }
    }
}

/// Trit 8: Does it think? — ML/AI capability detection.
fn probe_intelligence(ctx: &ScanContext) -> RawValue {
    // Count discrete binary ML signals. Each category is yes/no.
    let mut signals: u32 = 0;

    // Signal 1: ML-specific API endpoints present
    let ml_endpoints = [
        "/predict", "/inference", "/v1/models", "/v1/completions",
        "/classify", "/embed", "/generate", "/ai/", "/ml/",
    ];
    if ml_endpoints.iter().any(|ep| ctx.body_contains(ep)) {
        signals += 1;
    }

    // Signal 2: ML framework or service references
    let ml_frameworks = [
        "tensorflow", "pytorch", "openai", "anthropic", "hugging",
        "copilot", "gpt", "llm",
    ];
    if ml_frameworks.iter().any(|fw| ctx.body_contains(fw)) {
        signals += 1;
    }

    // Signal 3: Personalization/recommendation indicators
    let personalization = ["recommended for you", "you might like", "personalized"];
    if personalization.iter().any(|sig| ctx.body_contains(sig)) {
        signals += 1;
    }

    // Signal 4: Search with ranking (implies ML-driven relevance)
    if ctx.body_contains("search") && ctx.body_contains("results") {
        signals += 1;
    }

    // Signal 5: ML-specific headers
    if ctx.has_header("x-model-version") || ctx.has_header("x-inference-time") {
        signals += 1;
    }

    RawValue::Numeric(signals as f64)
}

// ── WHERE Probes ─────────────────────────────────────────────────────────────

/// Trit 9: Who can see it? — 3 binary visibility signals.
/// Signal 1: site responds (HTTP status != 0)
/// Signal 2: no auth challenge (no 401/403)
/// Signal 3: serves public content (200-299)
fn probe_visibility(ctx: &ScanContext) -> RawValue {
    let mut signals: u32 = 0;
    if ctx.http_status > 0 { signals += 1; }             // Responds at all
    if ctx.http_status != 401 && ctx.http_status != 403 { signals += 1; } // No auth challenge
    if (200..300).contains(&ctx.http_status) { signals += 1; }           // Serves content
    RawValue::Numeric(signals as f64)
}

/// Trit 10: Do I need to log in? — Auth model detection.
fn probe_auth_model(ctx: &ScanContext) -> RawValue {
    // WWW-Authenticate header = explicit auth challenge
    if ctx.has_header("www-authenticate") {
        let auth_val = ctx.header("www-authenticate").unwrap_or("").to_lowercase();
        if auth_val.contains("bearer") || auth_val.contains("negotiate") {
            return RawValue::Pattern("mfa".into());
        }
        return RawValue::Pattern("password".into());
    }

    // Login form detection
    let has_login_form = ctx.body_contains("type=\"password\"")
        || ctx.body_contains("type='password'")
        || ctx.body_contains("login")
            && (ctx.body_contains("<form") || ctx.body_contains("sign in"));

    // MFA/certificate indicators — ONLY check in auth context.
    // Wikipedia, blogs, and docs mention "two-factor", "security key", etc.
    // in article text. These are not auth challenges.
    let in_auth_context = has_login_form
        || ctx.has_header("www-authenticate")
        || ctx.body_contains("id=\"login\"")
        || ctx.body_contains("action=\"/login")
        || ctx.body_contains("action=\"/signin");

    if in_auth_context {
        let mfa_signals = [
            "two-factor", "2fa", "multi-factor", "mfa", "authenticator",
            "security key", "biometric", "client certificate",
        ];
        if mfa_signals.iter().any(|s| ctx.body_contains(s)) {
            return RawValue::Pattern("mfa".into());
        }
    }

    if has_login_form {
        return RawValue::Pattern("password".into());
    }

    RawValue::Pattern("none".into())
}

/// Trit 11: How many servers? — 6 binary infrastructure signals.
/// Signal 1: DNS resolves at all (any A/AAAA records)
/// Signal 2: Multiple DNS records (>1)
/// Signal 3: Many DNS records (>4)
/// Signal 4: CDN provider header present
/// Signal 5: CDN cache header present (x-cache/x-cache-hits)
/// Signal 6: Proxy chain detected (Via header)
fn probe_infra_scale(ctx: &ScanContext) -> RawValue {
    let mut signals: u32 = 0;

    // Signal 1: DNS resolves
    if ctx.dns_record_count > 0 { signals += 1; }

    // Signal 2: Multiple records
    if ctx.dns_record_count > 1 { signals += 1; }

    // Signal 3: Many records
    if ctx.dns_record_count > 4 { signals += 1; }

    // Signal 4: CDN provider header
    let cdn_provider = ctx.has_header("cf-ray")
        || ctx.has_header("x-cdn")
        || ctx.has_header("x-amz-cf-id")
        || ctx.has_header("x-served-by")
        || ctx.has_header("x-vercel-id")
        || ctx.has_header("x-github-request-id")
        || ctx.has_header("x-fastly-request-id")
        || ctx.has_header("x-timer")
        || ctx.has_header("x-netlify-request-id");
    if cdn_provider { signals += 1; }

    // Signal 5: CDN cache header
    if ctx.has_header("x-cache") || ctx.has_header("x-cache-hits") { signals += 1; }

    // Signal 6: Proxy chain
    if ctx.has_header("via") { signals += 1; }

    RawValue::Numeric(signals as f64)
}

/// Trit 12: What connection? — Protocol from port scan + headers.
fn probe_connection_protocol(ctx: &ScanContext) -> RawValue {
    // WebSocket detection
    if ctx.has_header("upgrade") || ctx.body_contains("websocket") || ctx.body_contains("wss://") {
        return RawValue::Pattern("websocket".into());
    }

    // gRPC detection
    if ctx.content_type.contains("grpc") || ctx.has_header("grpc-status") {
        return RawValue::Pattern("websocket".into()); // gRPC uses HTTP/2, same trit as WebSocket
    }

    // Raw TCP (IoT protocols on non-HTTP ports)
    let non_http_ports: Vec<u16> = ctx
        .open_ports
        .iter()
        .filter(|&&p| p != 80 && p != 443 && p != 8080 && p != 8443)
        .copied()
        .collect();
    if !non_http_ports.is_empty() && ctx.http_status == 0 {
        return RawValue::Pattern("tcp".into());
    }

    RawValue::Pattern("http".into())
}

/// Trit 13: What era? — 6 binary modern protocol stack signals.
/// Each signal is a header or feature that didn't exist before its era.
/// Signal 1: alt-svc header (HTTP/3, 2020s)
/// Signal 2: permissions-policy header (2021+)
/// Signal 3: NEL or report-to header (2019+)
/// Signal 4: cross-origin policy headers (COEP/COOP, 2020+)
/// Signal 5: content-security-policy header (2012+, but indicates active security)
/// Signal 6: modern JS framework detected in body (__NEXT_DATA__, __NUXT__, etc.)
fn probe_era(ctx: &ScanContext) -> RawValue {
    let mut signals: u32 = 0;

    // Signal 1: alt-svc (HTTP/3)
    if ctx.has_header("alt-svc") { signals += 1; }

    // Signal 2: permissions-policy
    if ctx.has_header("permissions-policy") { signals += 1; }

    // Signal 3: NEL or report-to
    if ctx.has_header("nel") || ctx.has_header("report-to") { signals += 1; }

    if ctx.has_header("cross-origin-embedder-policy")
        || ctx.has_header("cross-origin-opener-policy") { signals += 1; }

    // Signal 5: CSP
    if ctx.has_header("content-security-policy") { signals += 1; }

    // Signal 6: modern framework in body
    let modern_frameworks = ["__NEXT_DATA__", "__NUXT__", "svelte", "remix", "astro", "vite"];
    if modern_frameworks.iter().any(|f| ctx.body_contains(f)) { signals += 1; }

    RawValue::Numeric(signals as f64)
}

/// Trit 15: What kind of data? — Data freshness from headers and content.
fn probe_data_freshness(ctx: &ScanContext) -> RawValue {
    // Streaming/live indicators
    if ctx.content_type.contains("event-stream")
        || ctx.body_contains("websocket")
        || ctx.body_contains("real-time")
        || ctx.body_contains("live feed")
        || ctx.body_contains("streaming")
    {
        return RawValue::Pattern("live".into());
    }

    // Check Last-Modified / Date headers for freshness
    if let Some(cache_control) = ctx.header("cache-control") {
        if cache_control.contains("no-cache") || cache_control.contains("no-store") {
            return RawValue::Pattern("current".into()); // Dynamic, always fresh
        }
        if cache_control.contains("max-age=31536000") || cache_control.contains("immutable") {
            return RawValue::Pattern("historical".into()); // Immutable = archive
        }
    }

    // Dynamic content signals
    if ctx.body_contains("datetime") || ctx.body_contains("timestamp") || ctx.body_contains("ago") {
        return RawValue::Pattern("current".into());
    }

    RawValue::Pattern("current".into()) // Default: most sites serve current content
}

/// Trit 16: Is it real-time? — Latency measurement (actual TTFB).
/// Trit 16: Is it real-time? — 6 binary timeliness signals.
/// Signal 1: dynamic cache policy (no-cache/no-store)
/// Signal 2: freshness indicators in body (datetime, "ago", timestamps)
/// Signal 3: WebSocket detected (Upgrade header or ws/wss in body)
/// Signal 4: SSE/EventSource detected (event-stream content type)
/// Signal 5: gRPC detected (grpc content type or grpc-status header)
/// Signal 6: streaming content type or self-declared real-time
fn probe_latency_profile(ctx: &ScanContext) -> RawValue {
    let mut signals: u32 = 0;

    // Signal 1: dynamic cache policy
    if let Some(cc) = ctx.header("cache-control") {
        let cc_lower = cc.to_lowercase();
        if cc_lower.contains("no-cache") || cc_lower.contains("no-store") {
            signals += 1;
        }
    }

    // Signal 2: freshness indicators in body
    if ctx.body_contains("datetime") || ctx.body_contains("timestamp")
        || ctx.body_contains(" ago") || ctx.body_contains("updated") {
        signals += 1;
    }

    // Signal 3: WebSocket
    if ctx.has_header("upgrade") || ctx.body_contains("websocket") || ctx.body_contains("wss://") {
        signals += 1;
    }

    // Signal 4: SSE/EventSource
    if ctx.content_type.contains("event-stream") || ctx.body_contains("eventsource") {
        signals += 1;
    }

    // Signal 5: gRPC
    if ctx.content_type.contains("grpc") || ctx.has_header("grpc-status") {
        signals += 1;
    }

    // Signal 6: streaming content or self-declared
    if ctx.body_contains("real-time") || ctx.body_contains("realtime")
        || ctx.body_contains("live feed")
        || ctx.content_type.contains("x-ndjson")
        || ctx.content_type.contains("x-mixed-replace") {
        signals += 1;
    }

    RawValue::Numeric(signals as f64)
}

/// Trit 24: Does it remember me? — 3 binary cookie persistence signals.
/// Signal 1: has any Set-Cookie header
/// Signal 2: has persistent cookie (expires or max-age present)
/// Signal 3: has long-lived cookie (max-age > 86400 = 1 day)
fn probe_state_persistence(ctx: &ScanContext) -> RawValue {
    let mut signals: u32 = 0;

    if let Some(cookie) = ctx.header("set-cookie") {
        // Signal 1: any cookie at all
        signals += 1;

        let cookie_lower = cookie.to_lowercase();

        // Signal 2: persistent (has explicit expiry)
        if cookie_lower.contains("expires=") || cookie_lower.contains("max-age=") {
            signals += 1;
        }

        // Signal 3: long-lived (max-age > 86400)
        if cookie_lower.contains("max-age=") {
            if let Some(start) = cookie_lower.find("max-age=") {
                let rest = &cookie_lower[start + 8..];
                let value_str: String = rest.chars().take_while(|c| c.is_ascii_digit()).collect();
                if let Ok(seconds) = value_str.parse::<u64>() {
                    if seconds > 86400 { signals += 1; }
                }
            }
        }
    }

    RawValue::Numeric(signals as f64)
}

fn probe_encryption(ctx: &ScanContext) -> RawValue {

    if !ctx.tls_available {

    // Signal 2: HSTS header
    if ctx.has_header("strict-transport-security") {

    // Signal 3: Content Security Policy
    if ctx.has_header("content-security-policy") {
        signals += 1;
    }

    // Signal 4: security.txt exists
    if ctx.has_security_txt {
        signals += 1;
    }

    // Signal 5: X-Content-Type-Options
    if ctx.has_header("x-content-type-options") {
        signals += 1;
    }

    // Signal 6: X-Frame-Options
    if ctx.has_header("x-frame-options") {
        signals += 1;
    }

    RawValue::Numeric(signals as f64)
}

// ORCHESTRATOR — Full scan pipeline: target → measurements → address
// ═══════════════════════════════════════════════════════════════════════════

/// The result of a full scan.
#[derive(Debug, Clone)]
pub struct FullScanResult {
    /// The target that was scanned.
    pub target: ScanTarget,
    /// The raw scan context (all network data).
    pub context: ScanContext,
    /// The 27 raw measurements extracted from the context.
    pub measurements: Vec<RawValue>,
    /// The derived 27-trit address.
    pub address: CubeAddr,
    /// The trit derivation results with confidence.
    pub derivations: Vec<(Trit, Confidence)>,
    /// Scan hash (BLAKE3 of measurements).
    pub scan_hash: ScanHash,
}

/// Scan a live target and derive its 27-trit address.
///
/// This is the full pipeline:
/// 1. Parse target URL
/// 2. Collect raw network data (HTTP, DNS, TLS, auxiliary pages)
/// 3. Extract 27 measurements from the context
/// 4. Run each measurement through its DerivationRule
/// 5. Assemble the CubeAddr
///
/// No fixtures. No simulations. Real network inspection.
pub fn scan(url: &str) -> Result<FullScanResult, String> {
    scan_with_timeout(url, Duration::from_secs(10))
}

/// Scan with configurable timeout.
pub fn scan_with_timeout(url: &str, timeout: Duration) -> Result<FullScanResult, String> {
    // 1. Parse target
    let target = ScanTarget::from_url(url)?;

    // 2. Collect raw network data
    let context = collect(&target, timeout)?;

    // 3. Extract 27 measurements
    let measurements = extract_measurements(&context);

    // 4. Derive trits
    let rules = all_rules();
    let mut trits = [Trit::V1; DIMENSIONS];
    let mut derivations = Vec::with_capacity(DIMENSIONS);
    let mut scan_measurements = Vec::with_capacity(DIMENSIONS);

    for (i, (rule, raw)) in rules.iter().zip(measurements.iter()).enumerate() {
        let (trit, confidence) = rule
            .derive(raw)
            .map_err(|e| format!("derivation failed at dim {}: {}", i + 1, e))?;
        trits[i] = trit;
        derivations.push((trit, confidence));
        scan_measurements.push(ScanMeasurement {
            dim: i,
            raw: raw.clone(),
            confidence,
        });
    }

    let address = CubeAddr::new(trits);

    // 5. Compute scan hash
    let scan_result = ScanResult::new(
        target.url.clone(),
        0, // Timestamp filled by CRS at registration
        scan_measurements,
        trits,
    );

    Ok(FullScanResult {
        target,
        context,
        measurements,
        address,
        derivations,
        scan_hash: scan_result.scan_hash,
    })
}

/// Pretty-print a full scan result with dimension labels.
pub fn format_scan_report(result: &FullScanResult) -> String {
    use crate::schema::SCHEMA;

    let mut report = String::new();
    report.push_str(&format!("═══ CRS SCAN REPORT ═══\n"));
    report.push_str(&format!("Target: {}\n", result.target.url));
    report.push_str(&format!("Address: {}\n", result.address));
    report.push_str(&format!("HPTP-mandatory: {}\n", result.address.is_hptp_mandatory()));
    report.push_str(&format!("Scan hash: {}\n", result.scan_hash));
    report.push_str(&format!("───────────────────────\n"));

    let mut current_category = "";
    for (i, (trit, confidence)) in result.derivations.iter().enumerate() {
        let dim = &SCHEMA[i];
        let cat_prefix = dim.category.prefix();
        if cat_prefix != current_category {
            report.push_str(&format!("\n  {} — {}\n", cat_prefix, dim.category.root_question()));
            current_category = cat_prefix;
        }
        let conf_marker = match confidence {
            Confidence::High => "██",
            Confidence::Medium => "▓▓",
            Confidence::Low => "░░",
        };
        report.push_str(&format!(
            "  {:2}. {} {} → {} ({})\n",
            dim.number,
            conf_marker,
            dim.question,
            dim.label(*trit),
            trit
        ));
    }

    report.push_str(&format!("\n═══════════════════════\n"));
    report
}

    #[test]
    #[cfg(feature = "network-tests")]
    fn live_scan_github() {
        // This test makes real network calls to github.com.
        // It will fail if the network is unavailable.
        let result = match scan("https://github.com") {
            Ok(r) => r,
            Err(e) => {
                eprintln!("SKIP live_scan_github: {}", e);
                return;
            }
        };

        // Basic sanity checks on a live scan
        assert_eq!(result.derivations.len(), 27);
        assert_eq!(result.measurements.len(), 27);

        // GitHub should be:
        // - Corporate (trit 1 = 2)
        assert_eq!(result.address.dim(1), Trit::V2, "GitHub should be corporate");
        // - Everyone can see it (trit 9 = 3)
        assert_eq!(result.address.dim(9), Trit::V3, "GitHub should be visible to everyone");
        // - HTTPS (trit 25 should score high)
        let encryption_trit = result.address.dim(25);
        assert!(encryption_trit.value() >= 2, "GitHub should have at least basic TLS");

        // Print the full report
        let report = format_scan_report(&result);
        eprintln!("{}", report);
    }
