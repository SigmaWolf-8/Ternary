// NUMERICAL METHODS — No Math.random() anywhere
// ================================================================
const Num = {
    // 4th-order Runge-Kutta for dy/dt = f(t, y)
    rk4Step(f, t, y, dt) {
        const k1 = f(t, y);
        const k2 = f(t + dt/2, y.map((yi, i) => yi + dt/2 * k1[i]));
        const k3 = f(t + dt/2, y.map((yi, i) => yi + dt/2 * k2[i]));
        const k4 = f(t + dt, y.map((yi, i) => yi + dt * k3[i]));
        return y.map((yi, i) => yi + dt/6 * (k1[i] + 2*k2[i] + 2*k3[i] + k4[i]));
    },

    // Discrete Fourier Transform — magnitude spectrum
    dft(signal) {
        const N = signal.length;
        const mag = new Array(Math.floor(N/2));
        for (let k = 0; k < mag.length; k++) {
            let re = 0, im = 0;
            for (let n = 0; n < N; n++) {
                const angle = -2 * Math.PI * k * n / N;
                re += signal[n] * Math.cos(angle);
                im += signal[n] * Math.sin(angle);
            }
            mag[k] = Math.sqrt(re*re + im*im) / N;
        }
        return mag;
    },

    // Trapezoidal integration
    trapz(y, dx) {
        let sum = 0;
        for (let i = 1; i < y.length; i++) {
            sum += (y[i-1] + y[i]) / 2 * dx;
        }
        return sum;
    },

    // Deterministic initial phase distribution (uniform on [0, 2π])
    // Using irrational rotation (golden angle) — no RNG
    goldenPhases(N) {
        const phi = (1 + Math.sqrt(5)) / 2; // golden ratio
        return Array.from({length: N}, (_, i) => (2 * Math.PI * i * phi) % (2 * Math.PI));
    },

    // Deterministic natural frequency distribution for vasomotion oscillators
    // Lorentzian (Cauchy) distribution via quantile function
    lorentzianFreqs(N, center, width) {
        return Array.from({length: N}, (_, i) => {
            const u = (i + 0.5) / N; // uniform on (0,1) deterministically
            return center + width * Math.tan(Math.PI * (u - 0.5));
        });
    }
};

// PHYSICS COMPUTATIONS — Every function from V2.3 equations

const Phys = {
    // === Section 2.1: Distributed Parameters ===

    // L_v = ρ/(πr²)  [kg/m⁵]
    inertance(r) { return C.rho_blood / (Math.PI * r * r); },

    // C_v = 2πr³/(Eh)  [m⁵/Pa]
    compliance(r, E, h) { return 2 * Math.PI * r * r * r / (E * h); },

    // R_v = 8μ/(πr⁴)  [Pa·s/m⁵]
    resistance(r) { return 8 * C.mu_blood / (Math.PI * Math.pow(r, 4)); },

    // G_v = 2πr·L_p/h_wall  [m⁵/(Pa·s)] (transmural leakage)
    leakance(r, Lp, h_wall) { return 2 * Math.PI * r * Lp / h_wall; },

    // === Section 2.2: Wave Speed ===

    // c₀ = √(Eh/(2ρr))  [m/s] — Moens-Korteweg
    waveSpeed(E, h, r) { return Math.sqrt(E * h / (2 * C.rho_blood * r)); },

    // Z₀ = √(L_v/C_v) = √(ρEh/(2π²r⁵))  [Pa·s/m³]
    impedance(r, E, h) {
        return Math.sqrt(this.inertance(r) / this.compliance(r, E, h));
    },

    // === Section 2.2: Streaming Potential — Full Helmholtz-Smoluchowski ===
    // E_s = ζε·ΔP/(ηκ)  [V/m]
    streamingPotential(deltaP) {
        return C.zeta_endo * C.epsilon_w * deltaP / (C.mu_blood * C.sigma_blood);
    },

    // Shear stress → pressure gradient conversion for tube flow
    // τ_w = r/2 · dP/dz, so dP = 2τ_w/r per unit length
    shearToPressureGrad(tau_w, r) { return 2 * tau_w * 0.1 / r; }, // dyn/cm² to Pa, per segment length

    // === Section 2.5: EM Field from Biot-Savart ===
    // |B| ~ μ₀·I/(2πd) for current I at distance d
    biotSavart(I, d) { return C.mu_0 * I / (2 * Math.PI * d); },

    // === Section 2.6.1: Kramers Escape Rate ===
    // Γ = (ω_a·ω_b)/(2πγ) · exp(−ΔU/(k_B·T))
    kramersRate(deltaU, gamma, omega_a, omega_b) {
        return (omega_a * omega_b) / (2 * Math.PI * gamma) * Math.exp(-deltaU / (C.k_B * C.T));
    },

    // Piezo1 activation energy barrier as function of membrane voltage shift
    // ΔU(V_m) = ΔU₀ − z_eff·e·V_m  (voltage-dependent barrier)
    piezo1Barrier(deltaU0, z_eff, V_m) {
        return deltaU0 - z_eff * 1.602e-19 * V_m;
    },

    // SR SNR gain: SNR_out/SNR_in for bistable system (McNamara-Wiesenfeld theory)
    // SNR_out = (A²·Γ₀)/(4·D²) · exp(−ΔU/(D)) for signal amplitude A, noise intensity D
    srSNR(A_signal, deltaU, D_noise) {
        if (D_noise < 1e-30) return 0;
        const gamma0 = this.kramersRate(deltaU, 1e9, 1e10, 1e10);
        return (A_signal * A_signal * gamma0) / (4 * D_noise * D_noise) *
               Math.exp(-deltaU / (D_noise + 1e-30));
    },

    // === Section 2.6.2: Donnan Shift ===
    // Δ[Ca²⁺]/[Ca²⁺] = z·F·E_s·d/(R·T)
    donnanShift(Es, d) {
        return 2 * C.F * Math.abs(Es) * d / (C.R * C.T);
    },

    // Hill amplification: ΔNO/NO = n_Hill × Δ[Ca²⁺]/[Ca²⁺]
    hillAmplification(nHill, caShift) { return nHill * caShift; },

    // === Section 2.6.3: Coherent EM Summation ===
    // Interpolation: |B_total| = |B₁| × (√N + (N − √N) × κ_geom × C_phase)
    // κ_geom accounts for toroidal geometric cancellation
    emSummation(N, B1, C_phase, kappa_geom) {
        const sqrtN = Math.sqrt(N);
        return B1 * (sqrtN + (N - sqrtN) * kappa_geom * C_phase);
    },

    // === Section 4.3: Piezo1 Boltzmann Gating ===
    // P_open = 1/(1 + exp(−(τ_w − τ_half)/k_τ))
    piezo1Open(tau_w) {
        return 1 / (1 + Math.exp(-(tau_w - C.piezo1_tau_half) / C.piezo1_k_tau));
    },

    // === Section 4.3: Hydraulic permeability ===
    // L_p = L_p0·(1 + β·P_open)
    hydraulicPerm(Lp0, beta, pOpen) { return Lp0 * (1 + beta * pOpen); },

    // === Section 3.1: Murray's Law ===
    // r_daughter = r_parent × 2^(-1/3) for symmetric bifurcation
    murrayDaughter(r_parent) { return r_parent * Math.pow(2, -1/3); },

    // Segment count in fractal tree: 2^(d+1) − 2
    treeSegments(depth) { return Math.pow(2, depth + 1) - 2; },

    // === Section 3.1: Reflection Coefficient ===
    // Γ = (Z_d − Z_p)/(Z_d + Z_p) where Z_d = Z_d1·Z_d2/(Z_d1+Z_d2) for parallel daughters
    reflectionCoeff(Z_parent, Z_d1, Z_d2) {
        const Z_load = (Z_d1 * Z_d2) / (Z_d1 + Z_d2);
        return (Z_load - Z_parent) / (Z_load + Z_parent);
    },

    // === Section B: Womersley Number ===
    // α = r·√(ωρ/μ)
    womersley(r, f) {
        return r * Math.sqrt(2 * Math.PI * f * C.rho_blood / C.mu_blood);
    },

    // === Section 8.1: FitzHugh-Nagumo Lymphangion ===
    // dV/dt = V − V³/3 − W + I_stretch
    // dW/dt = ε(V + a − bW)
    fhnDerivatives(V, W, I_stretch, epsilon, a, b) {
        return [
            V - V*V*V/3 - W + I_stretch,
            epsilon * (V + a - b*W)
        ];
    }
};

                hz_check: Math.abs(delta_f) < 0.1     // §7.5 0.1 Hz rule
            }
        };
    }

        let V = -1.0, W = -0.5; // deterministic initial state (rest)
        const vHistory = [];
        const tLabels = [];
        const sampleRate = Math.max(1, Math.floor(steps / 300));

        for (let s = 0; s <= steps; s++) {
            const t = s * dt;
            if (s % sampleRate === 0) {
                vHistory.push(V);
                tLabels.push(t.toFixed(2));
            }
            if (s === steps) break;

            // PEMF forcing: Mechanism 2 contribution as current injection
            const I_pemf = NO_relative * 0.5 * Math.sin(2 * Math.PI * fPEMF * t);
            // Natural myogenic stretch input (from preload, including Mechanism 3)
            const I_stretch = 0.5 * (1 + preload_increase);

            // RK4 step
            const deriv = (tt, state) => {
                const [vv, ww] = state;
                const I_p = NO_relative * 0.5 * Math.sin(2 * Math.PI * fPEMF * tt);
                return Phys.fhnDerivatives(vv, ww, I_stretch + I_p, epsilon, a, b_fhn);
            };
            const newState = Num.rk4Step(deriv, t, [V, W], dt);
            V = newState[0];
            W = newState[1];
        }

        // Compute oscillation frequency via zero-crossing count
        let crossings = 0;
        for (let i = 1; i < vHistory.length; i++) {
            if (vHistory[i-1] < 0 && vHistory[i] >= 0) crossings++;
        }
        const measuredFreq = crossings / T_total;

        // Entrainment metric: how close is measured freq to PEMF freq
        const detuning = Math.abs(measuredFreq - fPEMF) / fPEMF;
        const entrained = detuning < 0.15;

        CoherenceState.update('lymph', entrained ? 0.8 - detuning : 0.3 + measuredFreq / fNat * 0.2, 'lymphangion_pemf');

            chart: { labels: tLabels, data: vHistory, label: 'Lymphangion FHN Membrane Potential V(t)' },

fsi: {
    title: 'fsi_1d_solver.py',
    desc: '1D FSI via method of characteristics (MOC). Telegraph equations solved: forward/backward Riemann invariants propagated at ±c₀ through elastic tube. No synthetic waveform — pressure emerges from cardiac flow input + wall mechanics.',
    tag: 'Method of Characteristics — 1D Telegraph',
    tagClass: 'tag-verified',
    status: '✓ MOC Solver',
    params: [
        { id: 'fsi-hr', label: 'Heart Rate (BPM)', min: 50, max: 120, value: 72, step: 1 },

        { id: 'fsi-E', label: 'Young\'s Modulus (kPa)', min: 200, max: 1200, value: 400, step: 50, unit: ' kPa' }
    ],
    run(p) {
        const hr = p['fsi-hr'], sv_mL = p['fsi-sv'], E = p['fsi-E'] * 1000;
        const h = 1.5e-3; // aortic wall thickness
        const r0 = 0.013; // 13 mm aortic radius
        const L_aorta = 0.4; // 40 cm aorta length
        const f = hr / 60;

        // Derived quantities
        const c0 = Phys.waveSpeed(E, h, r0);
        const Z0 = Phys.impedance(r0, E, h);
        const alpha = Phys.womersley(r0, f);
        const A0 = Math.PI * r0 * r0; // cross-section

        // MOC grid
        const Nz = 40; // spatial nodes
        const dz = L_aorta / Nz;
        const dt_cfl = 0.8 * dz / c0; // CFL condition
        const T_total = 3 / f; // 3 cardiac cycles
        const Nt = Math.ceil(T_total / dt_cfl);
        const dt = T_total / Nt;

        // State arrays: p(z,t), Q(z,t)
        let p_curr = new Float64Array(Nz + 1); // pressure
        let Q_curr = new Float64Array(Nz + 1); // flow rate
        let p_next = new Float64Array(Nz + 1);
        let Q_next = new Float64Array(Nz + 1);

        // Initialize at mean arterial pressure
        p_curr.fill(93.3 * 133.322); // 93.3 mmHg → Pa
        Q_curr.fill(0);

        // Distributed parameters
        const L_v = Phys.inertance(r0);
        const C_v = Phys.compliance(r0, E, h);
        const R_v = Phys.resistance(r0);

        // g(phase) is a physiological ejection waveform
        const Q_mean = (sv_mL * 1e-6) * f; // m³/s

        const pAtMid = []; // record midpoint pressure
        const tLabels = [];

        const Z_term = 3 * Z0;

            const t = n * dt;

            // Cardiac ejection waveform at inlet (z=0)
            const phase = (t * f) % 1;
            let Q_in;

            // ∂p/∂t = −(1/C_v)·∂Q/∂z − (G_v/C_v)·p

            }

            Q_next[0] = Q_in;
            // Forward characteristic: p = Z0 × Q + (incoming wave)
            p_next[0] = p_curr[0] + Z0 * (Q_next[0] - Q_curr[0]);

            // Outlet BC: Windkessel reflection
            // p_out = Z_term × Q_out
            Q_next[Nz] = Q_curr[Nz] + dt * (-(p_curr[Nz] - p_curr[Nz-1]) / (dz * L_v) - R_v / L_v * Q_curr[Nz]);
            p_next[Nz] = Z_term * Q_next[Nz] * 0.001 + p_curr[Nz] * 0.999; // relaxed BC

            // Swap
            [p_curr, p_next] = [p_next, p_curr];
            [Q_curr, Q_next] = [Q_next, Q_curr];

            // Record
            if (n % sampleEvery === 0) {
                pAtInlet.push(p_curr[0] / 133.322); // Pa → mmHg
                pAtMid.push(p_curr[Math.floor(Nz/2)] / 133.322);

            }

        const diastolic = Math.min(...lastCycle);

        const rr_ms = 60000 / hr;

        document.getElementById('hero-hrv').textContent = Math.round(rr_ms * 0.05); // ~5% of RR as proxy

            chart: { labels: tLabels, data: pAtInlet, label: 'Aortic Pressure p(0,t) [mmHg] — MOC Solution' },
            lines: [
                { t: `1D FSI solver: MOC Lax-Wendroff, Nz=${Nz}, CFL dt=${(dt*1000).toFixed(2)} ms`, c: '' },
                { t: `Aorta: L=${(L_aorta*100).toFixed(0)} cm, r=${(r0*1000).toFixed(1)} mm, E=${p['fsi-E']} kPa`, c: '' },
                { t: `c₀ = ${c0.toFixed(2)} m/s, Z₀ = ${(Z0/1e6).toFixed(3)} MPa·s/m³`, c: 'ln-val' },
                { t: `Womersley α = ${alpha.toFixed(2)}`, c: '' },
                { t: `Q_peak = ${(Q_peak*1e6).toFixed(1)} mL/s, Q_mean = ${(Q_mean*1e6).toFixed(1)} mL/s`, c: '' },
                { t: `Terminal Windkessel Z_term = ${(Z_term/1e6).toFixed(1)} MPa·s/m³ (3×Z₀)`, c: '' },

            }

        }, 50);

const vizCanvas = document.getElementById('vascularViz');
const vizCtx = vizCanvas.getContext('2d');
let vizTime = 0;

function resizeViz() {
    const dpr = window.devicePixelRatio || 1;
    vizCanvas.width = vizCanvas.clientWidth * dpr;
    vizCanvas.height = vizCanvas.clientHeight * dpr;
    vizCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
}
resizeViz();
window.addEventListener('resize', resizeViz);

// Build tree from Murray's law: r_d = r_p × 2^(-1/3), angle from optimal branching
const VIZ_TREE = [];
(function buildVizTree(x, y, length, angle, r, depth) {
    if (depth > 8 || r < 0.5) return;
    const endX = x + length * Math.sin(angle);
    const endY = y - length * Math.cos(angle);
    VIZ_TREE.push({ x, y, endX, endY, depth, r, length });
    const r_d = r * Math.pow(2, -1/3); // Murray's law
    const branchAngle = 0.4 * Math.pow(r_d / r, 0.5); // optimal branching angle
    buildVizTree(endX, endY, length * 0.72, angle + branchAngle, r_d, depth + 1);
    buildVizTree(endX, endY, length * 0.72, angle - branchAngle, r_d, depth + 1);
})(0, 0, 1, 0, 8, 0); // normalized coordinates

function drawVascular() {
    const w = vizCanvas.clientWidth, h = vizCanvas.clientHeight;
    vizCtx.clearRect(0, 0, w, h);

    const bg = vizCtx.createRadialGradient(w*0.5, h*0.7, 0, w*0.5, h*0.5, w*0.7);
    bg.addColorStop(0, '#1a0a12');
    bg.addColorStop(1, '#0a0506');
    vizCtx.fillStyle = bg;
    vizCtx.fillRect(0, 0, w, h);

    const scale = h * 0.18;
    const ox = w / 2, oy = h;

    // Draw with pulse wave propagation
    for (const seg of VIZ_TREE) {
        // Pulse wave: arrives at time = distance/c₀, visualized as brightness
        const dist = seg.depth; // proxy for wave travel
        const c0_viz = 6; // m/s proxy
        const wave = Math.sin(vizTime * 4 - dist * 0.8) * 0.5 + 0.5;

        const r_int = Math.max(180 - seg.depth * 12, 120);
        const g_int = Math.round(24 + wave * 50);
        const b_int = Math.round(80 + wave * 30);
        const a = Math.max(0.3, 0.95 - seg.depth * 0.08);

        vizCtx.beginPath();
        vizCtx.moveTo(ox + seg.x * scale, oy + seg.y * scale);
        vizCtx.lineTo(ox + seg.endX * scale, oy + seg.endY * scale);
        vizCtx.strokeStyle = `rgba(${r_int},${g_int},${b_int},${a})`;
        vizCtx.lineWidth = Math.max(0.5, seg.r * 0.3);
        vizCtx.stroke();
    }

    vizTime += 0.016;
    requestAnimationFrame(drawVascular);
}
drawVascular();

    document.getElementById(id)?.scrollIntoView({ behavior: 'smooth' });
    document.getElementById('navLinks').classList.remove('open');

const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => { if (entry.isIntersecting) entry.target.classList.add('visible'); });
}, { threshold: 0.1 });
document.querySelectorAll('.reveal').forEach(el => observer.observe(el));

window.addEventListener('scroll', () => {
    document.querySelector('.nav').classList.toggle('scrolled', window.scrollY > 50);
});

// PLENUMNET INITIALIZATION — §24.4 Step 1
// On page load: establish connection via server proxy, fetch safety limits, ternary state
// Auth: Handled by server.js proxy — PLENUMNET_API_KEY injected server-side from Replit Secrets
// ================================================================
(async function initPlenumNET() {
    PlenumNET.log('Initializing PPTPro V2.3 → PlenumNET connection');
    PlenumNET.log(`Proxy: ${PlenumNET.PROXY_BASE}/* → ${PlenumNET.DIRECT_BASE}/api/v1/*`);
    PlenumNET.log('Auth: server-side injection via PLENUMNET_API_KEY secret');

    // Check if proxy server is running (keycheck endpoint)
    try {
        const kc = await fetch('/api/proxy/keycheck').then(r => r.json()).catch(() => null);
        if (kc) {
            PlenumNET.log(`API Key: ${kc.configured ? kc.prefix : 'NOT CONFIGURED — add PLENUMNET_API_KEY to Replit Secrets'}`, kc.configured ? 'api-ok' : 'api-err');
        }
    } catch (e) { /* proxy not running — that's fine for static serving */ }

    // §24.4 Step 1: Check PlenumNET status
    await PlenumNET.checkStatus();

    // If connected, log platform info
    if (PlenumNET.connected) {
        PlenumNET.log('§24.4 workflow active: module → coherence → advise → ternary');
    } else {
        PlenumNET.log('Client-side computation active. PlenumNET sync will retry on module execution.');
        // Retry connection every 30 seconds
        setInterval(async () => {
            if (!PlenumNET.connected) {
                PlenumNET.log('Retrying PlenumNET connection...');
                await PlenumNET.checkStatus();
            }
        }, 30000);
    }
})();
