/// A: Scalar HMAC compute (2 derive_key calls).
pub fn bench_hmac_scalar() {
    let key = ternary_math::sponge::derive_key(
        b"PlenumNET-HB-HMAC", b"key-material", 48,
    );
    let tag = ternary_math::sponge::derive_key(
        b"PlenumNET-HB-TAG",
        &[key.as_slice(), b"heartbeat-payload".as_slice()].concat(),
        27,

    black_box(tag);
}

// ═══════════════════════════════════════════════════════════════════════
// A/B: Heartbeat pipeline ×26 — The headline number
// ═══════════════════════════════════════════════════════════════════════

/// A: Scalar 26-neighbor heartbeat pipeline.
pub fn bench_heartbeat_26_scalar() {
    for i in 0..26u8 {
        let mut km = Vec::with_capacity(49);
        km.extend_from_slice(b"key-material");
        km.push(i);

        let key = ternary_math::sponge::derive_key(b"PlenumNET-HB-HMAC", &km, 48);
        let tag = ternary_math::sponge::derive_key(
            b"PlenumNET-HB-TAG",
            &[key.as_slice(), b"hb-payload".as_slice()].concat(),
            27,
        );
        let tag2 = ternary_math::sponge::derive_key(
            b"PlenumNET-HB-TAG",
            &[key.as_slice(), b"hb-payload".as_slice()].concat(),
            27,
        );
        let mut diff: u8 = 0;
        for j in 0..tag.len() { diff |= tag[j] ^ tag2[j]; }
        black_box(diff);
    }
}

// ═══════════════════════════════════════════════════════════════════════
// A/B: PT26-DSA sign — 2 sponge calls in the hot path
// ═══════════════════════════════════════════════════════════════════════

/// A: PT26-DSA sign with scalar sponge.
pub fn bench_pt26_sign_scalar() {
    let addr: [u8; 13] = [2, 1, 3, 2, 1, 3, 2, 1, 3, 2, 1, 3, 2];
    let dest: [u8; 13] = [3, 3, 1, 1, 3, 1, 3, 3, 1, 2, 1, 3, 2];
    let weights: [u32; 9] = [208, 2, 123, 26, 111, 196, 99, 220, 14];

    // Sponge 1: msg hash
    let msg_hash = ternary_math::sponge::derive_key(
        b"PT26-MSG", b"benchmark message", 48,
    );

    // Walk: pure GF(3), same for both
    let h = 9;
    let mut checksum: u32 = 0;
    for step in 0..h {
        checksum = (checksum + weights[step % 9] * (step as u32 + 1)) % 333;
    }

    // Sponge 2: binding
    let mut mat = Vec::with_capacity(80);
    mat.extend_from_slice(&addr);
    mat.extend_from_slice(&dest);
    mat.extend_from_slice(&(checksum as u16).to_le_bytes());
    mat.extend_from_slice(&msg_hash);
    let binding = ternary_math::sponge::derive_key(b"PT26-BIND", &mat, 48);
    black_box(binding);
}

/// B: PT26-DSA sign with packed sponge.
pub fn bench_pt26_sign_packed() {
    let addr: [u8; 13] = [2, 1, 3, 2, 1, 3, 2, 1, 3, 2, 1, 3, 2];
    let dest: [u8; 13] = [3, 3, 1, 1, 3, 1, 3, 3, 1, 2, 1, 3, 2];
    let weights: [u32; 9] = [208, 2, 123, 26, 111, 196, 99, 220, 14];

    // Sponge 1: msg hash (PACKED)
    let msg_hash = ternary_math::sponge_packed::derive_key_packed(
        b"PT26-MSG", b"benchmark message", 48,
    );

    // Walk: identical GF(3)
    let h = 9;
    let mut checksum: u32 = 0;
    for step in 0..h {
        checksum = (checksum + weights[step % 9] * (step as u32 + 1)) % 333;
    }

    // Sponge 2: binding (PACKED)
    let mut mat = Vec::with_capacity(80);
    mat.extend_from_slice(&addr);
    mat.extend_from_slice(&dest);
    mat.extend_from_slice(&(checksum as u16).to_le_bytes());
    mat.extend_from_slice(&msg_hash);
    let binding = ternary_math::sponge_packed::derive_key_packed(b"PT26-BIND", &mat, 48);
    black_box(binding);
}

// ═══════════════════════════════════════════════════════════════════════
// QUICK A/B SMOKE TEST
// ═══════════════════════════════════════════════════════════════════════

/// Run all A/B benchmarks once and print results.
/// Used for quick validation, not statistical measurement.
pub fn ab_smoke_test() -> Vec<(&'static str, u64)> {
    let entries = ab_benchmark_entries();
    let mut results = Vec::with_capacity(entries.len());
    for (name, _target, func) in &entries {
        let start = std::time::Instant::now();
        func();
        let elapsed = start.elapsed().as_nanos() as u64;
        results.push((*name, elapsed));
    }
    results
}
