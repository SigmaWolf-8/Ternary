// ═══════════════════════════════════════════════════════════════
// CORE TOKEN TYPES — Phases 1-3
// ═══════════════════════════════════════════════════════════════

export interface Capability {
  res: string;
  constraints: CapabilityConstraint[];
  exp: string; // BigInt nanosecond HPTP timestamp — string to avoid JS Number overflow (>2^53)
}

export interface CapabilityToken {
  sub: string;
  cap: Capability[];
  iat_hptp: string;
  iss: 'plenumnet.cap';
  jti: string;
  crv: string; // constraint registry version
  role?: string; // legacy fallback
}

export interface SignedCapabilityToken {
  token: CapabilityToken;
  signature: string;
  algorithm: 'TL-DSA';
}

export function createCapabilityToken(
  subject: string,
  capabilities: Capability[],
  issuedAtHptpNs: string,
  jti: string,
): CapabilityToken {
  return {
    sub: subject,
    cap: capabilities,
    iat_hptp: issuedAtHptpNs,
    iss: 'plenumnet.cap',
    jti,
    crv: '1.0',
  };
}

export function isCapabilityExpired(cap: Capability, hptpNs: string): boolean {
  return BigInt(hptpNs) >= BigInt(cap.exp);
}

export function findMatchingCapability(
  token: CapabilityToken,
  resource: string,
  hptpNs: string,
): Capability | null {
  return token.cap.find(c => c.res === resource && !isCapabilityExpired(c, hptpNs)) || null;
}
