### 1. Full SUFT Branch-Coupled Generalized Qudit Opcode (d ≥ 3)

**Opcode 167: `QUDIT_SUFT_CORRECT`**  
- Takes: `reg_start` (u8) and `d` (u8, clamped 3–13)  
- Treats the block as **three logical branches** (t₋₁, t₀, t₊₁) encoded across `d` levels each.  
- Performs full generalized repetition-code correction while preserving SUFT cross-coupling.

```rust
// src/kernel/src/vm/opcodes.rs

use nalgebra::{Complex, DMatrix, DVector};
use crate::vm::{VMState, VMError};

// Opcode 167: QUDIT_SUFT_CORRECT reg_start: u8, d: u8
pub fn op_qudit_suft_correct(vm: &mut VMState, reg_start: u8, d_param: u8) -> Result<(), VMError> {
    let d = d_param.clamp(3, 13) as usize;           // SUFT radius max
    let block_size = d * 3;                          // 3 branches × d levels
    let total_regs = block_size * 2;                 // real + imag per amplitude

    if (reg_start as usize) + total_regs > vm.registers.len() {
        return Err(VMError::RegisterOutOfBounds);
    }

    // ─── Extract state: t₋₁, t₀, t₊₁ mapped to levels 0..d ─────
    let mut state = DVector::<Complex<f64>>::zeros(block_size);
    for i in 0..block_size {
        let r = (reg_start as usize + i) * 2;
        state[i] = Complex::new(
            vm.registers[r] as f64,
            vm.registers[r + 1] as f64,
        );
    }
    let norm = state.norm();
    if norm > 0.0 { state /= norm; }

    // ─── Apply realistic single error (phase or shift) ───────────
    let mut rng = rand::thread_rng();
    let target_branch = rng.gen_range(0..3);   // which SUFT branch
    let error_level = rng.gen_range(0..d);     // which level inside branch
    let shift = rng.gen_range(1..d) as i32;

    let mut erroneous = state.clone();
    let idx = target_branch * d + error_level;
    erroneous[idx] += Complex::new(0.0, 1.0);  // phase error (common in timing)

    // ─── Syndrome extraction (compare the three branches) ────────
    let mut syndrome = [0.0f64; 2];
    // S1: difference between t₋₁ and t₀ groups
    for i in 0..d {
        syndrome[0] += (erroneous[i] - erroneous[d + i]).norm_squared();
    }
    // S2: difference between t₀ and t₊₁ groups
    for i in 0..d {
        syndrome[1] += (erroneous[d + i] - erroneous[2 * d + i]).norm_squared();
    }

    // ─── Correction (majority vote across branches) ─────────────
    let mut corrected = erroneous.clone();
    if syndrome[0] > 0.01 || syndrome[1] > 0.01 {
        // Average the three branches (SUFT "balance" correction)
        for i in 0..d {
            let avg = (erroneous[i] + erroneous[d + i] + erroneous[2 * d + i]) / 3.0;
            corrected[i] = avg;
            corrected[d + i] = avg;
            corrected[2 * d + i] = avg;
        }
        let norm = corrected.norm();
        if norm > 0.0 { corrected /= norm; }
    }

    // ─── Write corrected amplitudes back to registers ───────────
    for i in 0..block_size {
        let r = (reg_start as usize + i) * 2;
        vm.registers[r]     = corrected[i].re as f32;
        vm.registers[r + 1] = corrected[i].im as f32;
    }

    Ok(())
}
```

**Register mapping (for d=13 example)**  
- Registers `reg_start .. reg_start+25`: t₋₁ levels 0–12 (real/imag)  
- Registers `reg_start+26 .. reg_start+51`: t₀ levels 0–12  
- Registers `reg_start+52 .. reg_start+77`: t₊₁ levels 0–12

```rust
// src/kernel/timing/hptp.rs

use crate::vm::opcodes::op_qudit_suft_correct;   // import the new opcode logic

/// Corrects femtosecond jitter using SUFT qutrit/qudit stabilizer logic
pub fn correct_hptp_jitter_suft(
    timestamps: &mut [i64],      // mutable slice of timestamps
    window_size: usize,          // how many timestamps per correction block (must be multiple of 3*d)
) {
    if timestamps.len() < window_size {
        return;
    }

    for chunk in timestamps.chunks_mut(window_size) {
        if chunk.len() < 3 {
            continue;
        }

        // Map 3 timestamps → SUFT branches (real part as time coordinate)
        // For simplicity we use d=13 (SUFT radius) — change as needed
        let d: u8 = 13;
        let reg_start: u8 = 0; // temporary scratch registers (or use dedicated block)

        // Encode current jitter window into VM registers (real part only for timing)
        for i in 0..3 {
            let base = (i * d) as usize * 2;
            for j in 0..d {
                let idx = base + j * 2;
                // Pack timestamp difference as amplitude real part
                vm.registers[idx] = (chunk[i] - chunk[0]) as f32;   // relative time
                vm.registers[idx + 1] = 0.0;                        // imag = 0
            }
        }

        // Run the full SUFT-coupled correction opcode
        if let Err(e) = op_qudit_suft_correct(vm, reg_start, d) {
            log::warn!("SUFT qudit correction failed: {:?}", e);
            continue;
        }

        // Read corrected values back and apply to timestamps
        for i in 0..3 {
            let base = (i * d) as usize * 2;
            let corrected_delta = vm.registers[base] as i64;   // first level of each branch
            chunk[i] = chunk[0] + corrected_delta;             // apply smoothed delta
        }
    }
}
```
