#### 3.2: Update Generation Service (`server/services/api-key.service.ts`)

```ts
// In generate() – add optional bind param
async generate(
  owner: string,
  name: string,
  scopes: string[],
  expiresDays = 90,
  bindToHPTP = false,
  toleranceSeconds = 5
) {
  const rawKey = randomBytes(32).toString('base64url');
  const keyHash = await hash(rawKey, { type: argon2.argon2id });

  let hpptBoundAt: Date | null = null;
  if (bindToHPTP) {
    hpptBoundAt = await hptpService.getCorrectedTimestamp();  // your HPTP function – assume returns Date or ISO string
    if (typeof hpptBoundAt === 'string') hpptBoundAt = new Date(hpptBoundAt);
  }

  const [inserted] = await db.insert(apiKeys).values({
    keyHash,
    name,
    owner,
    scopes,
    expiresAt: new Date(Date.now() + expiresDays * 86400000),
    hpptBoundAt,
    hpptToleranceSeconds: toleranceSeconds,
  }).returning({ id: apiKeys.id });

  // Schedule rotation if desired (from Priority 1)
  if (inserted.id) await this.scheduleRotation(inserted.id);

  return { key: `plm_${rawKey}`, name, scopes, boundToHPTP: !!hpptBoundAt };
},
```

```ts
// Add to router
router.get('/stats', async (req, res) => {
  const ownerFilter = req.user?.isAdmin ? {} : { owner: req.user.id };  // admin sees all

  // Basic stats
  const totalKeys = await db.$count(apiKeys, { ...ownerFilter, revokedAt: null });
  const activeKeys = await db.$count(apiKeys, { ...ownerFilter, revokedAt: null, expiresAt: gt(apiKeys.expiresAt, new Date()) });

  // Usage last 7 days (group by day)
  const usageLast7d = await db
    .select({
      date: sql<string>`date_trunc('day', ${apiKeyLogs.timestamp})`.as('date'),
      count: sql<number>`count(*)`.as('count'),
    })
    .from(apiKeyLogs)
    .where(and(
      gte(apiKeyLogs.timestamp, sql`now() - interval '7 days'`),
      // join to api_keys if needed for owner filter
    ))
    .groupBy(sql`date_trunc('day', ${apiKeyLogs.timestamp})`)
    .orderBy(sql`date`);

  // Top endpoints
  const topEndpoints = await db
    .select({
      endpoint: apiKeyLogs.endpoint,
      count: sql<number>`count(*)`.as('count'),
    })
    .from(apiKeyLogs)
    .where(gte(apiKeyLogs.timestamp, sql`now() - interval '30 days'`))
    .groupBy(apiKeyLogs.endpoint)
    .orderBy(desc(sql`count(*)`))
    .limit(10);

  // Simple anomaly: >300% increase day-over-day
  const anomalies = []; // logic: compare consecutive days in usageLast7d

  res.json({ totalKeys, activeKeys, usageLast7d, topEndpoints, anomalies });
});
```

```ts
// In POST /api/keys/generate validation
const schema = z.object({
  name: z.string().min(3),
  scopes: z.array(z.string()).min(1),
  template: z.enum(Object.keys(SCOPE_TEMPLATES) as [string, ...string[]]).optional(),
```
