// libternary/utils/matrix-utils.ts
import {
  SATURNIAN_MATRIX,
  MAGIC_CONSTANT,
  TERNARY_BALANCE_CENTER
} from '../../shared/saturnian-blueprint';

/**
 * Returns a flattened version of the Saturnian matrix (useful for seeding or indexing)
 */
export function getSaturnianFlattened(): readonly number[] {
  return SATURNIAN_MATRIX.flat();
}

/**
 * Returns one of the three cyclic permutations of the matrix rows
 * @param shift 0, 1, or 2 – corresponds to ternary gauge-like rotation
 */
export function getSaturnianCyclic(shift: 0 | 1 | 2 = 0): readonly (readonly number[])[] {
  const rows = SATURNIAN_MATRIX;
  return [
    rows[(0 + shift) % 3],
    rows[(1 + shift) % 3],
    rows[(2 + shift) % 3]
  ] as const;
}

/**
 * Validates that a 3×3 matrix has the Saturnian magic property (sum = 333)
 * Pure function – for tests or compile-time checks
 */
export function isSaturnianMagic(matrix: readonly (readonly number[])[]): boolean {
  if (matrix.length !== 3 || matrix.some(row => row.length !== 3)) return false;
  const sums = [
    matrix[0][0] + matrix[0][1] + matrix[0][2],
    matrix[1][0] + matrix[1][1] + matrix[1][2],
    matrix[2][0] + matrix[2][1] + matrix[2][2],
    // columns
    matrix[0][0] + matrix[1][0] + matrix[2][0],
    matrix[0][1] + matrix[1][1] + matrix[2][1],
    matrix[0][2] + matrix[1][2] + matrix[2][2],
    // diagonals
    matrix[0][0] + matrix[1][1] + matrix[2][2],
    matrix[0][2] + matrix[1][1] + matrix[2][0]
  ];
  return sums.every(sum => sum === MAGIC_CONSTANT);
}

// Quick static check (can be used in tests)
if (!isSaturnianMagic(SATURNIAN_MATRIX)) {
  throw new Error("Saturnian blueprint matrix is invalid");
