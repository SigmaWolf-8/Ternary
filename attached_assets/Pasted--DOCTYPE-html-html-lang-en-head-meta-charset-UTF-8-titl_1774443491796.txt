<script>
// Toggle any layer by id
function toggle(layerId, visible) {
  const g = document.getElementById(layerId);
  if (g) g.style.display = visible ? 'inline' : 'none';
}

// Reset everything
function resetAll() {
  document.querySelectorAll('g[id]').forEach(g => g.style.display = 'inline');
  document.querySelectorAll('#controls input[type="checkbox"]').forEach(cb => cb.checked = true);
}
</script>
