pub const FULL_CIRCLE_DEG: f64 = 364.0;
pub const RADIAN_DEG: f64 = 13.0;
pub const TWO_PI_TERNARY: f64 = 28.0;
pub const TAU_TRIBONACCI: f64 = 1.839286755214161;

#[inline]
pub fn std_deg_to_ternary_deg(std_deg: f64) -> f64 {
    std_deg * (FULL_CIRCLE_DEG / 360.0)
}

#[inline]
pub fn ternary_deg_to_resonance(tern_deg: f64) -> f64 {
    let mod13 = tern_deg % 13.0;
    let dist = mod13.min(13.0 - mod13);
    1.0 - (dist / 6.5)
}

### Rust – Minimal BalancedFloat bridge (ternary-math/src/bfloat.rs)
```rust
use crate::tribonacci::TritVec;

#[derive(Clone, Debug)]
pub struct BFloat {
    pub sign: i8,
    pub integer: TritVec,
    pub fraction: Vec<u8>,
    pub exponent: i32,
}

impl BFloat {
    pub fn from_f64(x: f64, prec: usize) -> Self {
        // Simplified MVP – real impl needs full carry logic
        let sign = if x < 0.0 { -1 } else { 1 };
        let abs = x.abs();
        let int_part = abs.floor() as u64;
        let mut trits = vec![];
        let mut n = int_part;
        while n > 0 {
            trits.push((n % 3) as u8);
            n /= 3;
        }
        trits.reverse();
        let mut frac = vec![];
        let mut f = abs - int_part as f64;
        for _ in 0..prec {
            f *= 3.0;
            frac.push(f.floor() as u8);
            f -= frac.last().unwrap() as f64;
        }
        BFloat { sign, integer: TritVec::new(trits), fraction: frac, exponent: 0 }
    }

    pub fn to_f64(&self) -> f64 {
        let mut val = 0.0_f64;
        let mut p = 1.0;
        for &t in &self.integer.trits {
            val += t as f64 * p;
            p *= 3.0;
        }
        let mut fp = 1.0 / 3.0;
        for &t in &self.fraction {
            val += t as f64 * fp;
            fp /= 3.0;
        }
        val * self.sign as f64
    }

    pub fn sin(&self) -> Self {
        let x = self.to_f64();
        BFloat::from_f64(x.sin(), 60)
    }
}
```

### JavaScript – Frontend API call example (Astrology App)
```js
async function fetchTernaryEphemeris(planet, jd, lat = 53.54, lon = -113.49) {
  try {
    const response = await fetch('https://plenumnet.replit.app/api/v1/ephemeris', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        planet,
        jd,
        include_resonance: true,
        observer: { lat, lon, alt: 0 }
      })
    });

    if (!response.ok) throw new Error(`HTTP ${response.status}`);
    const data = await response.json();

    console.log(`${planet.toUpperCase()} @ JD ${jd}:`);
    console.log(`  Ternary longitude: ${data.ternary_longitude.toFixed(3)}°`);
    console.log(`  Resonance:         ${(data.resonance * 100).toFixed(1)}%`);

    return data;
  } catch (err) {
    console.error('API call failed:', err);
    return null;
  }
}

// Example usage in your chart init
fetchTernaryEphemeris("sun", 2460740.0);      // current approx JD
fetchTernaryEphemeris("moon", 2460740.0);
```
