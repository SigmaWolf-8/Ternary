# TL-DSA Optimization — NTT + Expanded Keys
# File: src/kernel/src/crypto/tl_dsa.rs
# Single file. No changes to ternary_lattice.rs.
# Two optimizations in one pass.

## Current State

TL-DSA-87 roundtrip: **1,441 µs** (keygen 723, sign 383, verify 357).

Two bottlenecks:
1. All polynomial multiplies use schoolbook O(n²) — NTT O(n log n) exists but isn't called
2. Matrix A is re-expanded from seed in keygen, sign, AND verify (56 sponge hashes each time)

## Target

Sub-800 µs roundtrip for DSA-87. Two changes:
- **Schoolbook → NTT** for all polynomial multiplication (~30-50% speedup)
- **Pre-NTT expanded keys** — compute NTT(A) once in keygen, store in keys, never re-expand

## Parameters (for reference)

| Variant | n | k | l | k×l poly muls per A×v |
|---------|---|---|---|-----------------------|
| TL-DSA-44 | 256 | 4 | 4 | 16 |
| TL-DSA-65 | 256 | 6 | 5 | 30 |
| TL-DSA-87 | 256 | 8 | 7 | 56 |

## Change 3: Modify `keygen`

Replace the entire `keygen` function body. Current code expands matrix A, multiplies
with schoolbook, and stores seed only. New code expands A, converts to NTT, multiplies
with NTT, and stores the NTT form in both keys.

Find:
```rust
pub fn keygen(variant: TlDsaVariant, seed: &[i8]) -> CryptoResult<(TlDsaPublicKey, TlDsaSecretKey)> {
    let params = variant.params();
    let n = params.n;
    let k = params.k;
    let l = params.l;

    let rho = dsa_hash(DOMAIN_MATRIX_EXPAND, &[seed, &[0i8]], 243);
    let sigma = dsa_hash(DOMAIN_SECRET_SAMPLE, &[seed, &[1i8]], 243);
    let signing_seed = dsa_hash(DOMAIN_SIGNING_SEED, &[seed, &[0i8, 1, -1]], 243);

    let matrix_a = expand_matrix_a(&rho, k, l, n);

    let secret_s1 = sample_noise_vec(&sigma, l, n, 0, params.eta);
    let secret_s2 = sample_noise_vec(&sigma, k, n, l as u16, params.eta);

    let public_t = matrix_a.mul_vec(&secret_s1)?;

    let pk = TlDsaPublicKey {
        variant,
        matrix_a_seed: rho.clone(),
        public_t: public_t.clone(),
    };

    let sk = TlDsaSecretKey {
        variant,
        matrix_a_seed: rho,
        secret_s1,
        secret_s2,
        public_t,
        signing_seed,
    };

    Ok((pk, sk))
}
```

Replace with:
```rust
pub fn keygen(variant: TlDsaVariant, seed: &[i8]) -> CryptoResult<(TlDsaPublicKey, TlDsaSecretKey)> {
    let params = variant.params();
    let n = params.n;
    let k = params.k;
    let l = params.l;

    let rho = dsa_hash(DOMAIN_MATRIX_EXPAND, &[seed, &[0i8]], 243);
    let sigma = dsa_hash(DOMAIN_SECRET_SAMPLE, &[seed, &[1i8]], 243);
    let signing_seed = dsa_hash(DOMAIN_SIGNING_SEED, &[seed, &[0i8, 1, -1]], 243);

    // Expand matrix A from seed, then convert to NTT form ONCE.
    // NTT form is stored in keys — sign and verify never re-expand.
    let matrix_a = expand_matrix_a(&rho, k, l, n);
    let ntt_a = matrix_a.to_ntt();

    let secret_s1 = sample_noise_vec(&sigma, l, n, 0, params.eta);
    let secret_s2 = sample_noise_vec(&sigma, k, n, l as u16, params.eta);

    // Use NTT matrix-vector multiply instead of schoolbook
    let public_t = ntt_a.mul_vec(&secret_s1)?;

    let pk = TlDsaPublicKey {
        variant,
        matrix_a_seed: rho.clone(),
        ntt_a: ntt_a.clone(),  // NOTE: NttMatrix must derive Clone, or clone manually
        public_t: public_t.clone(),
    };

    let sk = TlDsaSecretKey {
        variant,
        matrix_a_seed: rho,
        ntt_a,
        secret_s1,
        secret_s2,
        public_t,
        signing_seed,
    };

    Ok((pk, sk))
}
```

**NOTE on Clone:** `NttMatrix` must implement `Clone`. Check if it already derives
Clone in `ternary_lattice.rs`. If not, add `#[derive(Clone)]` to the `NttMatrix`
struct definition. This is the ONE change needed in `ternary_lattice.rs`.

## Change 5: Modify `verify`

Same pattern — remove matrix expansion, use pre-stored NTT form from public key.

Find:
```rust
pub fn verify(pk: &TlDsaPublicKey, message: &[i8], sig: &TlDsaSignature) -> CryptoResult<bool> {
    let params = pk.variant.params();
    let n = params.n;
    let k = params.k;
    let l = params.l;

    if sig.z.polys.len() != l {
        return Ok(false);
    }

    for poly in &sig.z.polys {
        if poly.l_infinity_norm() > 1 {
            return Ok(false);
        }
    }

    let matrix_a = expand_matrix_a(&pk.matrix_a_seed, k, l, n);

    let c = sample_challenge(&sig.challenge_hash, n, params.tau);

    let az = matrix_a.mul_vec(&sig.z)?;

    let mut ct_polys = Vec::with_capacity(k);
    for i in 0..k {
        let cti = c.ring_mul(&pk.public_t.polys[i])?;
        ct_polys.push(cti);
    }
```

Replace with:
```rust
pub fn verify(pk: &TlDsaPublicKey, message: &[i8], sig: &TlDsaSignature) -> CryptoResult<bool> {
    let params = pk.variant.params();
    let _n = params.n;
    let k = params.k;
    let l = params.l;

    if sig.z.polys.len() != l {
        return Ok(false);
    }

    for poly in &sig.z.polys {
        if poly.l_infinity_norm() > 1 {
            return Ok(false);
        }
    }

    // Matrix A is pre-computed in NTT form — no expansion needed
    let c = sample_challenge(&sig.challenge_hash, params.n, params.tau);

    let az = pk.ntt_a.mul_vec(&sig.z)?;

    let mut ct_polys = Vec::with_capacity(k);
    for i in 0..k {
        let cti = ntt_ring_mul(&c, &pk.public_t.polys[i], NTT_Q);
        ct_polys.push(cti);
    }
```

The rest of the verify function (w_prime, mu, expected_hash comparison) stays unchanged.
